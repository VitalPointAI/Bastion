# Phase 05 Plan 16: Step Content Area Fix

**Gap:** Clicking a step doesn't show step details. The StepNavigator shows steps but PlanningDashboard doesn't render step-specific content.

## Root Cause

`PlanningDashboard.tsx` (lines 169-196) only renders `StepNavigator` in the main area. There's no conditional rendering of step-specific components (COAList, ApprovalPanel, ROEPanel, DocumentExport) based on `workflowState.context.currentStep`.

## Solution

Add a step content area to PlanningDashboard that renders different components based on the currently selected step:

| Step | Component(s) |
|------|--------------|
| planning_initiation | Info text |
| mission_analysis | Info text |
| coa_development | COAList + COAEditor |
| coa_analysis | COAList (read-only) |
| coa_comparison | COAList (read-only) |
| coa_approval | ApprovalPanel |
| plan_development | ROEPanel + DocumentExport |
| plan_approval | ApprovalPanel |

```typescript
const renderStepContent = () => {
  const currentStep = workflowState?.context.currentStep;
  if (!currentStep || !selectedPlanId || !selectedPlan) {
    return (
      <div className="step-content-placeholder">
        Select a step to view details
      </div>
    );
  }

  switch (currentStep) {
    case 'planning_initiation':
      return (
        <div className="step-content-info">
          <h4>Planning Initiation</h4>
          <p>Receive mission, analyze higher headquarters guidance, and issue warning order.</p>
          <ul>
            <li>Review strategic objectives linked to this plan</li>
            <li>Identify initial timeline constraints</li>
            <li>Prepare warning order for subordinate units</li>
          </ul>
        </div>
      );

    case 'mission_analysis':
      return (
        <div className="step-content-info">
          <h4>Mission Analysis</h4>
          <p>Analyze the mission to identify tasks, constraints, and planning considerations.</p>
          <ul>
            <li>Identify specified, implied, and essential tasks</li>
            <li>Determine constraints and limitations</li>
            <li>Develop initial commander's intent</li>
          </ul>
        </div>
      );

    case 'coa_development':
    case 'coa_analysis':
    case 'coa_comparison':
      return (
        <>
          <COAList
            planId={selectedPlanId}
            coas={coas}
            onCOAsChange={handleCOAsChange}
            onEditCOA={handleEditCOA}
          />
          {editingCOA !== undefined && (

              coa={editingCOA}

          planId={selectedPlanId}
          workflowState={workflowState}
          userDID={userDID}
          isCommander={true} // TODO: Check actual role

      ) : null;

    case 'plan_development':
      return (
        <>
          <ROEPanel
            checkResult={roeCheckResult}
            isCommander={true} // TODO: Check actual role
            onRequestOverride={handleROEOverride}
          />
          <DocumentExport
            planId={selectedPlanId}
            planName={selectedPlan.name}

          planId={selectedPlanId}
```

```typescript
export async function checkROE(planId: string): Promise<ROECheckResult> {
  const response = await fetch(`/api/planning/plans/${planId}/roe/check`);
  if (!response.ok) {
    // Return empty result if no ROE configured yet
    return { approved: true, violations: [], warnings: [], requiresOverride: false };
  }
  return response.json();
}

export async function requestROEOverride(
  planId: string,
  justification: string,
  commanderDID: string
): Promise<void> {
  const response = await fetch(`/api/planning/plans/${planId}/roe/override`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ justification, commanderDID }),
  });
  if (!response.ok) {
    throw new Error('Failed to request ROE override');
  }
}
```
