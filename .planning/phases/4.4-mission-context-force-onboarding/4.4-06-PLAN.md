// backend/src/resources/service.ts
import { ResourceStore } from './store.js';
import {
  Resource, ResourceCategory, Personnel, Consumable,
  ResourceStatus, ResourceWithCategory
} from './schemas.js';

// Valid status values per CONTEXT.md
const VALID_STATUSES = ['FMC', 'PMC', 'NMC'] as const;
type StatusType = typeof VALID_STATUSES[number];

export class ResourceService {
  private store: ResourceStore;

  constructor() {
    this.store = new ResourceStore();
  }

  async initialize(): Promise<void> {
    await this.store.ensureTablesExist();
    await this.seedDefaultCategories();
  }

  // --- Categories ---

  async createCategory(data: {
    name: string;
    parentId?: string;
    description?: string;
  }): Promise<ResourceCategory> {
    // Validate parent exists if specified
    if (data.parentId) {
      const parent = await this.store.getCategory(data.parentId);
      if (!parent) throw new Error('Parent category not found');
    }
    return this.store.createCategory(data);
  }

  async getCategories(parentId?: string): Promise<ResourceCategory[]> {
    return this.store.getCategories(parentId);
  }

  async getCategoryTree(): Promise<CategoryNode[]> {
    // Return hierarchical tree of all categories
    const all = await this.store.getAllCategories();
    return this.buildCategoryTree(all);
  }

  private buildCategoryTree(categories: ResourceCategory[]): CategoryNode[] {
    const nodeMap = new Map<string, CategoryNode>();
    const roots: CategoryNode[] = [];

    // Create nodes
    for (const cat of categories) {
      nodeMap.set(cat.id, { ...cat, children: [] });
    }

    // Build tree
    for (const cat of categories) {
      const node = nodeMap.get(cat.id)!;
      if (cat.parentId) {
        const parent = nodeMap.get(cat.parentId);
        if (parent) parent.children.push(node);
      } else {
        roots.push(node);
      }
    }

    return roots;
  }

  private async seedDefaultCategories(): Promise<void> {
    // Check if categories exist
    const existing = await this.store.getAllCategories();
    if (existing.length > 0) return;

    // Seed military equipment taxonomy
    const root = await this.store.createCategory({ name: 'Equipment' });
    await this.store.createCategory({ name: 'Weapons Systems', parentId: root.id });
    await this.store.createCategory({ name: 'Vehicles', parentId: root.id });
    await this.store.createCategory({ name: 'Aircraft', parentId: root.id });
    await this.store.createCategory({ name: 'Vessels', parentId: root.id });
    await this.store.createCategory({ name: 'Communications', parentId: root.id });
    await this.store.createCategory({ name: 'Sensors', parentId: root.id });

    const consumables = await this.store.createCategory({ name: 'Consumables' });
    await this.store.createCategory({ name: 'Ammunition', parentId: consumables.id });
    await this.store.createCategory({ name: 'Fuel', parentId: consumables.id });
    await this.store.createCategory({ name: 'Medical Supplies', parentId: consumables.id });
    await this.store.createCategory({ name: 'Rations', parentId: consumables.id });
  }

  // --- Resources ---

  async createResource(data: {
    missionId: string;
    categoryId?: string;
    name: string;
    serialNumber?: string;
    sidc?: string;
    specifications?: Record<string, any>;
    status?: string;
    statusReason?: string;
    location?: GeoJSON.Point;
  }): Promise<Resource> {
    // Validate status if provided
    if (data.status && !VALID_STATUSES.includes(data.status as StatusType)) {
      throw new Error(`Invalid status: ${data.status}. Must be FMC, PMC, or NMC`);
    }

    // Validate category if provided
    if (data.categoryId) {
      const cat = await this.store.getCategory(data.categoryId);
      if (!cat) throw new Error('Category not found');
    }

    return this.store.createResource({
      ...data,
      status: (data.status || 'FMC') as ResourceStatus
    });
  }

  async getResource(id: string): Promise<ResourceWithCategory | null> {
    return this.store.getResource(id);
  }

  async listResources(missionId: string, filters?: {
    categoryId?: string;
    status?: string;
    search?: string;
  }): Promise<ResourceWithCategory[]> {
    return this.store.listResources(missionId, filters);
  }

  async updateResource(id: string, data: Partial<Resource>): Promise<Resource> {
    if (data.status && !VALID_STATUSES.includes(data.status as StatusType)) {
      throw new Error(`Invalid status: ${data.status}`);
    }
    return this.store.updateResource(id, data);
  }

  async updateResourceStatus(id: string, status: string, reason?: string): Promise<Resource> {
    if (!VALID_STATUSES.includes(status as StatusType)) {
      throw new Error(`Invalid status: ${status}`);
    }
    return this.store.updateResource(id, { status: status as ResourceStatus, statusReason: reason });
  }

  async updateResourceLocation(id: string, location: GeoJSON.Point): Promise<Resource> {
    return this.store.updateLocation(id, location);
  }

  async deleteResource(id: string): Promise<void> {
    await this.store.deleteResource(id);
  }

  // --- Bulk Import ---

  async bulkImportResources(missionId: string, rows: BulkResourceRow[]): Promise<{
    imported: number;
    failed: number;
    errors: { row: number; error: string }[];
  }> {
    const errors: { row: number; error: string }[] = [];
    let imported = 0;
    let failed = 0;

    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      try {
        // Map CSV columns to resource fields
        await this.store.createResource({
          missionId,
          name: row.name,
          categoryId: row.categoryId,
          serialNumber: row.serialNumber,
          sidc: row.sidc,
          status: (row.status || 'FMC') as ResourceStatus,
          statusReason: row.statusReason,
          specifications: row.specifications ? JSON.parse(row.specifications) : undefined,
          location: row.latitude && row.longitude ? {
            type: 'Point',
            coordinates: [parseFloat(row.longitude), parseFloat(row.latitude)]
          } : undefined
        });
        imported++;
      } catch (error) {
        failed++;
        errors.push({ row: i + 1, error: (error as Error).message });
      }
    }

    return { imported, failed, errors };
  }

  // --- Personnel ---

  async createPersonnel(data: {
    missionId: string;
    did?: string;
    name: string;
    rank?: string;
    specialty?: string;
    unitId?: string;
    readinessStatus?: string;
    readinessReason?: string;
  }): Promise<Personnel> {
    // Validate unit exists if specified
    if (data.unitId) {
      // Would check via CommandStore, for now just pass through
    }
    return this.store.createPersonnel(data);
  }

  async getPersonnel(id: string): Promise<Personnel | null> {
    return this.store.getPersonnel(id);
  }

  async listPersonnel(missionId: string, filters?: {
    unitId?: string;
    readinessStatus?: string;
    specialty?: string;
  }): Promise<Personnel[]> {
    return this.store.listPersonnel(missionId, filters);
  }

  async updatePersonnel(id: string, data: Partial<Personnel>): Promise<Personnel> {
    return this.store.updatePersonnel(id, data);
  }

  async updateReadiness(id: string, status: string, reason?: string): Promise<Personnel> {
    return this.store.updatePersonnel(id, { readinessStatus: status, readinessReason: reason });
  }

  async bulkImportPersonnel(missionId: string, rows: BulkPersonnelRow[]): Promise<{
    imported: number;
    failed: number;
    errors: { row: number; error: string }[];
  }> {
    const errors: { row: number; error: string }[] = [];
    let imported = 0;
    let failed = 0;

    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      try {
        await this.store.createPersonnel({
          missionId,
          name: row.name,
          did: row.did,
          rank: row.rank,
          specialty: row.specialty,
          unitId: row.unitId,
          readinessStatus: row.readinessStatus || 'ready'
        });
        imported++;
      } catch (error) {
        failed++;
        errors.push({ row: i + 1, error: (error as Error).message });
      }
    }

    return { imported, failed, errors };
  }

  // --- Consumables ---

  async createConsumable(data: {
    missionId: string;
    categoryId?: string;
    name: string;
    supplyPointId?: string;
    quantity: number;
    unit?: string;
    reorderLevel?: number;
  }): Promise<Consumable> {
    return this.store.createConsumable(data);
  }

  async getConsumable(id: string): Promise<Consumable | null> {
    return this.store.getConsumable(id);
  }

  async listConsumables(missionId: string, filters?: {
    categoryId?: string;
    supplyPointId?: string;
    belowReorderLevel?: boolean;
  }): Promise<Consumable[]> {
    return this.store.listConsumables(missionId, filters);
  }

  async updateConsumableQuantity(id: string, quantity: number): Promise<Consumable> {
    return this.store.updateConsumable(id, { quantity });
  }

  async getResourceStats(missionId: string): Promise<{
    totalResources: number;
    byStatus: Record<string, number>;
    byCategory: Record<string, number>;
    personnelCount: number;
    consumableLowStock: number;
  }> {
    return this.store.getResourceStats(missionId);
  }
}

// Lazy singleton
let resourceService: ResourceService | null = null;

export async function getResourceService(): Promise<ResourceService> {
  if (!resourceService) {

  }
  return resourceService;
}

import { Router, Request, Response } from 'express';
import { getResourceService } from '../resources/service.js';

export const resourceRouter = Router();

// --- Categories ---

// GET /api/resources/categories
resourceRouter.get('/categories', async (req: Request, res: Response) => {
  try {
    const service = await getResourceService();
    const format = req.query.format as string;

    if (format === 'tree') {
      const tree = await service.getCategoryTree();
      res.json({ categories: tree });
    } else {
      const parentId = req.query.parentId as string;
      const categories = await service.getCategories(parentId);
      res.json({ categories });
    }
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// POST /api/resources/categories
resourceRouter.post('/categories', async (req: Request, res: Response) => {
  try {
    const service = await getResourceService();
    const category = await service.createCategory(req.body);
    res.status(201).json(category);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// --- Resources ---

// GET /api/missions/:missionId/resources
resourceRouter.get('/missions/:missionId/resources', async (req: Request, res: Response) => {
  try {
    const service = await getResourceService();
    const resources = await service.listResources(req.params.missionId as string, {
      categoryId: req.query.categoryId as string,
      status: req.query.status as string,
      search: req.query.search as string
    });
    res.json({ resources });
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// GET /api/resources/:id
resourceRouter.get('/resources/:id', async (req: Request, res: Response) => {
  try {
    const service = await getResourceService();
    const resource = await service.getResource(req.params.id as string);
    if (!resource) {
      return res.status(404).json({ error: 'Resource not found' });
    }
    res.json(resource);
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

    const service = await getResourceService();

      missionId: req.params.missionId as string,

  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

resourceRouter.patch('/resources/:id/location', async (req: Request, res: Response) => {
