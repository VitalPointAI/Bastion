// backend/src/command/service.ts
import { CommandStore } from './store.js';
import { Unit, CommandRelationship, JointResponsibility, HierarchyNode } from './schemas.js';

// Valid relationship types per CONTEXT.md
const RELATIONSHIP_TYPES = ['OPCON', 'TACON', 'ADCON', 'COCOM', 'DS', 'GS', 'GSR', 'R'] as const;
type RelationshipType = typeof RELATIONSHIP_TYPES[number];

// Joint responsibilities
const JOINT_RESPONSIBILITIES = ['JTF_COMD', 'JFLCC', 'JFMCC', 'JFACC', 'JFSOCC', 'JFMCC'] as const;

export class CommandService {
  private store: CommandStore;

  constructor() {
    this.store = new CommandStore();
  }

  async initialize(): Promise<void> {
    await this.store.ensureTablesExist();
  }

  // --- Units ---

  async createUnit(data: {
    missionId: string;
    name: string;
    unitType?: string;
    sidc?: string;
    parentDID?: string;
  }): Promise<Unit> {
    // Validate SIDC format if provided (basic check - 15-20 chars)
    if (data.sidc && (data.sidc.length < 10 || data.sidc.length > 30)) {
      throw new Error('Invalid SIDC format');
    }
    return this.store.createUnit(data);
  }

  async getUnit(id: string): Promise<Unit | null> {
    return this.store.getUnit(id);
  }

  async listUnits(missionId: string): Promise<Unit[]> {
    return this.store.listUnits(missionId);
  }

  async updateUnit(id: string, data: Partial<Unit>): Promise<Unit> {
    return this.store.updateUnit(id, data);
  }

  async deleteUnit(id: string): Promise<void> {
    // Check if unit has relationships
    const unit = await this.store.getUnit(id);
    if (!unit) throw new Error('Unit not found');

    // Delete all relationships first
    await this.store.deleteRelationshipsForUnit(id);
    await this.store.deleteUnit(id);
  }

  // --- Command Relationships ---

  async createRelationship(data: {
    missionId: string;
    superiorUnitId: string;
    subordinateUnitId: string;
    relationshipType: string;
    effectiveFrom?: Date;
    effectiveTo?: Date;
    notes?: string;
  }): Promise<{ relationship: CommandRelationship; warnings: string[] }> {
    const warnings: string[] = [];

    // Validate relationship type
    if (!RELATIONSHIP_TYPES.includes(data.relationshipType as RelationshipType)) {
      throw new Error(`Invalid relationship type: ${data.relationshipType}`);
    }

    // Validate units exist
    const superior = await this.store.getUnit(data.superiorUnitId);
    const subordinate = await this.store.getUnit(data.subordinateUnitId);
    if (!superior) throw new Error('Superior unit not found');
    if (!subordinate) throw new Error('Subordinate unit not found');

    // Check for self-reference
    if (data.superiorUnitId === data.subordinateUnitId) {
      throw new Error('Unit cannot be superior and subordinate to itself');
    }

    // Check for cycles (would this create a loop?)
    const wouldCycle = await this.detectCycleIfAdded(
      data.missionId,
      data.superiorUnitId,
      data.subordinateUnitId
    );

    if (wouldCycle) {
      // Per CONTEXT.md: advisory warning, allow override
      warnings.push('This relationship creates a circular command structure. Proceeding anyway per user override.');
    }

    // Check for duplicate
    const existing = await this.store.getRelationship(
      data.missionId,
      data.superiorUnitId,
      data.subordinateUnitId,
      data.relationshipType
    );
    if (existing) {
      throw new Error('Relationship already exists');
    }

    const relationship = await this.store.createRelationship(data);
    return { relationship, warnings };
  }

  async getRelationships(missionId: string): Promise<CommandRelationship[]> {
    return this.store.getRelationships(missionId);
  }

  async updateRelationship(id: string, data: Partial<CommandRelationship>): Promise<CommandRelationship> {
    return this.store.updateRelationship(id, data);
  }

  async deleteRelationship(id: string): Promise<void> {
    await this.store.deleteRelationship(id);
  }

  // --- Hierarchy ---

  async getHierarchy(missionId: string): Promise<HierarchyNode[]> {
    // Returns tree structure with depth and path
    return this.store.getHierarchy(missionId);
  }

  async getHierarchyAsTree(missionId: string): Promise<TreeNode[]> {
    // Transform flat hierarchy into nested tree for frontend
    const flat = await this.store.getHierarchy(missionId);
    return this.buildTree(flat);
  }

  private buildTree(nodes: HierarchyNode[]): TreeNode[] {
    const nodeMap = new Map<string, TreeNode>();
    const roots: TreeNode[] = [];

    // Create tree nodes
    for (const node of nodes) {
      nodeMap.set(node.id, {
        id: node.id,
        name: node.name,
        unitType: node.unitType,
        sidc: node.sidc,
        depth: node.depth,
        children: [],
        relationships: []
      });
    }

    // Build parent-child relationships
    for (const node of nodes) {
      const treeNode = nodeMap.get(node.id)!;
      if (node.depth === 0) {
        roots.push(treeNode);
      } else {
        // Find parent from path
        const parentId = node.path[node.path.length - 2];
        const parent = nodeMap.get(parentId);
        if (parent) {
          parent.children.push(treeNode);
        }
      }
    }

    return roots;
  }

  // --- Joint Responsibilities ---

  async assignResponsibility(unitId: string, responsibility: string): Promise<JointResponsibility> {
    if (!JOINT_RESPONSIBILITIES.includes(responsibility as any)) {
      throw new Error(`Invalid joint responsibility: ${responsibility}`);
    }
    return this.store.assignResponsibility(unitId, responsibility);
  }

  async getResponsibilities(unitId: string): Promise<JointResponsibility[]> {
    return this.store.getResponsibilities(unitId);
  }

  async removeResponsibility(unitId: string, responsibility: string): Promise<void> {
    await this.store.removeResponsibility(unitId, responsibility);
  }

  async getMultiHattedPositions(missionId: string): Promise<{ unit: Unit; responsibilities: string[] }[]> {
    return this.store.getMultiHattedPositions(missionId);
  }

  // --- Cycle Detection ---

  private async detectCycleIfAdded(
    missionId: string,
    superiorId: string,
    subordinateId: string
  ): Promise<boolean> {
    // Use DFS to check if subordinate can reach superior through existing relationships
    const relationships = await this.store.getRelationships(missionId);

    const visited = new Set<string>();
    const stack = [subordinateId];

    while (stack.length > 0) {
      const current = stack.pop()!;
      if (current === superiorId) {
        return true; // Found cycle
      }
      if (visited.has(current)) continue;
      visited.add(current);

      // Find relationships where current is superior (looking for subordinates)
      for (const rel of relationships) {
        if (rel.superiorUnitId === current) {
          stack.push(rel.subordinateUnitId);
        }
      }
    }

    return false;
  }

  async validateHierarchy(missionId: string): Promise<{ valid: boolean; warnings: string[] }> {
    const warnings: string[] = [];

    // Check for orphan units (no relationships)
    const units = await this.store.listUnits(missionId);
    const relationships = await this.store.getRelationships(missionId);

    const connectedUnits = new Set<string>();
    for (const rel of relationships) {
      connectedUnits.add(rel.superiorUnitId);
      connectedUnits.add(rel.subordinateUnitId);
    }

    const orphans = units.filter(u => !connectedUnits.has(u.id));
    if (orphans.length > 0) {
      warnings.push(`${orphans.length} units have no command relationships: ${orphans.map(u => u.name).join(', ')}`);
    }

    // Check for cycles
    const hasCycles = await this.store.detectCycles(missionId);
    if (hasCycles) {
      warnings.push('Circular command relationships detected');
    }

    return { valid: warnings.length === 0, warnings };
  }
}

// Lazy singleton
let commandService: CommandService | null = null;

export async function getCommandService(): Promise<CommandService> {
  if (!commandService) {
    commandService = new CommandService();
    await commandService.initialize();
  }
  return commandService;
}

interface TreeNode {
  id: string;
  name: string;
  unitType?: string;
  sidc?: string;
  depth: number;
  children: TreeNode[];
  relationships: string[];
}

// backend/src/api/command.ts
import { Router, Request, Response } from 'express';
import { getCommandService } from '../command/service.js';

export const commandRouter = Router();

// --- Units ---

// GET /api/missions/:missionId/units
commandRouter.get('/missions/:missionId/units', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    const units = await service.listUnits(req.params.missionId as string);
    res.json({ units });
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// GET /api/units/:id
commandRouter.get('/units/:id', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    const unit = await service.getUnit(req.params.id as string);
    if (!unit) {
      return res.status(404).json({ error: 'Unit not found' });
    }
    res.json(unit);
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// POST /api/missions/:missionId/units
commandRouter.post('/missions/:missionId/units', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    const unit = await service.createUnit({
      missionId: req.params.missionId as string,
      ...req.body
    });
    res.status(201).json(unit);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// PATCH /api/units/:id
commandRouter.patch('/units/:id', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    const unit = await service.updateUnit(req.params.id as string, req.body);
    res.json(unit);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// DELETE /api/units/:id
commandRouter.delete('/units/:id', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    await service.deleteUnit(req.params.id as string);
    res.status(204).send();
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });

});

    res.status(500).json({ error: (error as Error).message });

});

  try {
    const service = await getCommandService();
    const { relationship, warnings } = await service.createRelationship({
      missionId: req.params.missionId as string,
      ...req.body
    });
    res.status(201).json({ relationship, warnings });
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// PATCH /api/relationships/:id
commandRouter.patch('/relationships/:id', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    const relationship = await service.updateRelationship(req.params.id as string, req.body);
    res.json(relationship);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }

// DELETE /api/relationships/:id
commandRouter.delete('/relationships/:id', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();
    await service.deleteRelationship(req.params.id as string);
    res.status(204).send();
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }

// --- Hierarchy ---

// GET /api/missions/:missionId/hierarchy
commandRouter.get('/missions/:missionId/hierarchy', async (req: Request, res: Response) => {
  try {
    const service = await getCommandService();

    if (format === 'tree') {
      const tree = await service.getHierarchyAsTree(req.params.missionId as string);
