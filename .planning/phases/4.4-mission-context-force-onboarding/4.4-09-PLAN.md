**MissionDetail.tsx:**
Tabbed interface showing overview, command hierarchy, resources, and map.

```typescript
// MissionDetail.tsx
import { useState, useEffect, useContext } from 'react';
import { Tab, Tabs, TabList, TabPanel } from 'react-tabs';
import 'react-tabs/style/react-tabs.css';
import { UserContext } from '../AuthWrapper';
import { missionService, MissionWithParticipants } from '../../lib/mission-service';
import { CommandEditor } from './command/CommandEditor';
import { ResourceCatalog } from './resources/ResourceCatalog';
import { PersonnelList } from './resources/PersonnelList';
import { ConsumablesPanel } from './resources/ConsumablesPanel';
import { MissionMap } from './map/MissionMap';
import './MissionDetail.css';

interface MissionDetailProps {
  missionId: string;
  onBack: () => void;
}

export function MissionDetail({ missionId, onBack }: MissionDetailProps) {
  const [mission, setMission] = useState<MissionWithParticipants | null>(null);
  const [loading, setLoading] = useState(true);
  const [transitioning, setTransitioning] = useState(false);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadMission();
  }, [missionId]);

  const loadMission = async () => {
    setLoading(true);
    try {
      const data = await missionService.getMission(missionId, userDID);
      setMission(data);
    } catch (error) {
      console.error('Failed to load mission:', error);
    }
    setLoading(false);
  };

  const handleTransition = async (event: 'ACTIVATE' | 'COMPLETE' | 'ARCHIVE' | 'REOPEN') => {
    if (!confirm(`Are you sure you want to ${event.toLowerCase()} this mission?`)) return;

    setTransitioning(true);
    try {
      await missionService.transitionMission(missionId, event, userDID);
      loadMission();
    } catch (error) {
      alert((error as Error).message);
    }
    setTransitioning(false);
  };

  const handleDuplicate = async () => {
    const newName = prompt('Enter name for duplicated mission:', `${mission?.name} (Copy)`);
    if (!newName) return;

    const includeResources = confirm('Include resources in duplicate?');

    try {
      const newMission = await missionService.duplicateMission(missionId, newName, includeResources, userDID);
      alert(`Mission duplicated: ${newMission.name}`);
    } catch (error) {
      alert((error as Error).message);
    }
  };

  if (loading) {
    return <div className="mission-detail loading">Loading mission...</div>;
  }

  if (!mission) {
    return <div className="mission-detail error">Mission not found</div>;
  }

  const statusColors: Record<string, string> = {
    planning: 'var(--status-blue)',
    active: 'var(--status-green)',
    complete: 'var(--status-amber)',
    archived: 'var(--text-secondary)'
  };

  const getAvailableTransitions = (status: string) => {
    switch (status) {
      case 'planning': return [{ event: 'ACTIVATE', label: 'Activate Mission' }];
      case 'active': return [{ event: 'COMPLETE', label: 'Mark Complete' }];
      case 'complete': return [
        { event: 'REOPEN', label: 'Reopen Mission' },
        { event: 'ARCHIVE', label: 'Archive Mission' }
      ];
      default: return [];
    }
  };

  return (
    <div className="mission-detail">
      <header className="mission-header">
        <button className="back-button" onClick={onBack}>← Back to Missions</button>

        <div className="mission-info">
          <h1>{mission.name}</h1>
          <span
            className="status-badge"
            style={{ backgroundColor: statusColors[mission.status] }}
          >
            {mission.status.toUpperCase()}
          </span>
        </div>

        <div className="mission-actions">
          {getAvailableTransitions(mission.status).map(t => (
            <button
              key={t.event}
              className={`action-button ${t.event.toLowerCase()}`}
              onClick={() => handleTransition(t.event as any)}
              disabled={transitioning}
            >
              {t.label}
            </button>
          ))}
          <button className="action-button duplicate" onClick={handleDuplicate}>
            Duplicate
          </button>
        </div>
      </header>

      <Tabs className="mission-tabs">
        <TabList>
          <Tab>Overview</Tab>
          <Tab>Command</Tab>
          <Tab>Resources</Tab>
          <Tab>Personnel</Tab>
          <Tab>Consumables</Tab>
          <Tab>Map</Tab>
        </TabList>

        <TabPanel>
          <div className="overview-panel">
            <section className="info-section">
              <h3>Mission Information</h3>
              {mission.description && <p>{mission.description}</p>}
              <dl>
                <dt>Status</dt>
                <dd>{mission.status}</dd>
                <dt>Created By</dt>
                <dd>{mission.createdBy}</dd>
                <dt>Created At</dt>
                <dd>{new Date(mission.createdAt).toLocaleString()}</dd>
                <dt>Last Updated</dt>
                <dd>{new Date(mission.updatedAt).toLocaleString()}</dd>
              </dl>
            </section>

            <section className="participants-section">
              <h3>Participants ({mission.participants.length})</h3>
              <ul className="participant-list">
                {mission.participants.map(p => (
                  <li key={p.id}>
                    <span className="participant-did">{p.participantDID}</span>
                    <span className="participant-role">{p.role}</span>
                    <span className="participant-joined">
                      Joined {new Date(p.joinedAt).toLocaleDateString()}
                    </span>
                  </li>
                ))}
              </ul>
            </section>

            {mission.areaOfOps && (
              <section className="ao-preview">
                <h3>Area of Operations</h3>
                <MissionMap missionId={missionId} />
              </section>
            )}
          </div>
        </TabPanel>

        <TabPanel>
          <CommandEditor missionId={missionId} />
        </TabPanel>

        <TabPanel>
          <ResourceCatalog missionId={missionId} />
        </TabPanel>

        <TabPanel>
          <PersonnelList missionId={missionId} />
        </TabPanel>

        <TabPanel>
          <ConsumablesPanel missionId={missionId} />
        </TabPanel>

        <TabPanel>
          <div className="full-map-panel">
            <MissionMap
              missionId={missionId}
              websocketUrl={import.meta.env.VITE_WS_URL}
            />
          </div>
        </TabPanel>
      </Tabs>
    </div>
  );
}
```

```typescript
// MissionDashboard.tsx (update existing file)
import { useState } from 'react';
import { MissionList } from './MissionList';
import { MissionWizard } from './wizard/MissionWizard';
import { MissionDetail } from './MissionDetail';
import { Mission } from '../../lib/mission-service';
import './MissionDashboard.css';

type ViewState = 'list' | 'wizard' | 'detail';

export function MissionDashboard() {
  const [view, setView] = useState<ViewState>('list');
  const [selectedMissionId, setSelectedMissionId] = useState<string | null>(null);

  const handleSelectMission = (id: string) => {
    setSelectedMissionId(id);
    setView('detail');
  };

  const handleCreateMission = () => {
    setView('wizard');
  };

  const handleWizardComplete = (mission: Mission) => {
    setSelectedMissionId(mission.id);
    setView('detail');
  };

  const handleBack = () => {
    setView('list');
    setSelectedMissionId(null);
  };

  return (
    <div className="mission-dashboard">
      {view === 'list' && (
        <MissionList
          onSelectMission={handleSelectMission}
          onCreateMission={handleCreateMission}
        />
      )}
      {view === 'wizard' && (
        <MissionWizard
          onComplete={handleWizardComplete}
          onCancel={handleBack}
        />
      )}
      {view === 'detail' && selectedMissionId && (
        <MissionDetail
          missionId={selectedMissionId}
          onBack={handleBack}
        />
      )}
    </div>
  );
}
```
