```typescript
// MilSymbolMarker.tsx
import { useMemo } from 'react';
import { Marker, Popup } from 'react-leaflet';
import L from 'leaflet';
import ms from 'milsymbol';
import './MissionMap.css';

interface MilSymbolMarkerProps {
  position: [number, number];
  sidc: string;
  name: string;
  status?: 'FMC' | 'PMC' | 'NMC';
  additionalInfo?: Record<string, string>;
  onClick?: () => void;
}

// SIDC modifier positions for status
const STATUS_MODIFIERS: Record<string, string> = {
  FMC: '', // No modifier for fully capable
  PMC: 'X', // Damaged
  NMC: 'XX' // Destroyed/NMC
};

export function MilSymbolMarker({
  position,
  sidc,
  name,
  status = 'FMC',
  additionalInfo,
  onClick
}: MilSymbolMarkerProps) {
  // Generate milsymbol icon
  const icon = useMemo(() => {
    try {
      // Apply status modifier if applicable
      const options: any = {
        size: 35,
        // Common optional graphics
        quantity: additionalInfo?.quantity,
        staffComments: additionalInfo?.staffComments,
        uniqueDesignation: additionalInfo?.uniqueDesignation,
        higherFormation: additionalInfo?.higherFormation,
        reinforcedReduced: status === 'PMC' ? '-' : undefined,
        // Style options
        civilianColor: false,
        colorMode: 'Light',
        fill: true,
        frame: true,
        icon: true,
        infoBackground: 'white',
        infoColor: 'black',
        infoSize: 40,
        monoColor: '',
        outlineColor: 'black',
        outlineWidth: 3,
        strokeWidth: 4,
      };

      // Create symbol
      const symbol = new ms.Symbol(sidc, options);
      const anchor = symbol.getAnchor();
      const size = symbol.getSize();

      return L.divIcon({
        className: `milsymbol-marker ${status.toLowerCase()}`,
        html: symbol.asSVG(),
        iconAnchor: [anchor.x, anchor.y],
        iconSize: [size.width, size.height],
        popupAnchor: [0, -anchor.y]
      });
    } catch (error) {
      console.error('Invalid SIDC:', sidc, error);
      // Fallback to simple marker
      return L.divIcon({
        className: 'milsymbol-fallback',
        html: '<div class="fallback-marker">?</div>',
        iconSize: [24, 24],
        iconAnchor: [12, 12]
      });
    }
  }, [sidc, status, additionalInfo]);

  return (
    <Marker
      position={position}
      icon={icon}
      eventHandlers={{
        click: onClick
      }}
    >
      <Popup>
        <div className="milsymbol-popup">
          <h4>{name}</h4>
          <p className={`status ${status.toLowerCase()}`}>{status}</p>
          {additionalInfo && Object.entries(additionalInfo).map(([key, value]) => (
            <p key={key}><strong>{key}:</strong> {value}</p>
          ))}
        </div>
      </Popup>
    </Marker>
  );
}

// Helper function to create icon without React component (for realtime layer)
export function createMilSymbolIcon(sidc: string, size: number = 35): L.DivIcon {
  try {
    const symbol = new ms.Symbol(sidc, { size });
    const anchor = symbol.getAnchor();

    return L.divIcon({
      className: 'milsymbol-marker',
      html: symbol.asSVG(),
      iconAnchor: [anchor.x, anchor.y],
      iconSize: [symbol.getSize().width, symbol.getSize().height]
    });
  } catch {
    return L.divIcon({
      className: 'milsymbol-fallback',
      html: '<div class="fallback-marker">?</div>',
      iconSize: [24, 24],
      iconAnchor: [12, 12]
    });
  }
}
```

```typescript
// SensorCoverage.tsx
import { useMemo } from 'react';
import { Circle, Polygon, Popup } from 'react-leaflet';
import './MissionMap.css';

interface Sensor {
  id: string;
  name: string;
  type: 'airborne' | 'ground' | 'maritime' | 'space' | 'autonomous';
  position: [number, number];
  coverageRadius?: number; // in meters
  coveragePolygon?: [number, number][];
  range: number;
  resolution?: string;
  status: 'active' | 'inactive' | 'maintenance';
}

interface SensorCoverageProps {
  sensors: Sensor[];
  showGradient?: boolean;
  selectedSensorId?: string;
  onSelectSensor?: (id: string) => void;
}

// Sensor type colors
const SENSOR_COLORS: Record<string, string> = {
  airborne: '#00BFFF',    // Deep sky blue
  ground: '#32CD32',      // Lime green
  maritime: '#4169E1',    // Royal blue
  space: '#9400D3',       // Dark violet
  autonomous: '#FF8C00'   // Dark orange
};

export function SensorCoverage({
  sensors,
  showGradient = false,
  selectedSensorId,
  onSelectSensor
}: SensorCoverageProps) {
  // Filter active sensors
  const activeSensors = useMemo(() =>
    sensors.filter(s => s.status === 'active'),
    [sensors]
  );

  const renderCoverage = (sensor: Sensor) => {
    const color = SENSOR_COLORS[sensor.type] || '#888888';
    const isSelected = sensor.id === selectedSensorId;
    const opacity = isSelected ? 0.5 : 0.2;
    const weight = isSelected ? 3 : 1;

    // Use polygon if defined, otherwise circle
    if (sensor.coveragePolygon && sensor.coveragePolygon.length >= 3) {
      return (
        <Polygon
          key={sensor.id}
          positions={sensor.coveragePolygon}
          pathOptions={{
            color,
            fillColor: color,
            fillOpacity: opacity,
            weight
          }}
          eventHandlers={{
            click: () => onSelectSensor?.(sensor.id)
          }}
        >
          <Popup>
            <div className="sensor-popup">
              <h4>{sensor.name}</h4>
              <p className="sensor-type">{sensor.type}</p>
              <p>Range: {sensor.range}km</p>
              {sensor.resolution && <p>Resolution: {sensor.resolution}</p>}
            </div>
          </Popup>
        </Polygon>
      );
    }

    // Default to circle coverage
    return (
      <Circle
        key={sensor.id}
        center={sensor.position}
        radius={sensor.coverageRadius || sensor.range * 1000} // Convert km to m
        pathOptions={{
          color,
          fillColor: color,
          fillOpacity: showGradient ? 0 : opacity,
          weight,
          // Gradient effect via CSS
          className: showGradient ? 'sensor-gradient' : ''
        }}
        eventHandlers={{
          click: () => onSelectSensor?.(sensor.id)
        }}
      >
        <Popup>
          <div className="sensor-popup">
            <h4>{sensor.name}</h4>
            <p className="sensor-type">{sensor.type}</p>
            <p>Range: {sensor.range}km</p>
            {sensor.resolution && <p>Resolution: {sensor.resolution}</p>}
          </div>
        </Popup>
      </Circle>
    );
  };

  return (
    <>
      {activeSensors.map(renderCoverage)}
    </>
  );
}

// Sensor layer legend
export function SensorLegend() {
  return (
    <div className="sensor-legend">
      <h4>Sensor Types</h4>
      {Object.entries(SENSOR_COLORS).map(([type, color]) => (
        <div key={type} className="legend-item">
          <span className="legend-color" style={{ backgroundColor: color }} />
          <span className="legend-label">{type}</span>
        </div>
      ))}
    </div>
  );
}
```

```typescript
// RealtimeTracker.tsx
import { useEffect, useRef, useCallback } from 'react';
import { useMap } from 'react-leaflet';
import L from 'leaflet';
import 'leaflet-realtime';
import { createMilSymbolIcon } from './MilSymbolMarker';
import './MissionMap.css';

interface TrackedAsset {
  id: string;
  name: string;
  sidc: string;
  position: [number, number];
  timestamp: string;
  speed?: number;
  heading?: number;
}

interface RealtimeTrackerProps {
  websocketUrl?: string;
  missionId: string;
  showTrails?: boolean;
  maxTrailLength?: number;
  onAssetClick?: (assetId: string) => void;
}

export function RealtimeTracker({
  websocketUrl,
  missionId,
  showTrails = true,
  maxTrailLength = 50,
  onAssetClick
}: RealtimeTrackerProps) {
  const map = useMap();
  const realtimeLayerRef = useRef<L.Realtime | null>(null);
  const trailsRef = useRef<Map<string, L.Polyline>>(new Map());
  const positionHistoryRef = useRef<Map<string, [number, number][]>>(new Map());

  // Initialize realtime layer
  useEffect(() => {
    if (realtimeLayerRef.current) {
      map.removeLayer(realtimeLayerRef.current);
    }

    // Create realtime layer with manual updates (WebSocket push)
    const realtimeLayer = (L as any).realtime(undefined, {
      start: false, // Don't auto-poll
      getFeatureId: (feature: any) => feature.properties.id,
      pointToLayer: (feature: any, latlng: L.LatLng) => {
        const { sidc, name, id } = feature.properties;
        const marker = L.marker(latlng, {
          icon: createMilSymbolIcon(sidc)
        });

        marker.bindPopup(`<strong>${name}</strong>`);
        marker.on('click', () => onAssetClick?.(id));

        return marker;
      }
    });

    realtimeLayer.addTo(map);
    realtimeLayerRef.current = realtimeLayer;

    // Handle realtime events for trails
    realtimeLayer.on('update', (e: any) => {
      if (!showTrails) return;

      Object.keys(e.update).forEach((id: string) => {
        const layer = e.update[id];
        const latlng = layer.getLatLng();

        // Update position history
        const history = positionHistoryRef.current.get(id) || [];
        history.push([latlng.lat, latlng.lng]);

        // Trim to max length
        if (history.length > maxTrailLength) {
          history.shift();
        }
        positionHistoryRef.current.set(id, history);

        // Update trail polyline
        let trail = trailsRef.current.get(id);
        if (!trail) {
          trail = L.polyline([], {
            color: 'rgba(0, 191, 255, 0.6)',
            weight: 2,
            dashArray: '5, 5'
          });
          trail.addTo(map);
          trailsRef.current.set(id, trail);
        }
        trail.setLatLngs(history);
      });
    });

    realtimeLayer.on('enter', (e: any) => {
      // New asset appeared - initialize trail
      if (showTrails) {
        Object.keys(e.enter).forEach((id: string) => {
          positionHistoryRef.current.set(id, []);
        });
      }
    });

    realtimeLayer.on('exit', (e: any) => {
      // Asset removed - clean up trail
      Object.keys(e.exit).forEach((id: string) => {
        const trail = trailsRef.current.get(id);
        if (trail) {
          map.removeLayer(trail);
          trailsRef.current.delete(id);
        }
        positionHistoryRef.current.delete(id);
      });
    });

    // Connect to WebSocket if URL provided
    let ws: WebSocket | null = null;
    if (websocketUrl) {
      ws = new WebSocket(`${websocketUrl}?missionId=${missionId}`);

      ws.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data);

          // Expect GeoJSON FeatureCollection
          if (data.type === 'FeatureCollection') {
            realtimeLayer.update(data);
          } else if (data.type === 'Feature') {
            // Single feature update
            realtimeLayer.update({
              type: 'FeatureCollection',
              features: [data]
            });
          }
        } catch (error) {
          console.error('Failed to parse realtime data:', error);
        }
      };

      ws.onerror = (error) => {
        console.error('WebSocket error:', error);
      };
    }

    return () => {
      if (realtimeLayerRef.current) {
        map.removeLayer(realtimeLayerRef.current);
      }
      // Clean up trails
      trailsRef.current.forEach(trail => map.removeLayer(trail));
      trailsRef.current.clear();
      positionHistoryRef.current.clear();

      if (ws) {
        ws.close();
      }
    };
  }, [map, websocketUrl, missionId, showTrails, maxTrailLength, onAssetClick]);

  // Method to manually push updates (for non-WebSocket usage)
  const pushUpdate = useCallback((assets: TrackedAsset[]) => {
    if (!realtimeLayerRef.current) return;

    const geojson = {
      type: 'FeatureCollection',
      features: assets.map(asset => ({
        type: 'Feature',
        properties: {
          id: asset.id,
          name: asset.name,
          sidc: asset.sidc,
          speed: asset.speed,
          heading: asset.heading,
          timestamp: asset.timestamp
        },
        geometry: {
          type: 'Point',
          coordinates: [asset.position[1], asset.position[0]] // GeoJSON is [lng, lat]
        }
      }))
    };

    realtimeLayerRef.current.update(geojson);
  }, []);

  // Expose pushUpdate via ref or return
  return null; // This component doesn't render anything visible
}

// Trail toggle control
export function TrailControl({
  showTrails,
  onToggle
}: {
  showTrails: boolean;
  onToggle: (show: boolean) => void;
}) {
  return (
    <div className="trail-control">
      <label>
        <input
          type="checkbox"
          checked={showTrails}
          onChange={(e) => onToggle(e.target.checked)}
        />
        Show movement trails
      </label>
    </div>
  );
}
```

```typescript
// MissionMap.tsx
import { useState, useEffect, useContext, useMemo } from 'react';
import { MapContainer, TileLayer, GeoJSON, LayersControl } from 'react-leaflet';
import { UserContext } from '../../AuthWrapper';
import { missionService, Mission } from '../../../lib/mission-service';
import { commandService, Unit } from '../../../lib/command-service';
import { resourceService, ResourceWithCategory, CategoryNode } from '../../../lib/resource-service';
import { MilSymbolMarker } from './MilSymbolMarker';
import { SensorCoverage } from './SensorCoverage';
import { RealtimeTracker } from './RealtimeTracker';
import { LayerControls } from './LayerControls';
import './MissionMap.css';

interface MissionMapProps {
  missionId: string;
  websocketUrl?: string;
  onResourceClick?: (resourceId: string) => void;
  onUnitClick?: (unitId: string) => void;
}

interface LayerState {
  showAO: boolean;
  showUnits: boolean;
  showResources: boolean;
  showSensors: boolean;
  showSensorCoverage: boolean;
  sensorGradient: boolean;
  showTrails: boolean;
  resourceCategories: string[];
}

const DEFAULT_LAYERS: LayerState = {
  showAO: true,
  showUnits: true,
  showResources: true,
  showSensors: true,
  showSensorCoverage: true,
  sensorGradient: false,
  showTrails: true,
  resourceCategories: []
};

export function MissionMap({
  missionId,
  websocketUrl,
  onResourceClick,
  onUnitClick
}: MissionMapProps) {
  const [mission, setMission] = useState<Mission | null>(null);
  const [units, setUnits] = useState<Unit[]>([]);
  const [resources, setResources] = useState<ResourceWithCategory[]>([]);
  const [categories, setCategories] = useState<CategoryNode[]>([]);
  const [sensors, setSensors] = useState<any[]>([]); // Sensor type from backend
  const [layers, setLayers] = useState<LayerState>(DEFAULT_LAYERS);
  const [loading, setLoading] = useState(true);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadMapData();
  }, [missionId]);

  const loadMapData = async () => {
    setLoading(true);
    try {
      const [missionData, unitList, resourceList, categoryTree] = await Promise.all([
        missionService.getMission(missionId, userDID),
        commandService.listUnits(missionId, userDID),
        resourceService.listResources(missionId, {}, userDID),
        resourceService.getCategoryTree(userDID)
      ]);

      setMission(missionData);
      setUnits(unitList);
      setResources(resourceList);
      setCategories(categoryTree);

      // TODO: Load sensors when sensor registration is implemented
      setSensors([]);

      // Initialize with all categories selected
      const flatCats = flattenCategories(categoryTree);
      setLayers(prev => ({
        ...prev,
        resourceCategories: flatCats.map(c => c.id)
      }));
    } catch (error) {
      console.error('Failed to load map data:', error);
    }
    setLoading(false);
  };

  const flattenCategories = (nodes: CategoryNode[]): { id: string; name: string }[] => {
    let result: { id: string; name: string }[] = [];
    for (const node of nodes) {
      result.push({ id: node.id, name: node.name });
      if (node.children.length > 0) {
        result = result.concat(flattenCategories(node.children));
      }
    }
    return result;
  };

  const handleLayerChange = (key: keyof LayerState, value: any) => {
    setLayers(prev => ({ ...prev, [key]: value }));
  };

  // Filter resources by selected categories
  const filteredResources = useMemo(() =>
    resources.filter(r =>
      !r.categoryId || layers.resourceCategories.includes(r.categoryId)
    ),
    [resources, layers.resourceCategories]
  );

  // Calculate map center from AO or default
  const mapCenter = useMemo((): [number, number] => {
    if (mission?.areaOfOps) {
      // Calculate centroid of polygon
      const coords = mission.areaOfOps.coordinates[0];
      const lat = coords.reduce((sum, c) => sum + c[1], 0) / coords.length;
      const lng = coords.reduce((sum, c) => sum + c[0], 0) / coords.length;
      return [lat, lng];
    }
    return [0, 0]; // Default to world center
  }, [mission]);

  if (loading) {
    return <div className="mission-map loading">Loading map...</div>;
  }

  return (
    <div className="mission-map-container">
      <MapContainer
        center={mapCenter}
        zoom={mission?.areaOfOps ? 10 : 2}
        className="mission-map"
      >
        <TileLayer
          url="https://tiles.stadiamaps.com/tiles/alidade_smooth_dark/{z}/{x}/{y}{r}.png"
          attribution='&copy; <a href="https://stadiamaps.com/">Stadia Maps</a>'
        />

        {/* Area of Operations */}
        {layers.showAO && mission?.areaOfOps && (
          <GeoJSON
            data={mission.areaOfOps}
            style={{
              color: 'var(--accent-blue)',
              weight: 2,
              fillColor: 'var(--accent-blue)',
              fillOpacity: 0.1
            }}
          />
        )}

        {/* Units with MIL-STD-2525D symbols */}
        {layers.showUnits && units.map(unit => {
          // Skip units without location (would need location from DB)
          if (!unit.sidc) return null;
          // For now, place units at mission center (location tracking TBD)
          return (
            <MilSymbolMarker
              key={unit.id}
              position={mapCenter}
              sidc={unit.sidc}
              name={unit.name}
              onClick={() => onUnitClick?.(unit.id)}
            />
          );
        })}

        {/* Resources with MIL-STD-2525D symbols */}
        {layers.showResources && filteredResources.map(resource => {
          if (!resource.location || !resource.sidc) return null;
          const coords = resource.location.coordinates;
          return (
            <MilSymbolMarker
              key={resource.id}
              position={[coords[1], coords[0]]}
              sidc={resource.sidc}
              name={resource.name}
              status={resource.status}
              onClick={() => onResourceClick?.(resource.id)}
            />
          );
        })}

        {/* Sensor coverage */}
        {layers.showSensorCoverage && (
          <SensorCoverage
            sensors={sensors}
            showGradient={layers.sensorGradient}
          />
        )}

        {/* Real-time tracking */}
        <RealtimeTracker
          missionId={missionId}
          websocketUrl={websocketUrl}
          showTrails={layers.showTrails}
        />
      </MapContainer>

      <LayerControls
        layers={layers}
        availableCategories={flattenCategories(categories)}
        onLayerChange={handleLayerChange}
      />
    </div>
  );
}
```
