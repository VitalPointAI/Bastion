  <files>
    backend/src/api/sensors.ts

  </files>
  <action>
    1. Create backend/src/api/sensors.ts:
       - Import sensorStore from '../sensors/sensor-store'
       - Create Express router: sensorRouter
       - POST /api/sensors - Create sensor (validates with SensorInputSchema)
       - GET /api/sensors?missionId=X&category=Y - List with filters
       - GET /api/sensors/:id - Get single sensor
       - PATCH /api/sensors/:id - Update sensor
       - PATCH /api/sensors/:id/status - Update status only
       - PATCH /api/sensors/:id/location - Update location (lat/lng)
       - DELETE /api/sensors/:id - Delete sensor
       - GET /api/sensors/coverage/:missionId - Get all sensor coverage polygons for map
       - POST /api/sensors/:id/coverage - Define/update coverage polygon
       - All endpoints use sensorStore methods (from 4.4-01)

    2. Mount in backend/src/index.ts:
       - Import sensorRouter from './api/sensors'
       - app.use('/api/sensors', sensorRouter)

    3. Create frontend/src/lib/sensor-service.ts:
       ```typescript
       const API_BASE = '/api/sensors';

       export const sensorService = {
         createSensor: (data) => fetch(API_BASE, { method: 'POST', ... }),
         getSensors: (missionId, category?) => fetch(`${API_BASE}?missionId=${missionId}${category ? `&category=${category}` : ''}`),
         getSensor: (id) => fetch(`${API_BASE}/${id}`),
         updateSensor: (id, data) => fetch(`${API_BASE}/${id}`, { method: 'PATCH', ... }),
         updateStatus: (id, status) => fetch(`${API_BASE}/${id}/status`, { method: 'PATCH', body: JSON.stringify({ status }) }),
         updateLocation: (id, lat, lng) => fetch(`${API_BASE}/${id}/location`, { method: 'PATCH', body: JSON.stringify({ lat, lng }) }),
         deleteSensor: (id) => fetch(`${API_BASE}/${id}`, { method: 'DELETE' }),
         getCoverage: (missionId) => fetch(`${API_BASE}/coverage/${missionId}`),
         updateCoverage: (id, polygon) => fetch(`${API_BASE}/${id}/coverage`, { method: 'POST', body: JSON.stringify({ polygon }) }),
       };
       ```
  </action>
  <verify>
    - TypeScript compiles: `cd backend && npx tsc --noEmit`
    - Backend starts without errors: `cd backend && npm run dev` (briefly test)
    - Sensor API reachable: `curl http://localhost:3001/api/sensors?missionId=test` returns 200 (empty array OK)
  </verify>
