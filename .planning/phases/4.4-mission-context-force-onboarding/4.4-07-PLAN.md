import axios from 'axios';

const API_BASE = import.meta.env.VITE_API_URL || 'http://localhost:3001';

export interface ResourceCategory {
  id: string;
  parentId?: string;
  name: string;
  description?: string;
  createdAt: string;
}

export interface CategoryNode extends ResourceCategory {
  children: CategoryNode[];
}

export interface Resource {
  id: string;
  missionId: string;
  categoryId?: string;
  name: string;
  serialNumber?: string;
  sidc?: string;
  specifications?: Record<string, any>;
  status: 'FMC' | 'PMC' | 'NMC';
  statusReason?: string;
  location?: GeoJSON.Point;
  lastLocationUpdate?: string;
  createdAt: string;
}

export interface ResourceWithCategory extends Resource {
  categoryName?: string;
}

export interface Personnel {
  id: string;
  missionId: string;
  did?: string;
  name: string;
  rank?: string;
  specialty?: string;
  unitId?: string;
  readinessStatus: string;
  readinessReason?: string;
  createdAt: string;
}

export interface Consumable {
  id: string;
  missionId: string;
  categoryId?: string;
  name: string;
  supplyPointId?: string;
  quantity: number;
  unit?: string;
  reorderLevel?: number;
  createdAt: string;
}

export interface BulkImportResult {
  imported: number;
  failed: number;
  errors: { row: number; error: string }[];
}

export interface ResourceStats {
  totalResources: number;
  byStatus: Record<string, number>;
  byCategory: Record<string, number>;
  personnelCount: number;
  consumableLowStock: number;
}

class ResourceService {
  private getHeaders(userDID?: string) {
    const headers: Record<string, string> = { 'Content-Type': 'application/json' };
    if (userDID) headers['X-DID'] = userDID;
    return headers;
  }

  // --- Categories ---

  async getCategoryTree(userDID: string): Promise<CategoryNode[]> {
    const response = await axios.get(`${API_BASE}/api/resources/categories?format=tree`, {
      headers: this.getHeaders(userDID)
    });
    return response.data.categories;
  }

  async createCategory(data: Partial<ResourceCategory>, userDID: string): Promise<ResourceCategory> {
    const response = await axios.post(`${API_BASE}/api/resources/categories`, data, {
      headers: this.getHeaders(userDID)
    });
    return response.data;
  }

  // --- Resources ---

  async listResources(missionId: string, filters: {
    categoryId?: string;
    status?: string;
    search?: string;
  }, userDID: string): Promise<ResourceWithCategory[]> {
    const params = new URLSearchParams();
    if (filters.categoryId) params.set('categoryId', filters.categoryId);
    if (filters.status) params.set('status', filters.status);
    if (filters.search) params.set('search', filters.search);

    const response = await axios.get(
      `${API_BASE}/api/resources/missions/${missionId}/resources?${params}`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.resources;
  }

  async getResource(id: string, userDID: string): Promise<ResourceWithCategory> {
    const response = await axios.get(`${API_BASE}/api/resources/resources/${id}`, {
      headers: this.getHeaders(userDID)
    });
    return response.data;
  }

  async createResource(missionId: string, data: Partial<Resource>, userDID: string): Promise<Resource> {
    const response = await axios.post(
      `${API_BASE}/api/resources/missions/${missionId}/resources`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async updateResource(id: string, data: Partial<Resource>, userDID: string): Promise<Resource> {
    const response = await axios.patch(`${API_BASE}/api/resources/resources/${id}`, data, {
      headers: this.getHeaders(userDID)
    });
    return response.data;
  }

  async updateResourceStatus(id: string, status: string, reason: string, userDID: string): Promise<Resource> {
    const response = await axios.patch(`${API_BASE}/api/resources/resources/${id}/status`, {
      status, reason
    }, { headers: this.getHeaders(userDID) });
    return response.data;
  }

  async deleteResource(id: string, userDID: string): Promise<void> {
    await axios.delete(`${API_BASE}/api/resources/resources/${id}`, {
      headers: this.getHeaders(userDID)
    });
  }

  async bulkImportResources(missionId: string, rows: any[], userDID: string): Promise<BulkImportResult> {
    const response = await axios.post(
      `${API_BASE}/api/resources/missions/${missionId}/resources/bulk`,
      { rows },
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  // --- Personnel ---

  async listPersonnel(missionId: string, filters: {
    unitId?: string;
    readinessStatus?: string;
    specialty?: string;
  }, userDID: string): Promise<Personnel[]> {
    const params = new URLSearchParams();
    if (filters.unitId) params.set('unitId', filters.unitId);
    if (filters.readinessStatus) params.set('readinessStatus', filters.readinessStatus);
    if (filters.specialty) params.set('specialty', filters.specialty);

    const response = await axios.get(
      `${API_BASE}/api/resources/missions/${missionId}/personnel?${params}`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.personnel;
  }

  async createPersonnel(missionId: string, data: Partial<Personnel>, userDID: string): Promise<Personnel> {
    const response = await axios.post(
      `${API_BASE}/api/resources/missions/${missionId}/personnel`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async updatePersonnel(id: string, data: Partial<Personnel>, userDID: string): Promise<Personnel> {
    const response = await axios.patch(`${API_BASE}/api/resources/personnel/${id}`, data, {
      headers: this.getHeaders(userDID)
    });
    return response.data;
  }

  async updateReadiness(id: string, status: string, reason: string, userDID: string): Promise<Personnel> {
    const response = await axios.patch(`${API_BASE}/api/resources/personnel/${id}/readiness`, {
      status, reason
    }, { headers: this.getHeaders(userDID) });
    return response.data;
  }

  async bulkImportPersonnel(missionId: string, rows: any[], userDID: string): Promise<BulkImportResult> {
    const response = await axios.post(
      `${API_BASE}/api/resources/missions/${missionId}/personnel/bulk`,
      { rows },
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  // --- Consumables ---

  async listConsumables(missionId: string, filters: {
    categoryId?: string;
    belowReorderLevel?: boolean;
  }, userDID: string): Promise<Consumable[]> {
    const params = new URLSearchParams();
    if (filters.categoryId) params.set('categoryId', filters.categoryId);
    if (filters.belowReorderLevel) params.set('belowReorderLevel', 'true');

    const response = await axios.get(
      `${API_BASE}/api/resources/missions/${missionId}/consumables?${params}`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.consumables;
  }

  async createConsumable(missionId: string, data: Partial<Consumable>, userDID: string): Promise<Consumable> {
    const response = await axios.post(
      `${API_BASE}/api/resources/missions/${missionId}/consumables`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async updateConsumableQuantity(id: string, quantity: number, userDID: string): Promise<Consumable> {
    const response = await axios.patch(`${API_BASE}/api/resources/consumables/${id}/quantity`, {
      quantity
    }, { headers: this.getHeaders(userDID) });
    return response.data;
  }

  // --- Stats ---

  async getStats(missionId: string, userDID: string): Promise<ResourceStats> {
    const response = await axios.get(
      `${API_BASE}/api/resources/missions/${missionId}/resources/stats`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }
}

// BulkImporter.tsx
import { useState, useContext } from 'react';
import { Importer, ImporterField } from 'react-csv-importer';
import 'react-csv-importer/dist/index.css';
import { UserContext } from '../../AuthWrapper';
import { resourceService, BulkImportResult } from '../../../lib/resource-service';
import './ResourceCatalog.css';

interface BulkImporterProps {
  missionId: string;
  type: 'resources' | 'personnel';
  onComplete: (result: BulkImportResult) => void;
  onCancel: () => void;
}

export function BulkImporter({ missionId, type, onComplete, onCancel }: BulkImporterProps) {
  const [result, setResult] = useState<BulkImportResult | null>(null);
  const { userDID } = useContext(UserContext);

  const handleDataHandler = async (rows: any[], info: { startIndex: number }) => {
    let importResult: BulkImportResult;

    if (type === 'resources') {
      importResult = await resourceService.bulkImportResources(missionId, rows, userDID);
    } else {
      importResult = await resourceService.bulkImportPersonnel(missionId, rows, userDID);
    }

    setResult(importResult);
    return; // Don't throw - let user see partial results
  };

  const handleComplete = () => {
    if (result) {
      onComplete(result);
    }
  };

  return (
    <div className="bulk-importer">
      <header>
        <h3>Bulk Import {type === 'resources' ? 'Resources' : 'Personnel'}</h3>
        <button className="btn-close" onClick={onCancel}>Cancel</button>
      </header>

      <Importer
        dataHandler={handleDataHandler}
        defaultNoHeader={false}
        restartable={true}
        onClose={handleComplete}
      >
        {type === 'resources' ? (
          <>
            <ImporterField name="name" label="Resource Name" />
            <ImporterField name="categoryId" label="Category ID" optional />
            <ImporterField name="serialNumber" label="Serial Number" optional />
            <ImporterField name="sidc" label="Symbol Code (SIDC)" optional />
            <ImporterField name="status" label="Status (FMC/PMC/NMC)" optional />
            <ImporterField name="statusReason" label="Status Reason" optional />
            <ImporterField name="latitude" label="Latitude" optional />
            <ImporterField name="longitude" label="Longitude" optional />
          </>
        ) : (
          <>
            <ImporterField name="name" label="Name" />
            <ImporterField name="did" label="DID" optional />
            <ImporterField name="rank" label="Rank" optional />
            <ImporterField name="specialty" label="Specialty" optional />
            <ImporterField name="unitId" label="Unit ID" optional />
            <ImporterField name="readinessStatus" label="Readiness Status" optional />
          </>
        )}
      </Importer>

      {result && (
        <div className="import-result">
          <h4>Import Complete</h4>
          <p className="success">{result.imported} records imported</p>
          {result.failed > 0 && (
            <>
              <p className="error">{result.failed} records failed</p>
              <ul className="error-list">
                {result.errors.slice(0, 10).map((e, i) => (
                  <li key={i}>Row {e.row}: {e.error}</li>
                ))}
                {result.errors.length > 10 && (
                  <li>...and {result.errors.length - 10} more errors</li>
                )}
              </ul>
            </>
          )}
          <button onClick={() => onComplete(result)}>Done</button>
        </div>
      )}
    </div>
  );
}

// ResourceForm.tsx
import { useContext, useEffect, useState } from 'react';
import { useForm } from 'react-hook-form';
import { UserContext } from '../../AuthWrapper';
import { resourceService, Resource, CategoryNode, STATUS_LABELS } from '../../../lib/resource-service';
import './ResourceCatalog.css';

interface ResourceFormProps {
  missionId: string;
  resource?: Resource;
  onSubmit: () => void;
  onCancel: () => void;
}

interface FormData {
  name: string;
  categoryId: string;
  serialNumber?: string;
  sidc?: string;
  status: 'FMC' | 'PMC' | 'NMC';
  statusReason?: string;
  specifications?: string;
}

export function ResourceForm({ missionId, resource, onSubmit, onCancel }: ResourceFormProps) {
  const { register, handleSubmit, formState: { errors } } = useForm<FormData>({
    defaultValues: resource ? {
      name: resource.name,
      categoryId: resource.categoryId || '',
      serialNumber: resource.serialNumber,
      sidc: resource.sidc,
      status: resource.status,
      statusReason: resource.statusReason,
      specifications: resource.specifications ? JSON.stringify(resource.specifications) : ''
    } : { status: 'FMC' }
  });
  const [categories, setCategories] = useState<CategoryNode[]>([]);
  const [submitting, setSubmitting] = useState(false);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadCategories();
  }, []);

  const loadCategories = async () => {
    const tree = await resourceService.getCategoryTree(userDID);
    setCategories(tree);
  };

  // Flatten tree for select options
  const flattenCategories = (nodes: CategoryNode[], depth = 0): { id: string; name: string; depth: number }[] => {
    let result: { id: string; name: string; depth: number }[] = [];
    for (const node of nodes) {
      result.push({ id: node.id, name: node.name, depth });
      if (node.children.length > 0) {
        result = result.concat(flattenCategories(node.children, depth + 1));
      }
    }
    return result;
  };

  const submit = async (data: FormData) => {
    setSubmitting(true);
    try {
      const payload: any = {
        name: data.name,
        categoryId: data.categoryId || undefined,
        serialNumber: data.serialNumber || undefined,
        sidc: data.sidc || undefined,
        status: data.status,
        statusReason: data.statusReason || undefined,
        specifications: data.specifications ? JSON.parse(data.specifications) : undefined
      };

      if (resource) {
        await resourceService.updateResource(resource.id, payload, userDID);
      } else {
        await resourceService.createResource(missionId, payload, userDID);
      }
      onSubmit();
    } catch (error) {
      alert((error as Error).message);
    }
    setSubmitting(false);
  };

  const flatCategories = flattenCategories(categories);

  return (
    <form className="resource-form" onSubmit={handleSubmit(submit)}>
      <h3>{resource ? 'Edit Resource' : 'Add Resource'}</h3>

      <div className="form-group">
        <label>Name *</label>
        <input {...register('name', { required: 'Name is required' })} />
        {errors.name && <span className="error">{errors.name.message}</span>}
      </div>

      <div className="form-group">
        <label>Category</label>
        <select {...register('categoryId')}>
          <option value="">No category</option>
          {flatCategories.map(cat => (
            <option key={cat.id} value={cat.id}>
              {'  '.repeat(cat.depth)}{cat.name}
            </option>
          ))}
        </select>
      </div>

      <div className="form-group">
        <label>Serial Number</label>
        <input {...register('serialNumber')} />
      </div>

      <div className="form-group">
        <label>Symbol Code (SIDC)</label>
        <input {...register('sidc')} placeholder="e.g., SFGPUCI-----" />
        <small>MIL-STD-2525D symbol identification code</small>
      </div>

      <div className="form-row">
        <div className="form-group">
          <label>Status</label>
          <select {...register('status')}>
            {Object.entries(STATUS_LABELS).map(([key, label]) => (
              <option key={key} value={key}>{label} ({key})</option>
            ))}
          </select>
        </div>
        <div className="form-group">
          <label>Status Reason</label>
          <input {...register('statusReason')} placeholder="e.g., Scheduled maintenance" />
        </div>
      </div>

      <div className="form-group">
        <label>Specifications (JSON)</label>
        <textarea {...register('specifications')} rows={3} placeholder='{"range": "50km", "crew": 4}' />
      </div>

      <div className="form-actions">
        <button type="button" onClick={onCancel}>Cancel</button>
        <button type="submit" disabled={submitting}>
          {submitting ? 'Saving...' : (resource ? 'Update' : 'Create')}
        </button>
      </div>
    </form>
  );
}

// ResourceCatalog.tsx
import { useState, useEffect, useContext } from 'react';
import { UserContext } from '../../AuthWrapper';
import {
  resourceService, ResourceWithCategory, CategoryNode, ResourceStats
} from '../../../lib/resource-service';
import { ResourceForm } from './ResourceForm';
import { BulkImporter } from './BulkImporter';
import { AvailabilityBadge } from './AvailabilityBadge';
import './ResourceCatalog.css';

type ViewMode = 'list' | 'form' | 'import';

interface ResourceCatalogProps {
  missionId: string;
}

export function ResourceCatalog({ missionId }: ResourceCatalogProps) {
  const [resources, setResources] = useState<ResourceWithCategory[]>([]);
  const [categories, setCategories] = useState<CategoryNode[]>([]);
  const [stats, setStats] = useState<ResourceStats | null>(null);
  const [selectedCategory, setSelectedCategory] = useState<string | null>(null);
  const [statusFilter, setStatusFilter] = useState<string>('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [viewMode, setViewMode] = useState<ViewMode>('list');
  const [editingResource, setEditingResource] = useState<ResourceWithCategory | null>(null);
  const [loading, setLoading] = useState(true);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadData();
  }, [missionId, selectedCategory, statusFilter, searchQuery]);

  const loadData = async () => {
    setLoading(true);
    try {
      const [resourceList, categoryTree, statsData] = await Promise.all([
        resourceService.listResources(missionId, {
          categoryId: selectedCategory || undefined,
          status: statusFilter === 'all' ? undefined : statusFilter,
          search: searchQuery || undefined
        }, userDID),
        resourceService.getCategoryTree(userDID),
        resourceService.getStats(missionId, userDID)
      ]);
      setResources(resourceList);
      setCategories(categoryTree);
      setStats(statsData);
    } catch (error) {
      console.error('Failed to load resources:', error);
    }
    setLoading(false);
  };

  const handleEdit = (resource: ResourceWithCategory) => {
    setEditingResource(resource);
    setViewMode('form');
  };

  const handleDelete = async (id: string) => {
    if (confirm('Delete this resource?')) {
      await resourceService.deleteResource(id, userDID);
      loadData();
    }
  };

  const handleFormComplete = () => {
    setViewMode('list');
    setEditingResource(null);
    loadData();
  };

  const renderCategoryTree = (nodes: CategoryNode[], depth = 0): JSX.Element => (
    <ul className="category-tree">
      {nodes.map(node => (
        <li key={node.id}>
          <button
            className={`category-item ${selectedCategory === node.id ? 'selected' : ''}`}
            onClick={() => setSelectedCategory(node.id === selectedCategory ? null : node.id)}
            style={{ paddingLeft: `${8 + depth * 16}px` }}
          >
            {node.name}
          </button>
          {node.children.length > 0 && renderCategoryTree(node.children, depth + 1)}
        </li>
      ))}
    </ul>
  );

  return (
    <div className="resource-catalog">
      <aside className="category-sidebar">
        <h3>Categories</h3>
        <button
          className={`category-item ${!selectedCategory ? 'selected' : ''}`}
          onClick={() => setSelectedCategory(null)}
        >
          All Resources
        </button>
        {renderCategoryTree(categories)}
      </aside>

      <main className="resource-content">
        <header>
          <div className="search-bar">
            <input
              type="search"
              placeholder="Search resources..."
              value={searchQuery}
              onChange={e => setSearchQuery(e.target.value)}
            />
          </div>
          <div className="filters">
            <select value={statusFilter} onChange={e => setStatusFilter(e.target.value)}>
              <option value="all">All Status</option>
              <option value="FMC">FMC Only</option>
              <option value="PMC">PMC Only</option>

              <span className="value">{stats.byStatus.NMC || 0}</span>
              <span className="label">NMC</span>
            </div>
          </div>
        )}

        {viewMode === 'list' && (
          <div className="resource-grid">
            {loading ? (
              <div className="loading">Loading resources...</div>
            ) : resources.length === 0 ? (
              <div className="empty">No resources found</div>
            ) : (
              resources.map(resource => (
                <div key={resource.id} className="resource-card" onClick={() => handleEdit(resource)}>
                  <div className="resource-header">
                    <AvailabilityBadge status={resource.status} reason={resource.statusReason} />
                    {resource.categoryName && (
                      <span className="category-badge">{resource.categoryName}</span>
                    )}
                  </div>
                  <h4>{resource.name}</h4>
                  {resource.serialNumber && (
                    <p className="serial">S/N: {resource.serialNumber}</p>
                  )}
                  <div className="resource-actions">
                    <button onClick={(e) => { e.stopPropagation(); handleDelete(resource.id); }}>
                      Delete
                    </button>
                  </div>
                </div>
              ))
            )}
          </div>
        )}

        {viewMode === 'form' && (
          <ResourceForm
            missionId={missionId}
            resource={editingResource || undefined}
            onSubmit={handleFormComplete}
            onCancel={() => { setViewMode('list'); setEditingResource(null); }}
          />
        )}

        {viewMode === 'import' && (
          <BulkImporter
            missionId={missionId}
            type="resources"
            onComplete={(result) => { setViewMode('list'); loadData(); }}
            onCancel={() => setViewMode('list')}
          />
        )}
      </main>
    </div>
  );
}

// ConsumablesPanel.tsx
import { useState, useEffect, useContext } from 'react';
import { UserContext } from '../../AuthWrapper';
import { resourceService, Consumable } from '../../../lib/resource-service';
import './ResourceCatalog.css';

interface ConsumablesPanelProps {
  missionId: string;
}

export function ConsumablesPanel({ missionId }: ConsumablesPanelProps) {
  const [consumables, setConsumables] = useState<Consumable[]>([]);
  const [showLowStockOnly, setShowLowStockOnly] = useState(false);
  const [loading, setLoading] = useState(true);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadConsumables();
  }, [missionId, showLowStockOnly]);

  const loadConsumables = async () => {
    setLoading(true);
    const list = await resourceService.listConsumables(missionId, {
      belowReorderLevel: showLowStockOnly
    }, userDID);
    setConsumables(list);
    setLoading(false);
  };

  const updateQuantity = async (id: string, currentQty: number) => {
    const newQty = prompt('Enter new quantity:', currentQty.toString());
    if (newQty !== null) {
      await resourceService.updateConsumableQuantity(id, parseFloat(newQty), userDID);
      loadConsumables();
    }
  };

  const isLowStock = (c: Consumable) => c.reorderLevel && c.quantity <= c.reorderLevel;

  return (
    <div className="consumables-panel">
      <header>
        <h3>Consumables</h3>
        <label>
          <input
            type="checkbox"
            checked={showLowStockOnly}
            onChange={e => setShowLowStockOnly(e.target.checked)}
          />
          Low stock only
        </label>
      </header>

      {loading ? (
        <div className="loading">Loading...</div>
      ) : (
        <div className="consumables-grid">
          {consumables.map(item => (
            <div
              key={item.id}
              className={`consumable-card ${isLowStock(item) ? 'low-stock' : ''}`}
            >
              <h4>{item.name}</h4>
              <div className="quantity">
                <span className="value">{item.quantity}</span>
                <span className="unit">{item.unit || 'units'}</span>
              </div>
              {item.reorderLevel && (
                <div className="reorder-level">
                  Reorder at: {item.reorderLevel}
                </div>
              )}
              <button onClick={() => updateQuantity(item.id, item.quantity)}>
                Update Qty
              </button>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
