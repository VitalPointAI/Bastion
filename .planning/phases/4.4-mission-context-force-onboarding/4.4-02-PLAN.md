```typescript
// backend/src/mission/service.ts
import { MissionStore } from './store.js';
import { createActor } from 'xstate';
import { missionLifecycleMachine } from './mission-machine.js';

export class MissionService {
  private store: MissionStore;

  constructor() {
    this.store = new MissionStore();
  }

  async initialize(): Promise<void> {
    await this.store.ensureTablesExist();
  }

  // Mission CRUD
  async createMission(data: {
    name: string;
    description?: string;
    workspaceId?: string;
    areaOfOps?: GeoJSON.Polygon;
    createdBy: string;
  }): Promise<Mission> {
    // Validate AO if provided (basic GeoJSON structure check)
    if (data.areaOfOps) {
      this.validateGeoJSON(data.areaOfOps);
    }
    return this.store.createMission(data);
  }

  async getMission(id: string): Promise<Mission | null> {
    return this.store.getMission(id);
  }

  async getMissionWithParticipants(id: string): Promise<MissionWithParticipants | null> {
    const mission = await this.store.getMission(id);
    if (!mission) return null;
    const participants = await this.store.getParticipants(id);
    return { ...mission, participants };
  }

  async listMissions(filters?: { status?: string; createdBy?: string }): Promise<Mission[]> {
    return this.store.listMissions(filters);
  }

  async updateMission(id: string, data: Partial<Mission>): Promise<Mission> {
    return this.store.updateMission(id, data);
  }

  // Status transitions via XState
  async transitionMission(id: string, event: 'ACTIVATE' | 'COMPLETE' | 'ARCHIVE' | 'REOPEN'): Promise<Mission> {
    const mission = await this.store.getMission(id);
    if (!mission) throw new Error('Mission not found');

    // Create actor with current state
    const actor = createActor(missionLifecycleMachine, {
      snapshot: { value: mission.status, context: { missionId: id } }
    });

    // Check if transition is valid
    const nextState = actor.getSnapshot();
    if (!nextState.can({ type: event })) {
      throw new Error(`Invalid transition: ${mission.status} -> ${event}`);
    }

    // Map event to new status
    const statusMap: Record<string, string> = {
      'ACTIVATE': 'active',
      'COMPLETE': 'complete',
      'ARCHIVE': 'archived',
      'REOPEN': 'active'
    };

    return this.store.setMissionStatus(id, statusMap[event]);
  }

  // Participant management
  async addParticipant(missionId: string, participantDID: string, role: string, invitedBy: string): Promise<MissionParticipant> {
    return this.store.addParticipant(missionId, participantDID, role, invitedBy);
  }

  async removeParticipant(missionId: string, participantDID: string): Promise<void> {
    return this.store.removeParticipant(missionId, participantDID);
  }

  async getParticipants(missionId: string): Promise<MissionParticipant[]> {
    return this.store.getParticipants(missionId);
  }

  // Invite token workflow
  async createInvite(missionId: string, expirationHours: number = 72): Promise<{ token: string; expiresAt: Date }> {
    const { token, expiresAt } = await this.store.generateInviteToken(missionId, expirationHours);
    return { token, expiresAt };
  }

  async redeemInvite(token: string, redeemerDID: string): Promise<Mission> {
    // Validate token
    const isValid = await this.store.validateInviteToken(token);
    if (!isValid) {
      throw new Error('Invalid or expired invite token');
    }

    // Redeem and add as participant
    const mission = await this.store.redeemInviteToken(token, redeemerDID);
    await this.store.addParticipant(mission.id, redeemerDID, 'member', 'invite');

    return mission;
  }

  // Duplication
  async duplicateMission(id: string, options: {
    newName: string;
    includeResources?: boolean;
    createdBy: string;
  }): Promise<Mission> {
    const original = await this.getMission(id);
    if (!original) throw new Error('Mission not found');

    // Create new mission with copied structure
    const newMission = await this.createMission({
      name: options.newName,
      description: original.description,
      workspaceId: original.workspaceId,
      areaOfOps: original.areaOfOps,
      createdBy: options.createdBy
    });

    // TODO: If includeResources, copy resource assignments (Phase 4.4-06)

    return newMission;
  }

  private validateGeoJSON(geojson: any): void {
    if (!geojson.type || !geojson.coordinates) {
      throw new Error('Invalid GeoJSON: missing type or coordinates');
    }
    if (geojson.type !== 'Polygon' && geojson.type !== 'MultiPolygon') {
      throw new Error('Area of Operations must be a Polygon or MultiPolygon');
    }
  }
}

// Lazy singleton pattern
let missionService: MissionService | null = null;

export async function getMissionService(): Promise<MissionService> {
  if (!missionService) {
    missionService = new MissionService();
    await missionService.initialize();
  }
  return missionService;
}
```

```typescript
// backend/src/api/mission.ts
import { Router, Request, Response } from 'express';
import { getMissionService } from '../mission/service.js';

export const missionRouter = Router();

// GET /api/missions - List missions
missionRouter.get('/', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const missions = await service.listMissions({
      status: req.query.status as string,
      createdBy: req.query.createdBy as string
    });
    res.json({ missions });
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// GET /api/missions/:id - Get mission with participants
missionRouter.get('/:id', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const mission = await service.getMissionWithParticipants(req.params.id as string);
    if (!mission) {
      return res.status(404).json({ error: 'Mission not found' });
    }
    res.json(mission);
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// POST /api/missions - Create mission
missionRouter.post('/', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const createdBy = req.headers['x-did'] as string || req.query.did as string;
    if (!createdBy) {
      return res.status(401).json({ error: 'DID required' });
    }

    const mission = await service.createMission({
      ...req.body,
      createdBy
    });
    res.status(201).json(mission);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// PATCH /api/missions/:id - Update mission
missionRouter.patch('/:id', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const mission = await service.updateMission(req.params.id as string, req.body);
    res.json(mission);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// POST /api/missions/:id/transition - Change status
missionRouter.post('/:id/transition', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const { event } = req.body;
    if (!['ACTIVATE', 'COMPLETE', 'ARCHIVE', 'REOPEN'].includes(event)) {
      return res.status(400).json({ error: 'Invalid event' });
    }
    const mission = await service.transitionMission(req.params.id as string, event);
    res.json(mission);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// POST /api/missions/:id/duplicate - Duplicate mission
missionRouter.post('/:id/duplicate', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const createdBy = req.headers['x-did'] as string || req.query.did as string;
    const mission = await service.duplicateMission(req.params.id as string, {
      newName: req.body.newName,
      includeResources: req.body.includeResources,
      createdBy
    });
    res.status(201).json(mission);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// --- Participants ---

// GET /api/missions/:id/participants
missionRouter.get('/:id/participants', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const participants = await service.getParticipants(req.params.id as string);
    res.json({ participants });
  } catch (error) {
    res.status(500).json({ error: (error as Error).message });
  }
});

// POST /api/missions/:id/participants - Add participant directly
missionRouter.post('/:id/participants', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const invitedBy = req.headers['x-did'] as string || 'system';
    const { participantDID, role } = req.body;

    const participant = await service.addParticipant(
      req.params.id as string,
      participantDID,
      role || 'member',
      invitedBy
    );
    res.status(201).json(participant);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// DELETE /api/missions/:id/participants/:did - Remove participant
missionRouter.delete('/:id/participants/:did', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    await service.removeParticipant(req.params.id as string, req.params.did as string);
    res.status(204).send();
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// --- Invites ---

// POST /api/missions/:id/invites - Generate invite token
missionRouter.post('/:id/invites', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const expirationHours = req.body.expirationHours || 72;
    const invite = await service.createInvite(req.params.id as string, expirationHours);
    res.status(201).json(invite);
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});

// POST /api/missions/join - Redeem invite token
missionRouter.post('/join', async (req: Request, res: Response) => {
  try {
    const service = await getMissionService();
    const redeemerDID = req.headers['x-did'] as string || req.query.did as string;
    if (!redeemerDID) {
      return res.status(401).json({ error: 'DID required' });
    }

    const mission = await service.redeemInvite(req.body.token, redeemerDID);
    res.json({ mission, message: 'Successfully joined mission' });
  } catch (error) {
    res.status(400).json({ error: (error as Error).message });
  }
});
```
