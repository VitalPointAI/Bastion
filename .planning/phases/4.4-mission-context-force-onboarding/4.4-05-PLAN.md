// frontend/src/lib/command-service.ts
import axios from 'axios';

const API_BASE = import.meta.env.VITE_API_URL || 'http://localhost:3001';

export interface Unit {
  id: string;
  missionId: string;
  name: string;
  unitType?: string;
  sidc?: string;
  parentDID?: string;
  createdAt: string;
}

export interface CommandRelationship {
  id: string;
  missionId: string;
  superiorUnitId: string;
  subordinateUnitId: string;
  relationshipType: 'OPCON' | 'TACON' | 'ADCON' | 'COCOM' | 'DS' | 'GS' | 'GSR' | 'R';
  effectiveFrom?: string;
  effectiveTo?: string;
  notes?: string;
  createdAt: string;
}

export interface JointResponsibility {
  id: string;
  unitId: string;
  responsibility: string;
  assignedAt: string;
}

export interface HierarchyNode {
  id: string;
  name: string;
  unitType?: string;
  sidc?: string;
  depth: number;
  path: string[];
}

export interface TreeNode extends HierarchyNode {
  children: TreeNode[];
  relationships: string[];
  responsibilities?: string[];
}

export interface CreateRelationshipResult {
  relationship: CommandRelationship;
  warnings: string[];
}

class CommandService {
  private getHeaders(userDID?: string) {
    const headers: Record<string, string> = { 'Content-Type': 'application/json' };
    if (userDID) headers['X-DID'] = userDID;
    return headers;
  }

  // --- Units ---

  async listUnits(missionId: string, userDID: string): Promise<Unit[]> {
    const response = await axios.get(
      `${API_BASE}/api/command/missions/${missionId}/units`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.units;
  }

  async getUnit(id: string, userDID: string): Promise<Unit> {
    const response = await axios.get(
      `${API_BASE}/api/command/units/${id}`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async createUnit(missionId: string, data: Partial<Unit>, userDID: string): Promise<Unit> {
    const response = await axios.post(
      `${API_BASE}/api/command/missions/${missionId}/units`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async updateUnit(id: string, data: Partial<Unit>, userDID: string): Promise<Unit> {
    const response = await axios.patch(
      `${API_BASE}/api/command/units/${id}`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async deleteUnit(id: string, userDID: string): Promise<void> {
    await axios.delete(
      `${API_BASE}/api/command/units/${id}`,
      { headers: this.getHeaders(userDID) }
    );
  }

  // --- Relationships ---

  async listRelationships(missionId: string, userDID: string): Promise<CommandRelationship[]> {
    const response = await axios.get(
      `${API_BASE}/api/command/missions/${missionId}/relationships`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.relationships;
  }

  async createRelationship(
    missionId: string,
    data: Omit<CommandRelationship, 'id' | 'missionId' | 'createdAt'>,
    userDID: string
  ): Promise<CreateRelationshipResult> {
    const response = await axios.post(
      `${API_BASE}/api/command/missions/${missionId}/relationships`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async updateRelationship(id: string, data: Partial<CommandRelationship>, userDID: string): Promise<CommandRelationship> {
    const response = await axios.patch(
      `${API_BASE}/api/command/relationships/${id}`,
      data,
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async deleteRelationship(id: string, userDID: string): Promise<void> {
    await axios.delete(
      `${API_BASE}/api/command/relationships/${id}`,
      { headers: this.getHeaders(userDID) }
    );
  }

  // --- Hierarchy ---

  async getHierarchy(missionId: string, format: 'flat' | 'tree', userDID: string): Promise<HierarchyNode[] | TreeNode[]> {
    const response = await axios.get(
      `${API_BASE}/api/command/missions/${missionId}/hierarchy?format=${format}`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.hierarchy;
  }

  async validateHierarchy(missionId: string, userDID: string): Promise<{ valid: boolean; warnings: string[] }> {
    const response = await axios.post(
      `${API_BASE}/api/command/missions/${missionId}/hierarchy/validate`,
      {},
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  // --- Responsibilities ---

  async getResponsibilities(unitId: string, userDID: string): Promise<JointResponsibility[]> {
    const response = await axios.get(
      `${API_BASE}/api/command/units/${unitId}/responsibilities`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.responsibilities;
  }

  async assignResponsibility(unitId: string, responsibility: string, userDID: string): Promise<JointResponsibility> {
    const response = await axios.post(
      `${API_BASE}/api/command/units/${unitId}/responsibilities`,
      { responsibility },
      { headers: this.getHeaders(userDID) }
    );
    return response.data;
  }

  async removeResponsibility(unitId: string, responsibility: string, userDID: string): Promise<void> {
    await axios.delete(
      `${API_BASE}/api/command/units/${unitId}/responsibilities/${responsibility}`,
      { headers: this.getHeaders(userDID) }
    );
  }

  async getMultiHattedPositions(missionId: string, userDID: string): Promise<{ unit: Unit; responsibilities: string[] }[]> {
    const response = await axios.get(
      `${API_BASE}/api/command/missions/${missionId}/multi-hatted`,
      { headers: this.getHeaders(userDID) }
    );
    return response.data.positions;
  }
}

// UnitNode.tsx
import { useMemo } from 'react';
import ms from 'milsymbol';
import { Unit, JointResponsibility, JOINT_RESPONSIBILITY_LABELS } from '../../../lib/command-service';
import './CommandEditor.css';

interface UnitNodeProps {
  unit: Unit;
  responsibilities?: string[];
  relationshipType?: string;
  isSelected?: boolean;
  onClick?: () => void;
  onDragStart?: (e: React.DragEvent) => void;
  onDrop?: (e: React.DragEvent) => void;
}

export function UnitNode({
  unit,
  responsibilities = [],
  relationshipType,
  isSelected,
  onClick,
  onDragStart,
  onDrop
}: UnitNodeProps) {
  // Generate milsymbol if SIDC provided
  const symbolSvg = useMemo(() => {
    if (!unit.sidc) return null;
    try {
      const symbol = new ms.Symbol(unit.sidc, { size: 30 });
      return symbol.asSVG();
    } catch {
      return null;
    }
  }, [unit.sidc]);

  const isMultiHatted = responsibilities.length > 1;

  return (
    <div
      className={`unit-node ${isSelected ? 'selected' : ''} ${isMultiHatted ? 'multi-hatted' : ''}`}
      onClick={onClick}
      draggable
      onDragStart={onDragStart}
      onDragOver={(e) => e.preventDefault()}
      onDrop={onDrop}
    >
      {symbolSvg && (
        <div
          className="unit-symbol"
          dangerouslySetInnerHTML={{ __html: symbolSvg }}
        />
      )}
      <div className="unit-name">{unit.name}</div>
      {unit.unitType && (
        <div className="unit-type">{unit.unitType}</div>
      )}
      {relationshipType && (
        <div className={`relationship-badge ${relationshipType.toLowerCase()}`}>
          {relationshipType}
        </div>
      )}
      {responsibilities.length > 0 && (
        <div className="responsibility-badges">
          {responsibilities.map(r => (
            <span key={r} className="responsibility-badge">
              {JOINT_RESPONSIBILITY_LABELS[r] || r}
            </span>
          ))}
        </div>
      )}
    </div>
  );
}

// CommandTreeView.tsx
import { useEffect, useState, useContext } from 'react';
import { Tree, TreeNode } from 'react-organizational-chart';
import { UserContext } from '../../AuthWrapper';
import { commandService, TreeNode as CommandTreeNode, Unit } from '../../../lib/command-service';
import { UnitNode } from './UnitNode';
import './CommandEditor.css';

interface CommandTreeViewProps {
  missionId: string;
  selectedUnitId?: string;
  onSelectUnit: (unitId: string) => void;
  onCreateRelationship: (superiorId: string, subordinateId: string) => void;
}

export function CommandTreeView({
  missionId,
  selectedUnitId,
  onSelectUnit,
  onCreateRelationship
}: CommandTreeViewProps) {
  const [hierarchy, setHierarchy] = useState<CommandTreeNode[]>([]);
  const [units, setUnits] = useState<Map<string, Unit>>(new Map());
  const [loading, setLoading] = useState(true);
  const { userDID } = useContext(UserContext);
  const [draggedUnitId, setDraggedUnitId] = useState<string | null>(null);

  useEffect(() => {
    loadHierarchy();
  }, [missionId]);

  const loadHierarchy = async () => {
    setLoading(true);
    try {
      const [tree, unitList] = await Promise.all([
        commandService.getHierarchy(missionId, 'tree', userDID) as Promise<CommandTreeNode[]>,
        commandService.listUnits(missionId, userDID)
      ]);
      setHierarchy(tree);
      setUnits(new Map(unitList.map(u => [u.id, u])));
    } catch (error) {
      console.error('Failed to load hierarchy:', error);
    }
    setLoading(false);
  };

  const handleDragStart = (unitId: string) => (e: React.DragEvent) => {
    setDraggedUnitId(unitId);
    e.dataTransfer.setData('text/plain', unitId);
  };

  const handleDrop = (targetUnitId: string) => (e: React.DragEvent) => {
    e.preventDefault();
    const sourceUnitId = e.dataTransfer.getData('text/plain');
    if (sourceUnitId && sourceUnitId !== targetUnitId) {
      // Create relationship: target is superior, dragged is subordinate
      onCreateRelationship(targetUnitId, sourceUnitId);
    }
    setDraggedUnitId(null);
  };

  const renderNode = (node: CommandTreeNode): JSX.Element => {
    const unit = units.get(node.id);
    if (!unit) return <div key={node.id}>Unknown unit</div>;

    return (
      <TreeNode
        key={node.id}
        label={
          <UnitNode
            unit={unit}
            responsibilities={node.responsibilities}
            isSelected={selectedUnitId === node.id}
            onClick={() => onSelectUnit(node.id)}
            onDragStart={handleDragStart(node.id)}
            onDrop={handleDrop(node.id)}
          />
        }
      >
        {node.children.map(child => renderNode(child))}
      </TreeNode>
    );
  };

  if (loading) {
    return <div className="loading">Loading hierarchy...</div>;
  }

  if (hierarchy.length === 0) {
    return (
      <div className="empty-hierarchy">
        <p>No units in hierarchy. Add units and create relationships to build the command structure.</p>
      </div>
    );
  }

  // Render multiple root nodes if needed
  return (
    <div className="command-tree-view">
      {hierarchy.map(root => {
        const unit = units.get(root.id);
        if (!unit) return null;

        return (
          <Tree
            key={root.id}
            lineWidth="2px"
            lineColor="var(--accent-blue)"
            lineBorderRadius="4px"
            label={
              <UnitNode
                unit={unit}
                responsibilities={root.responsibilities}
                isSelected={selectedUnitId === root.id}
                onClick={() => onSelectUnit(root.id)}
                onDragStart={handleDragStart(root.id)}
                onDrop={handleDrop(root.id)}
              />
            }
          >
            {root.children.map(child => renderNode(child))}
          </Tree>
        );
      })}
    </div>
  );
}

// CommandMatrixView.tsx
import { useEffect, useState, useContext } from 'react';
import { UserContext } from '../../AuthWrapper';
import { commandService, Unit, CommandRelationship, RELATIONSHIP_LABELS } from '../../../lib/command-service';
import './CommandEditor.css';

interface CommandMatrixViewProps {
  missionId: string;
  onSelectRelationship: (relationshipId: string) => void;
  onDeleteRelationship: (relationshipId: string) => void;
}

export function CommandMatrixView({
  missionId,
  onSelectRelationship,
  onDeleteRelationship
}: CommandMatrixViewProps) {
  const [units, setUnits] = useState<Unit[]>([]);
  const [relationships, setRelationships] = useState<CommandRelationship[]>([]);
  const [loading, setLoading] = useState(true);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadData();
  }, [missionId]);

  const loadData = async () => {
    setLoading(true);
    try {
      const [unitList, relList] = await Promise.all([
        commandService.listUnits(missionId, userDID),
        commandService.listRelationships(missionId, userDID)
      ]);
      setUnits(unitList);
      setRelationships(relList);
    } catch (error) {
      console.error('Failed to load data:', error);
    }
    setLoading(false);
  };

  const getUnitName = (id: string) => units.find(u => u.id === id)?.name || 'Unknown';

  if (loading) return <div className="loading">Loading...</div>;

  return (
    <div className="command-matrix-view">
      <table>
        <thead>
          <tr>
            <th>Superior</th>
            <th>Relationship</th>
            <th>Subordinate</th>
            <th>Effective</th>
            <th>Actions</th>
          </tr>
        </thead>
        <tbody>
          {relationships.map(rel => (
            <tr key={rel.id} onClick={() => onSelectRelationship(rel.id)}>
              <td>{getUnitName(rel.superiorUnitId)}</td>
              <td>
                <span className={`relationship-type ${rel.relationshipType.toLowerCase()}`}>
                  {RELATIONSHIP_LABELS[rel.relationshipType]}
                </span>
              </td>
              <td>{getUnitName(rel.subordinateUnitId)}</td>
              <td>
                {rel.effectiveFrom ? new Date(rel.effectiveFrom).toLocaleDateString() : 'Always'}
                {rel.effectiveTo && ` - ${new Date(rel.effectiveTo).toLocaleDateString()}`}
              </td>
              <td>
                <button
                  className="btn-delete"
                  onClick={(e) => { e.stopPropagation(); onDeleteRelationship(rel.id); }}
                >
                  Delete
                </button>
              </td>
            </tr>
          ))}
          {relationships.length === 0 && (
            <tr>
              <td colSpan={5}>No relationships defined</td>
            </tr>
          )}
        </tbody>
      </table>
    </div>
  );
}

// RelationshipForm.tsx
import { useState, useContext } from 'react';
import { useForm } from 'react-hook-form';
import { UserContext } from '../../AuthWrapper';
import { commandService, Unit, RELATIONSHIP_LABELS } from '../../../lib/command-service';
import './CommandEditor.css';

interface RelationshipFormProps {
  missionId: string;
  units: Unit[];
  presetSuperiorId?: string;
  presetSubordinateId?: string;
  onSubmit: () => void;
  onCancel: () => void;
}

interface FormData {
  superiorUnitId: string;
  subordinateUnitId: string;
  relationshipType: string;
  effectiveFrom?: string;
  effectiveTo?: string;
  notes?: string;
}

export function RelationshipForm({
  missionId,
  units,
  presetSuperiorId,
  presetSubordinateId,
  onSubmit,
  onCancel
}: RelationshipFormProps) {
  const { register, handleSubmit, formState: { errors } } = useForm<FormData>({
    defaultValues: {
      superiorUnitId: presetSuperiorId || '',
      subordinateUnitId: presetSubordinateId || '',
      relationshipType: 'OPCON'
    }
  });
  const [submitting, setSubmitting] = useState(false);
  const [warnings, setWarnings] = useState<string[]>([]);
  const { userDID } = useContext(UserContext);

  const submit = async (data: FormData) => {
    setSubmitting(true);
    setWarnings([]);
    try {
      const result = await commandService.createRelationship(missionId, {
        superiorUnitId: data.superiorUnitId,
        subordinateUnitId: data.subordinateUnitId,
        relationshipType: data.relationshipType as any,
        effectiveFrom: data.effectiveFrom,
        effectiveTo: data.effectiveTo,
        notes: data.notes
      }, userDID);

      if (result.warnings.length > 0) {
        setWarnings(result.warnings);
      }
      onSubmit();
    } catch (error) {
      alert((error as Error).message);
    }
    setSubmitting(false);
  };

  return (
    <form className="relationship-form" onSubmit={handleSubmit(submit)}>
      <h3>Create Command Relationship</h3>

      <div className="form-group">
        <label>Superior Unit</label>
        <select {...register('superiorUnitId', { required: true })}>
          <option value="">Select unit...</option>
          {units.map(u => (
            <option key={u.id} value={u.id}>{u.name}</option>
          ))}
        </select>
      </div>

      <div className="form-group">
        <label>Relationship Type</label>
        <select {...register('relationshipType', { required: true })}>
          {Object.entries(RELATIONSHIP_LABELS).map(([key, label]) => (
            <option key={key} value={key}>{label} ({key})</option>
          ))}
        </select>
      </div>

      <div className="form-group">
        <label>Subordinate Unit</label>
        <select {...register('subordinateUnitId', { required: true })}>
          <option value="">Select unit...</option>
          {units.map(u => (
            <option key={u.id} value={u.id}>{u.name}</option>
          ))}
        </select>
      </div>

      <div className="form-row">
        <div className="form-group">
          <label>Effective From (optional)</label>
          <input type="date" {...register('effectiveFrom')} />
        </div>
        <div className="form-group">
          <label>Effective To (optional)</label>
          <input type="date" {...register('effectiveTo')} />
        </div>
      </div>

      <div className="form-group">
        <label>Notes (optional)</label>
        <textarea {...register('notes')} rows={2} />
      </div>

      {warnings.length > 0 && (
        <div className="warnings">
          {warnings.map((w, i) => (
            <div key={i} className="warning">{w}</div>
          ))}
        </div>
      )}

      <div className="form-actions">
        <button type="button" onClick={onCancel}>Cancel</button>
        <button type="submit" disabled={submitting}>
          {submitting ? 'Creating...' : 'Create Relationship'}
        </button>
      </div>
    </form>
  );
}

  const [presetRelationship, setPresetRelationship] = useState<{superiorId?: string; subordinateId?: string}>({});
  const [refreshKey, setRefreshKey] = useState(0);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadUnits();
  }, [missionId, refreshKey]);

  const loadUnits = async () => {
    const unitList = await commandService.listUnits(missionId, userDID);
    setUnits(unitList);
  };

  const handleCreateRelationship = (superiorId: string, subordinateId: string) => {
    setPresetRelationship({ superiorId, subordinateId });
    setShowAddRelationship(true);
  };

  const handleDeleteRelationship = async (relationshipId: string) => {
    if (confirm('Delete this relationship?')) {
      await commandService.deleteRelationship(relationshipId, userDID);
      setRefreshKey(k => k + 1);
    }
  };

  const refresh = () => {
    setShowAddUnit(false);
    setShowAddRelationship(false);
    setPresetRelationship({});
    setRefreshKey(k => k + 1);
  };

  return (
    <div className="command-editor">
      <header>
        <h2>Command Hierarchy</h2>
        <div className="view-toggle">
          <button
            className={viewMode === 'tree' ? 'active' : ''}
            onClick={() => setViewMode('tree')}
          >
            Tree View
          </button>
          <button
            className={viewMode === 'matrix' ? 'active' : ''}
            onClick={() => setViewMode('matrix')}
          >
            Matrix View
          </button>
        </div>
        <div className="actions">
          <button onClick={() => setShowAddUnit(true)}>+ Add Unit</button>
          <button onClick={() => setShowAddRelationship(true)}>+ Add Relationship</button>
        </div>
      </header>

      {viewMode === 'tree' ? (
        <CommandTreeView
          key={refreshKey}
          missionId={missionId}
          selectedUnitId={selectedUnitId || undefined}
          onSelectUnit={setSelectedUnitId}
          onCreateRelationship={handleCreateRelationship}
        />
      ) : (
        <CommandMatrixView
          key={refreshKey}
          missionId={missionId}
          onSelectRelationship={() => {}}
          onDeleteRelationship={handleDeleteRelationship}
        />
      )}

      {showAddUnit && (
        <div className="modal-overlay">
          <div className="modal">
            <UnitForm
              missionId={missionId}
              onSubmit={refresh}
              onCancel={() => setShowAddUnit(false)}
            />
          </div>
        </div>
      )}

      {showAddRelationship && (
        <div className="modal-overlay">
          <div className="modal">
            <RelationshipForm
              missionId={missionId}
              units={units}
              presetSuperiorId={presetRelationship.superiorId}
              presetSubordinateId={presetRelationship.subordinateId}
              onSubmit={refresh}
              onCancel={() => { setShowAddRelationship(false); setPresetRelationship({}); }}
            />
          </div>
        </div>
      )}
    </div>
  );
}
