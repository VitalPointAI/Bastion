```typescript
// frontend/src/lib/mission-service.ts
import axios from 'axios';

const API_BASE = import.meta.env.VITE_API_URL || 'http://localhost:3001';

export interface Mission {
  id: string;
  name: string;
  description?: string;
  workspaceId?: string;
  status: 'planning' | 'active' | 'complete' | 'archived';
  areaOfOps?: GeoJSON.Polygon;
  createdBy: string;
  createdAt: string;
  updatedAt: string;
}

export interface MissionParticipant {
  id: string;
  missionId: string;
  participantDID: string;
  role: string;
  joinedAt: string;
  invitedBy: string;
}

export interface MissionWithParticipants extends Mission {
  participants: MissionParticipant[];
}

export interface CreateMissionData {
  name: string;
  description?: string;
  workspaceId?: string;
  areaOfOps?: GeoJSON.Polygon;
}

export interface InviteToken {
  token: string;
  expiresAt: string;
}

class MissionService {
  private getHeaders(userDID?: string) {
    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
    };
    if (userDID) {
      headers['X-DID'] = userDID;
    }
    return headers;
  }

  async listMissions(userDID: string, filters?: { status?: string }): Promise<Mission[]> {
    const params = new URLSearchParams();
    if (filters?.status) params.set('status', filters.status);

    const response = await axios.get(`${API_BASE}/api/missions?${params}`, {
      headers: this.getHeaders(userDID),
    });
    return response.data.missions;
  }

  async getMission(id: string, userDID: string): Promise<MissionWithParticipants> {
    const response = await axios.get(`${API_BASE}/api/missions/${id}`, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  async createMission(data: CreateMissionData, userDID: string): Promise<Mission> {
    const response = await axios.post(`${API_BASE}/api/missions`, data, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  async updateMission(id: string, data: Partial<Mission>, userDID: string): Promise<Mission> {
    const response = await axios.patch(`${API_BASE}/api/missions/${id}`, data, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  async transitionMission(id: string, event: 'ACTIVATE' | 'COMPLETE' | 'ARCHIVE' | 'REOPEN', userDID: string): Promise<Mission> {
    const response = await axios.post(`${API_BASE}/api/missions/${id}/transition`, { event }, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  async duplicateMission(id: string, newName: string, includeResources: boolean, userDID: string): Promise<Mission> {
    const response = await axios.post(`${API_BASE}/api/missions/${id}/duplicate`, {
      newName,
      includeResources,
    }, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  // Participants
  async getParticipants(missionId: string, userDID: string): Promise<MissionParticipant[]> {
    const response = await axios.get(`${API_BASE}/api/missions/${missionId}/participants`, {
      headers: this.getHeaders(userDID),
    });
    return response.data.participants;
  }

  async addParticipant(missionId: string, participantDID: string, role: string, userDID: string): Promise<MissionParticipant> {
    const response = await axios.post(`${API_BASE}/api/missions/${missionId}/participants`, {
      participantDID,
      role,
    }, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  async removeParticipant(missionId: string, participantDID: string, userDID: string): Promise<void> {
    await axios.delete(`${API_BASE}/api/missions/${missionId}/participants/${encodeURIComponent(participantDID)}`, {
      headers: this.getHeaders(userDID),
    });
  }

  // Invites
  async createInvite(missionId: string, expirationHours: number, userDID: string): Promise<InviteToken> {
    const response = await axios.post(`${API_BASE}/api/missions/${missionId}/invites`, {
      expirationHours,
    }, {
      headers: this.getHeaders(userDID),
    });
    return response.data;
  }

  async redeemInvite(token: string, userDID: string): Promise<Mission> {
    const response = await axios.post(`${API_BASE}/api/missions/join`, { token }, {
      headers: this.getHeaders(userDID),
    });
    return response.data.mission;
  }
}

export const missionService = new MissionService();
```

```typescript
  name: string;
  description: string;
  participants: { did: string; role: string }[];
  areaOfOps: GeoJSON.Polygon | null;
}

const STEPS = ['Name', 'Participants', 'Area of Ops', 'Review'];

export function MissionWizard({ onComplete, onCancel }: Props) {
  const [currentStep, setCurrentStep] = useState(0);
  const [data, setData] = useState<WizardState>({ ... });
  const [submitting, setSubmitting] = useState(false);
  const { userDID } = useContext(UserContext);

  // Validation per step
  const canProceed = () => { ... };

  // Navigation
  const next = () => canProceed() && setCurrentStep(s => s + 1);
  const back = () => setCurrentStep(s => s - 1);

  // Submit
  const submit = async () => {
    setSubmitting(true);
    const mission = await missionService.createMission(data, userDID);
    // Add participants
    for (const p of data.participants) {
      await missionService.addParticipant(mission.id, p.did, p.role, userDID);
    }
    onComplete(mission);
  };

  // Render current step
  return (
    <div className="mission-wizard">
      <StepIndicator steps={STEPS} current={currentStep} />
      {currentStep === 0 && <NameStep data={data} onChange={...} />}
      {currentStep === 1 && <ParticipantsStep data={data} onChange={...} />}
      {currentStep === 2 && <AreaOfOpsStep data={data} onChange={...} />}
      {currentStep === 3 && <ReviewStep data={data} />}
      <WizardNavigation ... />
    </div>
  );
}
```

```typescript
import { MapContainer, TileLayer, FeatureGroup } from 'react-leaflet';
import { EditControl } from 'react-leaflet-draw';
import 'leaflet-draw/dist/leaflet.draw.css';

export function AreaOfOpsStep({ areaOfOps, onChange }: Props) {
  const handleCreated = (e: any) => {
    const layer = e.layer;
    const geojson = layer.toGeoJSON();
    onChange(geojson.geometry);
  };

  return (
    <div className="area-of-ops-step">
      <h3>Define Area of Operations</h3>
      <p>Draw a polygon on the map to define the mission boundary.</p>
      <MapContainer center={[0, 0]} zoom={2} style={{ height: '400px' }}>
        <TileLayer url="https://tiles.stadiamaps.com/tiles/alidade_smooth_dark/{z}/{x}/{y}{r}.png" />
        <FeatureGroup>
          <EditControl
            position="topright"
            onCreated={handleCreated}
            draw={{
              rectangle: false,
              circle: false,
              circlemarker: false,
              marker: false,
              polyline: false,
              polygon: { allowIntersection: false }
            }}
          />
        </FeatureGroup>
        {areaOfOps && <GeoJSON data={areaOfOps} />}
      </MapContainer>
    </div>
  );
}
```

```typescript
// MissionList.tsx
export function MissionList({ onSelectMission, onCreateMission }: Props) {
  const [missions, setMissions] = useState<Mission[]>([]);
  const [statusFilter, setStatusFilter] = useState<string>('all');
  const [loading, setLoading] = useState(true);
  const { userDID } = useContext(UserContext);

  useEffect(() => {
    loadMissions();
  }, [statusFilter]);

  const loadMissions = async () => {
    setLoading(true);
    const filter = statusFilter === 'all' ? undefined : { status: statusFilter };
    const data = await missionService.listMissions(userDID, filter);
    setMissions(data);
    setLoading(false);
  };

  return (
    <div className="mission-list">
      <header>
        <h2>Missions</h2>
        <button onClick={onCreateMission}>+ New Mission</button>
      </header>
      <div className="filters">
        <select value={statusFilter} onChange={e => setStatusFilter(e.target.value)}>
          <option value="all">All Missions</option>
          <option value="planning">Planning</option>
          <option value="active">Active</option>
          <option value="complete">Complete</option>
          <option value="archived">Archived</option>
        </select>
      </div>
      <div className="mission-grid">
        {missions.map(m => (
          <MissionCard key={m.id} mission={m} onClick={() => onSelectMission(m.id)} />
        ))}
      </div>
    </div>
  );
}
```

```typescript
type ViewState = 'list' | 'wizard' | 'detail';

export function MissionDashboard() {
  const [view, setView] = useState<ViewState>('list');
  const [selectedMissionId, setSelectedMissionId] = useState<string | null>(null);

  const handleSelectMission = (id: string) => {
    setSelectedMissionId(id);
    setView('detail');
  };

  const handleCreateMission = () => {
    setView('wizard');
  };

  const handleWizardComplete = (mission: Mission) => {
    setSelectedMissionId(mission.id);
    setView('detail');
  };

  return (
    <div className="mission-dashboard">
      {view === 'list' && (
        <MissionList
          onSelectMission={handleSelectMission}
          onCreateMission={handleCreateMission}
        />
      )}
      {view === 'wizard' && (
        <MissionWizard
          onComplete={handleWizardComplete}
          onCancel={() => setView('list')}
        />
      )}
      {view === 'detail' && selectedMissionId && (
        <MissionDetail
          missionId={selectedMissionId}
          onBack={() => setView('list')}
        />
      )}
    </div>
  );
}
```
