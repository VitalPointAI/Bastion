**Backend — Add PATCH endpoint in `backend/src/api/resources.ts`:**
Add after the existing `router.delete('/groups/:groupId', ...)` route:
```typescript
router.patch('/groups/:groupId', async (req: Request, res: Response) => {
  try {
    const { groupId } = req.params;
    const { name, description } = req.body;
    if (!name && !description) {
      return res.status(400).json({ error: 'Provide name or description to update' });
    }
    const store = getResourceGroupStore(); // or however the store is accessed
    const updated = await store.updateGroup(groupId, { name, description });
    res.json(updated);
  } catch (error) {
    res.status(500).json({ error: String(error) });
  }
});
```
