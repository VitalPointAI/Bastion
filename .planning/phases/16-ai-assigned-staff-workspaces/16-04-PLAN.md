// Role assignments
async getRoleAssignments(scenarioId: string): Promise<Record<string, RoleAssignment>> {
  const res = await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/role-assignments`);
  if (!res.ok) throw new Error('Failed to fetch role assignments');
  const data = await res.json();
  return data.roleAssignments;
}

async updateRoleAssignments(
  scenarioId: string,
  roleAssignments: Record<string, RoleAssignment>
): Promise<Record<string, RoleAssignment>> {
  const res = await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/role-assignments`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ roleAssignments }),
  });
  if (!res.ok) throw new Error('Failed to update role assignments');
  const data = await res.json();
  return data.roleAssignments;
}

// AI runs
async triggerAIRole(scenarioId: string, roleKey: string, payload?: Record<string, unknown>): Promise<void> {
  const res = await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/roles/${roleKey}/runs`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload ?? {}),
  });
  if (!res.ok) throw new Error('Failed to trigger AI role');
}

async getAIRuns(scenarioId: string, roleKey: string): Promise<AIRoleRun[]> {
  const res = await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/roles/${roleKey}/runs`);
  if (!res.ok) throw new Error('Failed to fetch AI runs');
  const data = await res.json();
  return data.runs;
}

async pauseAIRun(scenarioId: string, roleKey: string, runId: string): Promise<void> {
  await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/roles/${roleKey}/runs/${runId}/pause`, { method: 'PATCH' });
}

async resumeAIRun(scenarioId: string, roleKey: string, runId: string): Promise<void> {
  await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/roles/${roleKey}/runs/${runId}/resume`, { method: 'PATCH' });
}

async submitReview(
  scenarioId: string,
  roleKey: string,
  runId: string,
  feedback: ReviewFeedback
): Promise<void> {
  const res = await fetch(
    `${API_BASE}/exercise/scenarios/${scenarioId}/roles/${roleKey}/runs/${runId}/review`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(feedback),
    }
  );
  if (!res.ok) throw new Error('Failed to submit review');
}

async getAgentsForRole(scenarioId: string, roleKey: string): Promise<StaffAgentDef[]> {
  const res = await fetch(`${API_BASE}/exercise/scenarios/${scenarioId}/roles/${roleKey}/agents`);
  if (!res.ok) throw new Error('Failed to fetch agents');
  const data = await res.json();
  return data.agents;
}

      ))}
    </div>
  );
}

**`frontend/src/components/exercise/AIRoleWorkspace.tsx`:**

Main AI workspace container. Side-by-side layout: product panel (primary, wider) on left, channel feed (narrower) on right.

Props: `{ roleKey: string; scenarioId: string; exercisePhase?: string; isControllerView?: boolean }`

State:
- `agents: StaffAgentDef[]` — loaded via `getAgentsForRole` on mount
- `activeRun: AIRoleRun | null` — loaded via `getAIRuns` (latest non-complete run)
- `isBeginning: boolean` — true while triggerAIRole is in flight
- `channelEvents: AIChannelEvent[]` — populated by SSE subscription (set up in Plan 05 ChannelFeed component)

The workspace renders one of three states:
1. **Initial state** (no active run, or activeRun is complete/failed): `<AgentRosterCard>` centered in the workspace
2. **Active/running state** (activeRun status is queued|running|paused|awaiting_review): side-by-side layout with product panel + ChannelFeed (ChannelFeed is a stub `<div className="channel-feed-placeholder">` for now — Plan 05 wires the real component)
3. **Error state**: shows error message + "Try Again" button that re-triggers

```tsx
export function AIRoleWorkspace({ roleKey, scenarioId, exercisePhase, isControllerView }: AIRoleWorkspaceProps) {
  const [agents, setAgents] = useState<StaffAgentDef[]>([]);
  const [activeRun, setActiveRun] = useState<AIRoleRun | null>(null);
  const [isBeginning, setIsBeginning] = useState(false);

  useEffect(() => {
    exerciseService.getAgentsForRole(scenarioId, roleKey).then(setAgents).catch(console.error);
    exerciseService.getAIRuns(scenarioId, roleKey).then(runs => {
      const active = runs.find(r => ['queued','running','paused','awaiting_review'].includes(r.status));
      setActiveRun(active ?? null);
    }).catch(console.error);
  }, [scenarioId, roleKey]);

  const handleBegin = async () => {
    setIsBeginning(true);
    try {
      await exerciseService.triggerAIRole(scenarioId, roleKey);
      // Poll for run creation briefly then update activeRun
      setTimeout(async () => {
        const runs = await exerciseService.getAIRuns(scenarioId, roleKey);
        const active = runs.find(r => ['queued','running'].includes(r.status));
        if (active) setActiveRun(active);
        setIsBeginning(false);
      }, 1500);
    } catch (err) {
      setIsBeginning(false);
    }
  };

  const hasActiveRun = activeRun && !['complete','failed'].includes(activeRun.status);

  return (
    <div className="ai-role-workspace">
      {!hasActiveRun ? (
        <div className="ai-workspace-initial">
          <AgentRosterCard
            roleKey={roleKey}
            scenarioId={scenarioId}
            agents={agents}
            onBegin={handleBegin}
            isBeginning={isBeginning}
          />
        </div>
      ) : (
        <div className="ai-workspace-active">
          <div className="ai-product-panel">
            {/* Product panel placeholder — wired in Plan 06 (StaffWorkspace extension) */}
            <div className="aip-placeholder">Agent products will appear here as they are generated.</div>
          </div>
          <div className="ai-channel-panel">
            {/* ChannelFeed placeholder — replaced by real ChannelFeed component in Plan 05 */}
            <div className="channel-feed-placeholder" id={`channel-${roleKey}`}>
              <span className="cfp-label">Channel feed</span>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
```
