2. **Type definitions**:

   ```typescript
   /** Resource categories for sustainment tracking */
   export type ResourceCategory = 'ammunition' | 'fuel' | 'food_water' | 'medical' | 'maintenance_parts' | 'other';

   /** A single resource entry with consumption rate */
   export interface ResourceEntry {
     /** Resource category */
     category: ResourceCategory;
     /** Specific resource name (e.g., "5.56mm", "JP-8", "MRE") */
     name: string;
     /** Starting quantity */
     startingQuantity: number;
     /** Unit of measurement */
     unit: string;
     /** Consumption rate per phase (array indexed by phase number) */
     consumptionPerPhase: number[];
     /** Resupply rate per phase (if applicable) */
     resupplyPerPhase: number[];
     /** Critical threshold (below this = red flag) */
     criticalThreshold: number;
     /** Warning threshold (below this = amber flag) */
     warningThreshold: number;
   }

   /** Burndown data point for charting */
   export interface BurndownPoint {
     /** Phase number */
     phase: number;
     /** Phase name */
     phaseName: string;
     /** Remaining quantity after this phase */
     remaining: number;
     /** Consumed in this phase */
     consumed: number;
     /** Resupplied in this phase */
     resupplied: number;
     /** Risk level based on thresholds */
     riskLevel: 'green' | 'amber' | 'red';
   }

   /** Resource burndown for a single resource */
   export interface ResourceBurndown {
     /** Resource name */
     resourceName: string;
     /** Resource category */
     category: ResourceCategory;
     /** Unit */
     unit: string;
     /** Burndown points per phase */
     points: BurndownPoint[];
     /** Phase where resource is exhausted (null if sufficient) */
     exhaustionPhase: number | null;
     /** Shortfall amount if exhausted */
     shortfallAmount: number;
   }

   /** Phase-level risk assessment */
   export interface PhaseRiskAssessment {
     /** Phase number */
     phase: number;
     /** Phase name */
     phaseName: string;
     /** Overall risk for this phase */
     overallRisk: 'green' | 'amber' | 'red';
     /** Resources at risk in this phase */
     atRiskResources: Array<{
       resourceName: string;
       category: ResourceCategory;
       riskLevel: 'amber' | 'red';
       remainingPercent: number;
     }>;
     /** Sustainability assessment: can operations continue? */
     sustainable: boolean;
   }

   /** Complete sustainment model result for one COA */
   export interface SustainmentModelResult {
     /** COA identifier */
     coaId: string;
     /** COA name */
     coaName: string;
     /** Resource burndowns */
     burndowns: ResourceBurndown[];
     /** Phase risk assessments */
     phaseRisks: PhaseRiskAssessment[];
     /** Overall sustainment feasibility */
     overallFeasibility: 'feasible' | 'marginal' | 'infeasible';
     /** Phase where sustainment becomes critical (first red phase) */
     criticalPhase: number | null;
     /** Summary text */
     summary: string;
     /** Recommendations for improving sustainment */
     recommendations: string[];
   }

   /** Comparison result across multiple COAs */
   export interface SustainmentComparison {
     /** Results per COA */
     results: SustainmentModelResult[];
     /** Which COA has best sustainment profile */
     bestSustainedCoaId: string;
     /** Which COA has worst sustainment profile */
     worstSustainedCoaId: string;
     /** Resource categories that differentiate the COAs most */
     differentiatingResources: string[];
     /** Comparison summary */
     comparisonSummary: string;
   }
   ```
