     /** Assessment: strong, moderate, weak, unknown */
     assessment: 'strong' | 'moderate' | 'weak' | 'unknown';

     /** Force model identifier */
     id: string;
     /** Adversary name/designation */
     adversaryName: string;
     /** Overall force disposition: conventional, irregular, hybrid, peer */
     forceType: 'conventional' | 'irregular' | 'hybrid' | 'peer';
     /** Assessed capabilities by domain */
     capabilities: AdversaryCapability[];

     /** Known center of gravity */

     /** Critical vulnerabilities */
     criticalVulnerabilities: string[];
     /** Intelligence gaps requiring collection */
     intelligenceGaps: string[];
     /** Model generation timestamp */
     generatedAt: number;
   }

   /** Adversary COA (MLCOA or MDCOA) */
   export interface AdversaryCOA {
     /** COA identifier */
     id: string;
     /** MLCOA or MDCOA */
     coaType: 'MLCOA' | 'MDCOA';
     /** COA name/title */
     name: string;
     /** Narrative intelligence assessment (paragraph format per CONTEXT decision) */
     narrative: string;
     /** Key actions the adversary would take */
     keyActions: Array<{
       phase: string;
       action: string;
       objective: string;
       timing: string;
     }>;
     /** Indicators that would signal this COA is being adopted */
     indicators: string[];
     /** Decision points for friendly forces */
     decisionPoints: string[];
     /** Probability assessment (0-1) per INVARIANT 5 */
     probability: number;
     /** Confidence bounds on probability */
     probabilityBounds: { lower: number; upper: number };
     /** Risk to friendly forces if this COA occurs */
     riskAssessment: string;
   }

   /** Complete adversary analysis output */
   export interface AdversaryAnalysisOutput {

     forceModel: AdversaryForceModel;

     coas: AdversaryCOA[];

     mostLikelyCOAId: string;

     mostDangerousCOAId: string;

     analysisSummary: string;

     analysisConfidence: number;
     /** Known gaps in the analysis */
     analysisGaps: string[];
     /** Timestamp */
     analyzedAt: number;
   }
