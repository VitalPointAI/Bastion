    ```typescript
    export class DecisionService {
      /**
       * Get RACI matrix for a problem set. Auto-seeds defaults if empty.
       * Echelon is read from problem set metadata (default: 'operational').
       */
      async getRACIMatrix(problemSetId: string): Promise<RACIAssignment[]>;

      /**
       * Get all decisions a specific position needs to act on.
       * Queries RACI for position's R/A assignments, then finds pending decisions of those types.
       */
      async getPendingForPosition(problemSetId: string, position: string): Promise<Decision[]>;

      /**
       * Create a new decision request.
       * Validates decision_type exists in RACI matrix.
       */
      async createDecision(params: {
        problem_set_id: string;
        decision_type: string;
        title: string;
        description?: string;
        context_json?: Record<string, unknown>;
        requested_by?: string;
      }): Promise<Decision>;

      /**
       * Process a decision action (approve/reject/defer/request info).
       * Validates that the acting user's position has R or A role for this decision type.
       * Returns updated decision.
       */
      async actOnDecision(
        decisionId: string,
        action: DecisionStatus,
        actorDid: string,
        actorPosition: string,
      ): Promise<Decision>;

      /**
       * Update RACI assignment. Only commander/xo positions can modify.
       * RACI changes themselves should go through a decision gate (future — for now, direct update).
       */
      async updateRACIAssignment(
        problemSetId: string,
        decisionType: string,
        position: string,
        raciRole: RACIRole,
        actorPosition: string,
      ): Promise<RACIAssignment>;

      /**
       * Get decisions summary for dashboard display.
       * Returns counts by status and list of recent decisions.
       */
      async getDashboardSummary(problemSetId: string): Promise<{
        pending: number;
        approved: number;
        rejected: number;
        deferred: number;
        info_requested: number;
        recent: Decision[];
      }>;
    }

    export const decisionService = new DecisionService();
    ```

    Implementation details:
    - `getRACIMatrix`: Call `raciStore.getByProblemSet()`. If empty, call `raciStore.seedDefaults(problemSetId, echelon)` then re-query. For echelon, query `problem_sets` table for the problem set's echelon field (if it exists) or default to 'operational'.
    - `getPendingForPosition`: Call `raciStore.getByPosition()` to get R/A decision types, then `decisionStore.getPending()` and filter to matching types.
    - `actOnDecision`: Validate actor has R or A role for the decision's type. If not, throw error. Then call `decisionStore.updateStatus()`.
    - `updateRACIAssignment`: Check actorPosition is 'commander' or 'xo'. If not, throw error. Then call `raciStore.upsert()`.
    - `getDashboardSummary`: Query all decisions, count by status, return recent 20.
  </action>

    **1. Create `backend/src/api/decisions.ts`** REST API router:

    Following existing patterns (e.g., api/problem-sets.ts): `const router = Router(); export default router;`

    Endpoints:
    - `GET /api/decisions/:problemSetId` — List decisions for a problem set. Query params: `?status=pending&decision_type=mission_approval`. Calls `decisionService.getByProblemSet()` or `decisionStore.getByProblemSet()` with filters.
    - `GET /api/decisions/:problemSetId/pending/:position` — Get pending decisions for a position. Calls `decisionService.getPendingForPosition()`.
    - `GET /api/decisions/:problemSetId/summary` — Dashboard summary. Calls `decisionService.getDashboardSummary()`.
    - `POST /api/decisions/:problemSetId` — Create a decision. Body: `{ decision_type, title, description?, context_json? }`. Sets `requested_by` from auth session DID.
    - `PATCH /api/decisions/:problemSetId/:decisionId` — Act on a decision. Body: `{ action: 'approved'|'rejected'|'deferred'|'info_requested' }`. Requires actor position from session/body.
    - `GET /api/decisions/:problemSetId/raci` — Get RACI matrix (auto-seeds if empty). Calls `decisionService.getRACIMatrix()`.
    - `PUT /api/decisions/:problemSetId/raci` — Update RACI assignment. Body: `{ decision_type, position, raci_role }`. Calls `decisionService.updateRACIAssignment()`.

    All endpoints use `requireAuth` middleware. Standard error handling with try/catch.

    Add `import decisionsRouter from './api/decisions.js';` and mount: `app.use('/api/decisions', decisionsRouter);`
