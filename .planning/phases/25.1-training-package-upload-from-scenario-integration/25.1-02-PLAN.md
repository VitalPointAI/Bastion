    } catch (err: unknown) {
      if (err && typeof err === 'object' && 'status' in err && (err as { status: number }).status === 404) return null;

      return null;

  4. Move sidebar items from a const to a computed value inside the component, conditionally adding training-packages:
  ```typescript
  export function UnderstandTab({ problemSetId }: UnderstandTabProps) {
    const { mode } = useMode();
    const [selectedView, setSelectedView] = useState<UnderstandView>('strategic-docs');
    const [trainingDocCount, setTrainingDocCount] = useState(0);
    const [hasPendingExtraction, setHasPendingExtraction] = useState(false);

    const items: SidebarItem[] = [
      { id: 'strategic-docs', label: 'Strategic Documents' },
      { id: 'subscriptions', label: 'Data Sharing', tooltip: 'Manage cross-problem-set data subscriptions' },
      ...(mode === 'training' ? [{
        id: 'training-packages',
        label: `Training Packages${trainingDocCount > 0 ? ` (${trainingDocCount})` : ''}${hasPendingExtraction ? ' *' : ''}`,
      }] : []),
    ];

    return (
      <TabLayout items={items} selectedItem={selectedView} onSelectItem={(id) => setSelectedView(id as UnderstandView)}>
        {selectedView === 'strategic-docs' && <StrategicDashboard problemSetId={problemSetId} />}
        {selectedView === 'subscriptions' && <SubscriptionManager problemSetId={problemSetId} />}
        {selectedView === 'training-packages' && mode === 'training' && (
          <TrainingPackagesView
            problemSetId={problemSetId}
            onDocCountChange={setTrainingDocCount}
            onPendingChange={setHasPendingExtraction}
          />
        )}
      </TabLayout>
    );
  }
  ```
  Note: if selectedView is 'training-packages' but mode switches to operational, it won't render. Could also reset selectedView when mode changes (optional improvement).
