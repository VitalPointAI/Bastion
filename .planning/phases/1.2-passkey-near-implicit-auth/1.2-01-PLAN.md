```typescript
// PasskeyCredential - stored after successful registration
export interface PasskeyCredential {
  id: string;                      // UUID
  userId: string;                  // UUID of user
  credentialId: Buffer;            // WebAuthn credential ID (stored as bytes)
  publicKey: Buffer;               // Credential public key (stored as bytes)
  counter: bigint;                 // Signature counter for replay prevention
  transports: string;              // JSON array of transports (usb, ble, nfc, internal)
  prfSupported: boolean;           // Whether PRF extension is available
  nearImplicitAccountId: string;   // Derived NEAR account ID (64-char hex)
  createdAt: Date;
  lastUsedAt: Date | null;
}

// MagicLinkToken - for PRF fallback and recovery
export interface MagicLinkToken {
  id: string;                      // UUID
  token: string;                   // 64-char hex token (32 random bytes)
  email: string;                   // Email address
  expiresAt: Date;                 // 15-minute expiration
  used: boolean;                   // One-time use flag
  createdAt: Date;
}

// UserSession - replaces Privy JWT
export interface UserSession {
  id: string;                      // UUID
  userId: string;                  // UUID of user
```

```typescript
import { getPool } from '../lib/database.js';
import type { PasskeyCredential, CreatePasskeyInput, AuthUser } from './types.js';

let initialized = false;

async function ensureInitialized(): Promise<void> {
  if (initialized) return;

  const pool = getPool();

  // Create auth_users table
  await pool.query(`
    CREATE TABLE IF NOT EXISTS auth_users (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      email VARCHAR(255) UNIQUE NOT NULL,
      email_verified BOOLEAN DEFAULT false,
      near_account_id VARCHAR(64),
      privy_user_id VARCHAR(255),
      passkey_migrated BOOLEAN DEFAULT false,
      created_at TIMESTAMPTZ DEFAULT NOW(),
      updated_at TIMESTAMPTZ DEFAULT NOW()
    )
  `);

  // Create passkey_credentials table
  await pool.query(`
    CREATE TABLE IF NOT EXISTS passkey_credentials (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      user_id UUID NOT NULL REFERENCES auth_users(id) ON DELETE CASCADE,
      credential_id BYTEA NOT NULL,
      public_key BYTEA NOT NULL,
      counter BIGINT NOT NULL DEFAULT 0,
      transports TEXT NOT NULL DEFAULT '[]',
      prf_supported BOOLEAN DEFAULT false,
      near_implicit_account_id VARCHAR(64) NOT NULL,
      created_at TIMESTAMPTZ DEFAULT NOW(),
      last_used_at TIMESTAMPTZ
    )
  `);

  // Index for credential lookup during authentication
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_passkey_credential_id
    ON passkey_credentials (credential_id)
  `);

  // Index for user's passkeys
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_passkey_user_id
    ON passkey_credentials (user_id)
  `);

  initialized = true;
}
```
