/**
 * UserStore - User management with UUID as stable identifier
 *
 * ARCHITECTURE: The user's UUID is the anchor for MPC derivation path.
 * MPC path = `bastion,{user.id}` - never changes, survives email/passkey changes.
 */

import { getPool } from '../lib/database.js';
import { v4 as uuidv4 } from 'uuid';
import type { AuthUser, CreateUserInput } from './types.js';

let initialized = false;

async function ensureInitialized(): Promise<void> {
  if (initialized) return;

  const pool = getPool();

  // Create auth_users table - UUID is the PRIMARY stable identifier
  await pool.query(`
    CREATE TABLE IF NOT EXISTS auth_users (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      email VARCHAR(255) UNIQUE NOT NULL,
      alternate_email VARCHAR(255),
      email_verified BOOLEAN DEFAULT false,
      near_account_id VARCHAR(64),
      mpc_derivation_path VARCHAR(128),
      privy_user_id VARCHAR(255),
      passkey_migrated BOOLEAN DEFAULT false,
      totp_enabled BOOLEAN DEFAULT false,
      created_at TIMESTAMPTZ DEFAULT NOW(),
      updated_at TIMESTAMPTZ DEFAULT NOW()
    )
  `);

  // Index for email lookup
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_auth_users_email
    ON auth_users (email)
  `);

  // Index for alternate email (recovery)
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_auth_users_alternate_email
    ON auth_users (alternate_email) WHERE alternate_email IS NOT NULL
  `);

  // Index for Privy migration
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_auth_users_privy_id
    ON auth_users (privy_user_id) WHERE privy_user_id IS NOT NULL
  `);

  initialized = true;
}

export class UserStore {
  /**
   * Create new user with stable UUID
   * UUID becomes the MPC derivation path anchor: `bastion,{uuid}`
   */
  async createUser(input: CreateUserInput): Promise<AuthUser> {
    await ensureInitialized();
    const pool = getPool();

    const userId = uuidv4();
    const mpcDerivationPath = `bastion,${userId}`;

    const result = await pool.query(`
      INSERT INTO auth_users (id, email, alternate_email, mpc_derivation_path)
      VALUES ($1, $2, $3, $4)
      RETURNING *
    `, [userId, input.email, input.alternateEmail || null, mpcDerivationPath]);

    return this.mapRow(result.rows[0]);
  }

  /**
   * Find user by primary email
   */
  async findByEmail(email: string): Promise<AuthUser | null> {
    await ensureInitialized();
    const pool = getPool();

    const pool = getPool();

/**
 * PasskeyStore - WebAuthn credential management
 *
 * ARCHITECTURE: Passkeys are for AUTHENTICATION ONLY.
 * This store does NOT derive NEAR accounts from passkey keys.
 * NEAR accounts are created via MPC using the user's UUID.
 */

import { getPool } from '../lib/database.js';
import { v4 as uuidv4 } from 'uuid';
import type { PasskeyCredential, CreatePasskeyInput } from './types.js';

let initialized = false;

async function ensureInitialized(): Promise<void> {
  if (initialized) return;

  const pool = getPool();

  // Create passkey_credentials table
  // NOTE: No near_implicit_account_id - passkeys don't derive NEAR accounts
  await pool.query(`
    CREATE TABLE IF NOT EXISTS passkey_credentials (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      user_id UUID NOT NULL REFERENCES auth_users(id) ON DELETE CASCADE,
      credential_id BYTEA NOT NULL,
      public_key BYTEA NOT NULL,
      counter BIGINT NOT NULL DEFAULT 0,
      transports TEXT NOT NULL DEFAULT '[]',
      prf_supported BOOLEAN DEFAULT false,
      created_at TIMESTAMPTZ DEFAULT NOW(),
      last_used_at TIMESTAMPTZ
    )
  `);

  // Index for credential lookup during authentication
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_passkey_credential_id
    ON passkey_credentials (credential_id)
  `);

  // Index for user's passkeys
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_passkey_user_id
    ON passkey_credentials (user_id)
  `);

  initialized = true;
}

export class PasskeyStore {
  /**
   * Store new passkey credential after registration
   */
  async createCredential(input: CreatePasskeyInput): Promise<PasskeyCredential> {
    await ensureInitialized();
    const pool = getPool();

    const id = uuidv4();
    const transportsJson = JSON.stringify(input.transports);

    const result = await pool.query(`
      INSERT INTO passkey_credentials
        (id, user_id, credential_id, public_key, counter, transports, prf_supported)
      VALUES ($1, $2, $3, $4, $5, $6, $7)
      RETURNING *
    `, [
      id,
      input.userId,
      input.credentialId,
      input.publicKey,
      input.counter.toString(),
      transportsJson,
      input.prfSupported
    ]);

    return this.mapRow(result.rows[0]);
  }

  /**
   * Find credential by WebAuthn credential ID
   * Uses hex comparison as noted in research Pitfall #3
   */
  async findByCredentialId(credentialId: Buffer): Promise<PasskeyCredential | null> {
    await ensureInitialized();
    const pool = getPool();

    // Convert to hex for comparison (PostgreSQL BYTEA comparison)
    const credentialIdHex = credentialId.toString('hex');

    const result = await pool.query(`
      SELECT * FROM passkey_credentials
      WHERE encode(credential_id, 'hex') = $1
    `, [credentialIdHex]);

    return result.rows[0] ? this.mapRow(result.rows[0]) : null;
  }

  /**
   * Get all passkeys for a user
   */
  async findByUserId(userId: string): Promise<PasskeyCredential[]> {
    await ensureInitialized();
    const pool = getPool();

    const result = await pool.query(`
      SELECT * FROM passkey_credentials WHERE user_id = $1
    `, [userId]);

    return result.rows.map(row => this.mapRow(row));
  }

  /**
   * Update signature counter (replay prevention)
   */
  async updateCounter(id: string, counter: bigint): Promise<void> {
    await ensureInitialized();
    const pool = getPool();

    await pool.query(`
      UPDATE passkey_credentials
      SET counter = $2, last_used_at = NOW()
      WHERE id = $1
    `, [id, counter.toString()]);
  }

  /**
   * Delete all passkeys for user (recovery flow)
   */
  async deleteAllForUser(userId: string): Promise<void> {
    await ensureInitialized();
    const pool = getPool();

    await pool.query(`
      DELETE FROM passkey_credentials WHERE user_id = $1
    `, [userId]);
  }

  private mapRow(row: any): PasskeyCredential {
    return {
      id: row.id,
      userId: row.user_id,
      credentialId: row.credential_id,
      publicKey: row.public_key,
      counter: BigInt(row.counter),
      transports: row.transports,
      prfSupported: row.prf_supported,
      createdAt: new Date(row.created_at),
      lastUsedAt: row.last_used_at ? new Date(row.last_used_at) : null
    };
  }
}

// Singleton

export function getPasskeyStore(): PasskeyStore {
  if (!passkeyStore) {
    passkeyStore = new PasskeyStore();
  }
  return passkeyStore;
}

/**
 * TotpStore - TOTP secret management for 2FA and recovery
 *
 * TOTP is required for account recovery (email + TOTP).
 * Secrets are encrypted at rest using platform encryption key.
 */

import { getPool } from '../lib/database.js';
import { v4 as uuidv4 } from 'uuid';
import { authenticator } from 'otplib';
import { createCipheriv, createDecipheriv, randomBytes, scryptSync } from 'crypto';
import type { TotpCredential, CreateTotpInput } from './types.js';

// Encryption key derived from environment variable
// In production, this should be from HSM or secure key management
const ENCRYPTION_KEY = process.env.TOTP_ENCRYPTION_KEY || 'development-key-change-in-production';

let initialized = false;

async function ensureInitialized(): Promise<void> {
  if (initialized) return;

  const pool = getPool();

  await pool.query(`
    CREATE TABLE IF NOT EXISTS totp_credentials (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      user_id UUID UNIQUE NOT NULL REFERENCES auth_users(id) ON DELETE CASCADE,
      secret_encrypted TEXT NOT NULL,
      verified BOOLEAN DEFAULT false,
      created_at TIMESTAMPTZ DEFAULT NOW(),
      last_used_at TIMESTAMPTZ
    )
  `);

  // Unique constraint ensures one TOTP per user
  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_totp_user_id
    ON totp_credentials (user_id)
  `);

  initialized = true;
}

/**
 * Encrypt TOTP secret for storage
 */
function encryptSecret(secret: string): string {
  const key = scryptSync(ENCRYPTION_KEY, 'totp-salt', 32);
  const iv = randomBytes(16);
  const cipher = createCipheriv('aes-256-gcm', key, iv);

  let encrypted = cipher.update(secret, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  const authTag = cipher.getAuthTag().toString('hex');

  // Format: iv:authTag:encrypted
  return `${iv.toString('hex')}:${authTag}:${encrypted}`;
}

/**
 * Decrypt TOTP secret for verification
 */
function decryptSecret(encryptedData: string): string {
  const [ivHex, authTagHex, encrypted] = encryptedData.split(':');

  const key = scryptSync(ENCRYPTION_KEY, 'totp-salt', 32);
  const iv = Buffer.from(ivHex, 'hex');
  const authTag = Buffer.from(authTagHex, 'hex');

  const decipher = createDecipheriv('aes-256-gcm', key, iv);
  decipher.setAuthTag(authTag);

  let decrypted = decipher.update(encrypted, 'hex', 'utf8');
  decrypted += decipher.final('utf8');

  return decrypted;
}

export class TotpStore {
  /**
   * Generate and store new TOTP secret
   * Returns secret (base32) for QR code generation
   */
  async createTotp(userId: string): Promise<{ secret: string; id: string }> {
    await ensureInitialized();
    const pool = getPool();

    // Generate new secret
    const secret = authenticator.generateSecret();
    const secretEncrypted = encryptSecret(secret);
    const id = uuidv4();

    // Delete existing TOTP if any (re-enrollment)
    await pool.query(`DELETE FROM totp_credentials WHERE user_id = $1`, [userId]);

    await pool.query(`
      INSERT INTO totp_credentials (id, user_id, secret_encrypted)
      VALUES ($1, $2, $3)
    `, [id, userId, secretEncrypted]);

    return { secret, id };
  }

  /**
   * Verify TOTP code
   */
  async verifyCode(userId: string, code: string): Promise<boolean> {
    await ensureInitialized();
    const pool = getPool();

    const result = await pool.query(`
      SELECT * FROM totp_credentials WHERE user_id = $1
    `, [userId]);

    if (!result.rows[0]) {
      return false;
    }

    const secret = decryptSecret(result.rows[0].secret_encrypted);
    const isValid = authenticator.verify({ token: code, secret });

    if (isValid) {
      // Update last used timestamp
      await pool.query(`
        UPDATE totp_credentials
        SET last_used_at = NOW(), verified = true
        WHERE user_id = $1
      `, [userId]);
    }

    return isValid;
  }

  /**
   * Mark TOTP as verified (initial setup confirmation)
   */
  async markVerified(userId: string): Promise<void> {
    await ensureInitialized();
    const pool = getPool();

    await pool.query(`
      UPDATE totp_credentials
      SET verified = true
      WHERE user_id = $1
    `, [userId]);
  }

  /**
   * Check if user has verified TOTP
   */
  async hasVerifiedTotp(userId: string): Promise<boolean> {
    await ensureInitialized();
    const pool = getPool();

    const result = await pool.query(`
      SELECT verified FROM totp_credentials
      WHERE user_id = $1 AND verified = true
    `, [userId]);

    return result.rows.length > 0;
  }

  /**
   * Get TOTP credential for user
   */
  async findByUserId(userId: string): Promise<TotpCredential | null> {
    await ensureInitialized();
    const pool = getPool();

    const result = await pool.query(`
      SELECT * FROM totp_credentials WHERE user_id = $1
    `, [userId]);

    return result.rows[0] ? this.mapRow(result.rows[0]) : null;
  }

  /**
   * Delete TOTP for user
   */
  async deleteForUser(userId: string): Promise<void> {
    await ensureInitialized();
    const pool = getPool();

    await pool.query(`DELETE FROM totp_credentials WHERE user_id = $1`, [userId]);
  }

  private mapRow(row: any): TotpCredential {
    return {
      id: row.id,
      userId: row.user_id,
      secretEncrypted: row.secret_encrypted,
      verified: row.verified,
      createdAt: new Date(row.created_at),
      lastUsedAt: row.last_used_at ? new Date(row.last_used_at) : null
    };

/**

import { getPool } from '../lib/database.js';
import { v4 as uuidv4 } from 'uuid';
import { randomBytes } from 'crypto';
import type { MagicLinkToken, CreateMagicLinkInput } from './types.js';

const MAGIC_LINK_EXPIRY_MINUTES = 15;

let initialized = false;

async function ensureInitialized(): Promise<void> {
  if (initialized) return;

  const pool = getPool();

  await pool.query(`
    CREATE TABLE IF NOT EXISTS magic_link_tokens (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      token VARCHAR(64) UNIQUE NOT NULL,
      email VARCHAR(255) NOT NULL,
      purpose VARCHAR(20) NOT NULL DEFAULT 'auth',
      expires_at TIMESTAMPTZ NOT NULL,
      used BOOLEAN DEFAULT false,
      created_at TIMESTAMPTZ DEFAULT NOW()
    )
  `);

  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_magic_link_token ON magic_link_tokens (token)
  `);

  await pool.query(`
    CREATE INDEX IF NOT EXISTS idx_magic_link_email ON magic_link_tokens (email)
  `);

  initialized = true;
}

export class MagicLinkStore {
  /**
   * Create magic link token
   * Returns 64-char hex token
   */
  async createToken(input: CreateMagicLinkInput): Promise<string> {
    await ensureInitialized();
    const pool = getPool();

    // Invalidate existing unused tokens for this email
    await pool.query(`
      UPDATE magic_link_tokens
      SET used = true
      WHERE email = $1 AND used = false
    `, [input.email]);

    // Generate secure random token
    const token = randomBytes(32).toString('hex');
    const expiresAt = new Date(Date.now() + MAGIC_LINK_EXPIRY_MINUTES * 60 * 1000);

    await pool.query(`
      INSERT INTO magic_link_tokens (id, token, email, purpose, expires_at)
      VALUES ($1, $2, $3, $4, $5)
    `, [uuidv4(), token, input.email, input.purpose, expiresAt]);

    return token;
  }

  /**
   * Verify token and return email if valid
   * Marks token as used atomically
   */
  async verifyToken(token: string): Promise<{ email: string; purpose: string } | null> {
    await ensureInitialized();
    const pool = getPool();

    // Atomic: select + update in transaction
    const result = await pool.query(`
      UPDATE magic_link_tokens
      SET used = true
      WHERE token = $1 AND used = false AND expires_at > NOW()
      RETURNING email, purpose
    `, [token]);

    if (!result.rows[0]) {
      return null;
    }

    return {
      email: result.rows[0].email,
      purpose: result.rows[0].purpose
    };
  }

  /**
   * Get token record without marking as used (for checking)
   */
  async getToken(token: string): Promise<MagicLinkToken | null> {
    await ensureInitialized();
    const pool = getPool();

    const result = await pool.query(`
      SELECT * FROM magic_link_tokens WHERE token = $1
    `, [token]);

    return result.rows[0] ? this.mapRow(result.rows[0]) : null;
  }

  private mapRow(row: any): MagicLinkToken {
    return {
      id: row.id,
      token: row.token,
      email: row.email,
      purpose: row.purpose,
      expiresAt: new Date(row.expires_at),
      used: row.used,
      createdAt: new Date(row.created_at)
    };
  }
}

let magicLinkStore: MagicLinkStore | null = null;

export function getMagicLinkStore(): MagicLinkStore {
  if (!magicLinkStore) {
    magicLinkStore = new MagicLinkStore();
  }
  return magicLinkStore;
}

/**
 * PlatformSettingsStore - Deployment environment configuration
 *
 * Allows admin to configure auth policy based on deployment context:
 * - public: TOTP via phone apps
 * - enterprise: TOTP or hardware tokens
 * - classified: Hardware tokens only (no phones in SCIFs)
 */

import { getPool } from '../lib/database.js';
import { v4 as uuidv4 } from 'uuid';
import type { DeploymentConfig } from './types.js';

const DEFAULT_CONFIG: Omit<DeploymentConfig, 'id' | 'updatedAt' | 'updatedBy'> = {
  environment: 'public',
  allowedSecondFactors: ['totp'],
  requireSecondFactor: true,
  sessionDurationMinutes: 10080, // 7 days
  requireReauthForHighValue: false
};

let initialized = false;

async function ensureInitialized(): Promise<void> {
  if (initialized) return;

  const pool = getPool();

  await pool.query(`
    CREATE TABLE IF NOT EXISTS platform_settings (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      key VARCHAR(64) UNIQUE NOT NULL,
      value JSONB NOT NULL,
      updated_at TIMESTAMPTZ DEFAULT NOW(),
      updated_by UUID
    )
  `);

  // Insert default deployment config if not exists
  await pool.query(`
    INSERT INTO platform_settings (id, key, value)
    VALUES ($1, 'deployment_config', $2)
    ON CONFLICT (key) DO NOTHING
  `, [uuidv4(), JSON.stringify(DEFAULT_CONFIG)]);

  initialized = true;
}

export class PlatformSettingsStore {
  /**
   * Get deployment configuration
   */
  async getDeploymentConfig(): Promise<DeploymentConfig> {
    await ensureInitialized();
    const pool = getPool();

    const result = await pool.query(`
      SELECT * FROM platform_settings WHERE key = 'deployment_config'
    `);

    if (!result.rows[0]) {
      return {
        id: '',
        ...DEFAULT_CONFIG,
        updatedAt: new Date(),
        updatedBy: null
      };
    }

    const row = result.rows[0];
    const value = row.value;

    return {
      id: row.id,
      environment: value.environment || 'public',
      allowedSecondFactors: value.allowedSecondFactors || ['totp'],
      requireSecondFactor: value.requireSecondFactor ?? true,
      sessionDurationMinutes: value.sessionDurationMinutes || 10080,
      requireReauthForHighValue: value.requireReauthForHighValue || false,
      updatedAt: new Date(row.updated_at),
      updatedBy: row.updated_by
    };
  }

  /**
   * Update deployment configuration (admin only)
   */
  async updateDeploymentConfig(
    config: Partial<Omit<DeploymentConfig, 'id' | 'updatedAt' | 'updatedBy'>>,
    adminUserId: string
  ): Promise<DeploymentConfig> {
    await ensureInitialized();
    const pool = getPool();

    // Get current config
    const current = await this.getDeploymentConfig();

    // Merge with updates
    const updated = {
      environment: config.environment || current.environment,
      allowedSecondFactors: config.allowedSecondFactors || current.allowedSecondFactors,
      requireSecondFactor: config.requireSecondFactor ?? current.requireSecondFactor,
      sessionDurationMinutes: config.sessionDurationMinutes || current.sessionDurationMinutes,
      requireReauthForHighValue: config.requireReauthForHighValue ?? current.requireReauthForHighValue
    };

    const result = await pool.query(`
      UPDATE platform_settings
      SET value = $1, updated_at = NOW(), updated_by = $2
      WHERE key = 'deployment_config'
      RETURNING *
    `, [JSON.stringify(updated), adminUserId]);

    const row = result.rows[0];

    return {
      id: row.id,
      ...updated,
      updatedAt: new Date(row.updated_at),
      updatedBy: row.updated_by
    };
  }

  /**
   * Check if a second factor type is allowed
   */
  async isSecondFactorAllowed(factorType: 'totp' | 'hardware_token' | 'cac_piv'): Promise<boolean> {
    const config = await this.getDeploymentConfig();
    return config.allowedSecondFactors.includes(factorType);
  }

  /**
   * Get session duration in minutes
   */
  async getSessionDuration(): Promise<number> {
    const config = await this.getDeploymentConfig();
    return config.sessionDurationMinutes;
  }
}

let platformSettingsStore: PlatformSettingsStore | null = null;

export function getPlatformSettingsStore(): PlatformSettingsStore {
  if (!platformSettingsStore) {
    platformSettingsStore = new PlatformSettingsStore();
  }
  return platformSettingsStore;
}
