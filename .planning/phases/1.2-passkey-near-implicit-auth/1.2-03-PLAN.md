Implement magic link authentication as a fallback for browsers/devices that don't support the WebAuthn PRF extension. This ensures all users can authenticate even when PRF-based DID operations aren't available.

Purpose: PRF extension is required for DID operations but not universally supported. Magic links provide a fallback authentication path for account recovery and PRF-unsupported scenarios.

Output: MagicLinkService with token generation/verification, email sending placeholder, and REST API endpoints integrated with the existing auth router.

```typescript
import { MagicLinkStore } from './magic-link-store.js';
import { UserStore } from './passkey-store.js';
import { SessionStore } from './session-store.js';
import type { UserSession } from './types.js';

// Environment configuration
const APP_URL = process.env.APP_URL || 'http://localhost:5173';
const SENDGRID_API_KEY = process.env.SENDGRID_API_KEY;
const FROM_EMAIL = process.env.FROM_EMAIL || 'noreply@bastion.near';

export interface MagicLinkResult {
  success: boolean;
  message: string;
  // Only in dev mode for testing
  devToken?: string;
}

export interface MagicLinkVerifyResult {
  verified: boolean;
  session: UserSession;
  accountId: string | null;
  needsPasskeySetup: boolean;
}

export class MagicLinkService {
  private magicLinkStore: MagicLinkStore;
  private userStore: UserStore;
  private sessionStore: SessionStore;

  constructor() {
    this.magicLinkStore = new MagicLinkStore();
    this.userStore = new UserStore();
    this.sessionStore = new SessionStore();
  }

  /**
   * Send magic link to user's email
   *
   * Flow:
   * 1. Find or create user by email
   * 2. Generate secure token
   * 3. Send email with link
   *
   * Note: In dev mode without SENDGRID_API_KEY, returns token for testing
   */
  async sendMagicLink(email: string): Promise<MagicLinkResult> {
    // Normalize email
    const normalizedEmail = email.toLowerCase().trim();

    // Validate email format
    if (!this.isValidEmail(normalizedEmail)) {
      return { success: false, message: 'Invalid email format' };
    }

    // Find or create user
    let user = await this.userStore.findByEmail(normalizedEmail);
    if (!user) {
      user = await this.userStore.createUser(normalizedEmail);
    }

    // Generate token (invalidates previous tokens for this email)
    const token = await this.magicLinkStore.createToken(normalizedEmail);

    // Build magic link URL
    const magicLink = `${APP_URL}/auth/verify?token=${token}`;

    // Send email (or return token in dev mode)
    if (SENDGRID_API_KEY) {
      await this.sendEmail(normalizedEmail, magicLink);
      return {
        success: true,
        message: 'Magic link sent to your email'
      };
    } else {
      // Dev mode: return token for testing
      console.log('DEV MODE: Magic link URL:', magicLink);
      return {
        success: true,
        message: 'Magic link generated (check console in dev mode)',
        devToken: token
      };
    }
  }

  /**
   * Verify magic link token and create session
   *
   * Flow:
   * 1. Verify token (checks expiry, one-time use)
   * 2. Get user by email
   * 3. Mark email as verified
   * 4. Create session (prfAvailable: false)
   */
  async verifyMagicLink(token: string): Promise<MagicLinkVerifyResult> {
    // Verify token
    const email = await this.magicLinkStore.verifyToken(token);

    if (!email) {
      throw new Error('Invalid or expired magic link');
    }

    // Get user
    const user = await this.userStore.findByEmail(email);
    if (!user) {
      throw new Error('User not found');
    }

    // Mark email as verified
    if (!user.emailVerified) {
      await this.userStore.markEmailVerified(user.id);
    }

    // Create session (PRF not available via magic link)
    const session = await this.sessionStore.createSession({
      userId: user.id,
      accountId: user.nearAccountId || `pending:${user.id}`,
      prfAvailable: false
    });

    // Check if user needs to set up passkey
    const needsPasskeySetup = !user.nearAccountId;

    return {
      verified: true,
      session,
      accountId: user.nearAccountId,
      needsPasskeySetup
    };
  }

  /**
   * Request account recovery via magic link
   *
   * This is the same as sendMagicLink but with explicit recovery context.
   * After verification, user will be prompted to register a new passkey.
   */
  async requestRecovery(email: string): Promise<MagicLinkResult> {
    const result = await this.sendMagicLink(email);

    if (result.success) {
      result.message = 'Recovery link sent to your email. You will be able to set up a new passkey after verification.';
    }

    return result;
  }

  // Private: Email validation
  private isValidEmail(email: string): boolean {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return emailRegex.test(email);
  }

  // Private: Send email via SendGrid (or other provider)
  private async sendEmail(to: string, magicLink: string): Promise<void> {
    if (!SENDGRID_API_KEY) {
      throw new Error('Email provider not configured');
    }

    // SendGrid API call
    const response = await fetch('https://api.sendgrid.com/v3/mail/send', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${SENDGRID_API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        personalizations: [{ to: [{ email: to }] }],
        from: { email: FROM_EMAIL, name: 'BASTION' },
        subject: 'Your BASTION login link',
        content: [
          {
            type: 'text/plain',
            value: `Click this link to log in to BASTION:\n\n${magicLink}\n\nThis link expires in 15 minutes and can only be used once.`
          },
          {
            type: 'text/html',
            value: `
              <div style="font-family: sans-serif; max-width: 600px; margin: 0 auto;">
                <h2 style="color: #0a0a0a;">BASTION Login</h2>
                <p>Click the button below to log in:</p>
                <a href="${magicLink}"
                   style="display: inline-block; padding: 12px 24px; background: #3b82f6; color: white; text-decoration: none; border-radius: 6px; margin: 16px 0;">
                  Log in to BASTION
                </a>
                <p style="color: #666; font-size: 14px;">
                  This link expires in 15 minutes and can only be used once.
                </p>
                <p style="color: #999; font-size: 12px;">
                  If you didn't request this link, you can safely ignore this email.
                </p>
              </div>
            `
          }
        ]
      })
    });

    if (!response.ok) {
      const error = await response.text();
      console.error('SendGrid error:', error);
      throw new Error('Failed to send email');
    }
  }
}

// Singleton
let instance: MagicLinkService | null = null;
export function getMagicLinkService(): MagicLinkService {
  if (!instance) {
    instance = new MagicLinkService();
  }
  return instance;
}
```

Key features:

- Production uses SendGrid (configurable to other providers)
- 15-minute expiry, one-time use
- Sessions created with prfAvailable: false
- Detects if user needs passkey setup after magic link auth

```typescript
// ============================================
// Magic Link Endpoints (PRF Fallback)
// ============================================

/**
 * POST /api/auth/magic-link/send
 * Send magic link to email
 *
 * Body: { email: string }
 * Response: { success: boolean, message: string, devToken?: string }
 */
router.post('/magic-link/send', async (req, res) => {
  try {
    const { email } = req.body;

    if (!email || typeof email !== 'string') {
      return res.status(400).json({ error: 'Email is required' });
    }

    const service = getMagicLinkService();
    const result = await service.sendMagicLink(email);

    if (!result.success) {
      return res.status(400).json({ error: result.message });
    }

    res.json(result);
  } catch (error) {
    console.error('Magic link send error:', error);
    res.status(500).json({
      error: 'Failed to send magic link',
      details: error instanceof Error ? error.message : 'Unknown error'
    });
  }
});

/**
 * POST /api/auth/magic-link/verify
 * Verify magic link token and create session
 *
 * Body: { token: string }
 * Response: { verified: boolean, sessionToken: string, accountId: string | null, needsPasskeySetup: boolean }
 */
router.post('/magic-link/verify', async (req, res) => {
  try {
    const { token } = req.body;

    if (!token || typeof token !== 'string') {
      return res.status(400).json({ error: 'Token is required' });
    }

    const service = getMagicLinkService();
    const result = await service.verifyMagicLink(token);

    res.json({
      verified: result.verified,
      sessionToken: result.session.id,
      accountId: result.accountId,
      needsPasskeySetup: result.needsPasskeySetup
    });
  } catch (error) {
    console.error('Magic link verify error:', error);
    res.status(401).json({
      error: 'Verification failed',
      details: error instanceof Error ? error.message : 'Unknown error'
    });
  }
});

/**
 * POST /api/auth/recovery/request
 * Request account recovery via magic link
 *
 * Body: { email: string }
 * Response: { success: boolean, message: string }
 */
router.post('/recovery/request', async (req, res) => {
  try {
    const { email } = req.body;

    if (!email || typeof email !== 'string') {
      return res.status(400).json({ error: 'Email is required' });
    }

    const service = getMagicLinkService();
    const result = await service.requestRecovery(email);

    if (!result.success) {
      return res.status(400).json({ error: result.message });
    }

    // Never reveal whether email exists (security)
    res.json({
      success: true,
      message: 'If an account exists with this email, a recovery link has been sent.'
    });
  } catch (error) {
    console.error('Recovery request error:', error);
    // Always return success to prevent email enumeration
    res.json({
      success: true,
      message: 'If an account exists with this email, a recovery link has been sent.'
    });
  }
});
```

These endpoints complement the passkey flow:
- `/magic-link/send` - For PRF-unsupported browsers
- `/magic-link/verify` - Authenticates and returns session
- `/recovery/request` - For lost passkey recovery
