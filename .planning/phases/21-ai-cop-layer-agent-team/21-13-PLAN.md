  ```typescript
  async getStatus(req: Request, res: Response): Promise<void> {
    try {
      const workspaceId = req.query.workspaceId as string;
      if (!workspaceId) {
        res.status(400).json({ error: 'workspaceId required' });
        return;
      }

      const layers = await layerStore.queryLayers({ workspaceId });
      const hasLayers = layers.length > 0;
      const layerCount = layers.length;
      const draftCount = layers.filter(l => l.state === 'draft').length;
      const copCount = layers.filter(l => l.state === 'cop').length;

      // Check if generation is in progress by looking for recent agent activity
      const activity = activityBridge.getRecentActivity(10);
      const recentGeneration = activity.find(a =>
        a.action === 'generating' &&
        Date.now() - new Date(a.timestamp).getTime() < 60000
      );
      const status = recentGeneration ? 'generating' : (hasLayers ? 'ready' : 'idle');

      res.json({
        status, // 'idle' | 'generating' | 'ready'
        layerCount,
        draftCount,
        copCount,
        hasLayers,
      });
    } catch (error) {
      const message = error instanceof Error ? error.message : 'Unknown error';
      res.status(500).json({ error: message });
    }
  }
  ```

  ```typescript
  const [autoTriggered, setAutoTriggered] = useState(false);

  // Auto-trigger on first visit when objectives exist but no layers
  useEffect(() => {
    if (autoTriggered) return;

    async function checkAndTrigger() {
      try {
        const status = await copService.getStatus(workspaceId);
        if (status.status === 'idle' && !status.hasLayers) {
          // Check if workspace has objectives (via a lightweight check)
          const layers = await copService.queryLayers(workspaceId);
          if (layers.length === 0) {
            setGenerating(true);
            setAutoTriggered(true);
            await copService.triggerGeneration(workspaceId, 'default');
            setGenerating(false);
          }
        }
      } catch (err) {
        console.warn('[COP] Auto-trigger check failed:', err);
        setGenerating(false);
      }
    }

    checkAndTrigger();
  }, [workspaceId, autoTriggered]);
  ```
