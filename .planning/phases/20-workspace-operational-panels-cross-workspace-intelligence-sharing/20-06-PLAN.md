   async getPanelConfig(workspaceId: string, userDID: string): Promise<{ panelVisibility: Record<string, string[]>; defaultTab: string }> {
     const res = await this.fetch(`/api/workspaces/${workspaceId}/panel-config`, {
       headers: { 'X-DID': userDID },
       credentials: 'include',
     });
     if (!res.ok) throw new Error('Failed to fetch panel config');
     return res.json();
   }

   async updatePanelConfig(workspaceId: string, panelVisibility: Record<string, string[]>, userDID: string, defaultTab?: string): Promise<void> {
     const res = await this.fetch(`/api/workspaces/${workspaceId}/panel-config`, {
       method: 'PUT',
       headers: { 'Content-Type': 'application/json', 'X-DID': userDID },
       credentials: 'include',
       body: JSON.stringify({ panelVisibility, defaultTab }),
     });
