   ```sql
   -- Migration 052: ironclaw_concepts versioned concept store (Phase 66)
   -- TARGET DATABASE: ironclaw-postgres (NOT coalition_ops)
   -- This migration must be run against DATABASE_URL_IRONCLAW
   -- The ironclaw-postgres image (pgvector/pgvector:pg16) already has vector extension

   CREATE EXTENSION IF NOT EXISTS vector;

   CREATE TYPE ironclaw_concept_type AS ENUM (
     'actor', 'situation', 'assessment', 'preference',
     'lesson', 'intent', 'relationship', 'directive'
   );

   CREATE TABLE IF NOT EXISTS ironclaw_concepts (
     id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
     problem_set_id  TEXT,
     user_did        TEXT NOT NULL,
     concept_key     TEXT NOT NULL,
     concept_type    ironclaw_concept_type NOT NULL,
     current_value   JSONB NOT NULL,
     confidence      NUMERIC(4,3) NOT NULL DEFAULT 0.500,
     source_thread_id TEXT,
     version         INT NOT NULL DEFAULT 1,
     supersedes_id   UUID REFERENCES ironclaw_concepts(id),
     status          TEXT NOT NULL DEFAULT 'active',
     embedding       vector(1536),
     created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
     updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
     expires_at      TIMESTAMPTZ,
     UNIQUE (problem_set_id, user_did, concept_key, version)
   );

   CREATE INDEX IF NOT EXISTS idx_ironclaw_concepts_embedding
     ON ironclaw_concepts USING hnsw (embedding vector_cosine_ops);

   CREATE INDEX IF NOT EXISTS idx_ironclaw_concepts_lookup
     ON ironclaw_concepts (user_did, problem_set_id, concept_key, status);
   ```

   - Import `pg` and reuse the `getIronclawPool()` pattern from `routine-service.ts` (copy the function locally or import — follow the pattern of creating a local pool singleton)
   - Import `pgvector from 'pgvector/pg'` and call `pgvector.registerTypes(pool)` in `ensureReady()`
   - Class `ConceptStore` with methods:
     - `ensureReady()` — register pgvector types with pool (once)
     - `upsertConcept(input: ConceptUpsertInput): Promise<ConceptEntry>` — find existing active concept with same (user_did, concept_key, problem_set_id), mark it superseded, insert new version with incremented version number and supersedes_id FK
     - `getActive(userDid: string, problemSetId: string | null): Promise<ConceptEntry[]>` — all active concepts for user+problem_set
     - `getByKey(userDid: string, conceptKey: string, problemSetId: string | null): Promise<ConceptEntry | null>` — latest active version
     - `getVersionChain(userDid: string, conceptKey: string, problemSetId: string | null): Promise<ConceptEntry[]>` — all versions ordered by version ASC (use recursive CTE with supersedes_id)
     - `semanticSearch(embedding: number[], userDid: string, problemSetId: string | null, limit: number): Promise<(ConceptEntry & { similarity: number })[]>` — pgvector cosine search using `1 - (embedding <=> $1::vector) AS similarity`, filtering by user_did, status='active', and (problem_set_id = $3 OR problem_set_id IS NULL)
     - `retractByThread(threadId: string): Promise<number>` — mark all concepts with source_thread_id = threadId as 'retracted'; if a retracted concept is the latest version and has a superseded predecessor, re-activate the predecessor
     - `retractById(id: string): Promise<void>` — mark single concept as 'retracted'
     - `getConsolidationCandidates(problemSetId: string | null): Promise<{conceptKey: string, userDid: string, versions: number}[]>` — find concept_keys with 2+ active versions from different threads
   - Export singleton: `export const conceptStore = new ConceptStore();`
   - All queries use parameterized $N placeholders (no string interpolation)
   - All queries include `user_did = $N` in WHERE for access control
   - Use `rowToConceptEntry()` mapper function following autonomous-activity-store.ts pattern
