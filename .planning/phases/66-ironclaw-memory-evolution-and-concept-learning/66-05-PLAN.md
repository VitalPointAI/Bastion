   **Add directive:**
   - POST to concept upsert endpoint (or a dedicated directives endpoint — use the existing concepts API):
     ```typescript
     const handleAdd = async () => {
       if (!newDirective.trim()) return;
       const key = `directive:${Date.now()}`;
       // Optimistic add
       const tempId = crypto.randomUUID();
       setDirectives(prev => [{ id: tempId, text: newDirective.trim(), createdAt: new Date().toISOString() }, ...prev]);
       setNewDirective('');
       try {
         await fetch(`/api/ironclaw/${problemSetId}/concepts`, {
           method: 'POST',
           headers: { 'Content-Type': 'application/json' },
           body: JSON.stringify({
             conceptKey: key,
             conceptType: 'directive',
             value: { text: newDirective.trim() },
             confidence: 1.0,
           }),
         });
         // Refresh list
       } catch {
         setDirectives(prev => prev.filter(d => d.id !== tempId));
       }
     };
     ```
