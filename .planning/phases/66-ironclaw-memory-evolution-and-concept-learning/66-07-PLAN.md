   ```typescript
   let sidecarMemorySupported: boolean | null = null;

   async function checkSidecarMemorySupport(): Promise<boolean> {
     if (sidecarMemorySupported !== null) return sidecarMemorySupported;
     try {
       // Send a test /memory command to see if sidecar recognizes it
       const result = await ironclawClient.sendMessage('system', '/memory list');
       // If we get a meaningful response (not an error), memory commands work
       sidecarMemorySupported = result.response !== null && !result.response.includes('unknown command');
       console.log(`[ironclaw] sidecar memory support: ${sidecarMemorySupported ? 'YES' : 'NO'}`);
     } catch {
       sidecarMemorySupported = false;
       console.warn('[ironclaw] sidecar memory support check failed — assuming not supported');
     }
     return sidecarMemorySupported;
   }
   ```

3. **`pushConceptToSidecar(concept: ConceptEntry): Promise<void>`:**
   ```typescript
   async function pushConceptToSidecar(concept: ConceptEntry): Promise<void> {
     try {
       const supported = await checkSidecarMemorySupport();
       if (!supported) {
         // Fallback: send as a message that Ironclaw can process naturally
         console.log(`[ironclaw] sidecar sync skipped (memory commands not supported): ${concept.conceptKey}`);
         return;
       }
       const value = typeof concept.currentValue === 'object' && concept.currentValue !== null
         ? (concept.currentValue as Record<string, unknown>).text ?? JSON.stringify(concept.currentValue)
         : String(concept.currentValue);
       await ironclawClient.sendMessage('system', `/memory update ${concept.conceptKey} ${value}`);
     } catch (err) {
       console.error(`[ironclaw] sidecar push failed for ${concept.conceptKey}:`, err);
       // Non-blocking — sidecar sync is best-effort
     }
   }
   ```

4. **`forgetThread(threadId: string): Promise<void>`:**
   ```typescript
   async function forgetThread(threadId: string): Promise<void> {
     try {
       const supported = await checkSidecarMemorySupport();
       if (!supported) return;
       await ironclawClient.sendMessage('system', `/memory forget ${threadId}`);
     } catch (err) {
       console.error(`[ironclaw] sidecar forget failed for thread ${threadId}:`, err);
     }
   }
   ```

5. **`pushBulkConcepts(concepts: ConceptEntry[]): Promise<number>`:**
   - Iterate concepts, call pushConceptToSidecar for each
   - Add 100ms delay between calls to avoid flooding sidecar
   - Return count of successfully pushed
   - Used by consolidation job after merging

6. **Export:**

   ```typescript
     pushConceptToSidecar,

     pushBulkConcepts,
   ```
