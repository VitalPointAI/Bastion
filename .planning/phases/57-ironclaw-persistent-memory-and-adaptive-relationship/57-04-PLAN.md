       ```typescript
       import { getPool } from '../lib/database.js';
       // import pg-boss accessor following project pattern

       export async function registerMemoryCleanupJob(): Promise<void> {
         // Use the project's pg-boss pattern to register a recurring job
         // Job name: 'ironclaw-memory-cleanup'
         // Schedule: daily (cron: '0 3 * * *' -- 3am UTC)
         // Handler: delete expired rows from all three tables

         const handler = async () => {
           const pool = getPool();
           const results = await Promise.allSettled([
             pool.query(`DELETE FROM ironclaw_user_memory WHERE expires_at < NOW()`),
             pool.query(`DELETE FROM ironclaw_context_memory WHERE expires_at < NOW()`),
             // interaction_outcomes: delete records older than 90 days (raw data retention)
             pool.query(`DELETE FROM ironclaw_interaction_outcomes WHERE created_at < NOW() - INTERVAL '90 days'`),
           ]);
           const deleted = results.map((r, i) => {
             const tables = ['user_memory', 'context_memory', 'interaction_outcomes'];
             if (r.status === 'fulfilled') return `${tables[i]}: ${r.value.rowCount} rows`;
             return `${tables[i]}: error`;
           });
           console.log(`[ironclaw-memory-cleanup] Completed: ${deleted.join(', ')}`);
         };

         // Register with pg-boss following the project pattern found in step 1
         // If pg-boss is not easily accessible, fall back to a simple setInterval approach
         // that runs the cleanup every 24 hours (setTimeout pattern used elsewhere)
       }
       ```
