       ```typescript
       const API_BASE = '/api/resources';

       export const resourceService = {
         // Resources
         createResource: (data) => fetch(API_BASE, { method: 'POST', ... }),
         getResources: (missionId, category?) => fetch(`${API_BASE}?missionId=${missionId}...`),
         updateResource: (id, data) => fetch(`${API_BASE}/${id}`, { method: 'PATCH', ... }),
         updateStatus: (id, status) => fetch(`${API_BASE}/${id}/status`, { method: 'PATCH', ... }),
         deleteResource: (id) => fetch(`${API_BASE}/${id}`, { method: 'DELETE' }),
         bulkImport: (missionId, rows) => fetch(`${API_BASE}/bulk-import`, { method: 'POST', ... }),

         // Personnel
         createPersonnel: (data) => fetch(`${API_BASE}/personnel`, { method: 'POST', ... }),
         getPersonnel: (missionId, unitId?) => fetch(`${API_BASE}/personnel?...`),
         updatePersonnel: (id, data) => fetch(`${API_BASE}/personnel/${id}`, { method: 'PATCH', ... }),
         assignToUnit: (id, unitId) => fetch(`${API_BASE}/personnel/${id}/unit`, { method: 'PATCH', ... }),
         deletePersonnel: (id) => fetch(`${API_BASE}/personnel/${id}`, { method: 'DELETE' }),

         // Consumables
         createConsumable: (data) => fetch(`${API_BASE}/consumables`, { method: 'POST', ... }),
         getConsumables: (missionId, category?) => fetch(`${API_BASE}/consumables?...`),
         updateLevel: (id, level) => fetch(`${API_BASE}/consumables/${id}/level`, { method: 'PATCH', ... }),
         getLowStock: (missionId) => fetch(`${API_BASE}/consumables/low-stock?missionId=${missionId}`),
       };
       ```
