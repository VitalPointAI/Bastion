```typescript
import { useEffect, useState } from 'react';
import { MapContainer, TileLayer, Marker, Popup, Circle, useMap } from 'react-leaflet';
import L from 'leaflet';
import 'leaflet/dist/leaflet.css';
import './ValidityMap.css';

// Fix Leaflet default icon issue
delete (L.Icon.Default.prototype as unknown as { _getIconUrl?: unknown })._getIconUrl;
L.Icon.Default.mergeOptions({
  iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
  iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
  shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
});

export interface MapEvent {
  id: string;
  title: string;
  location: {
    name: string;
    latitude: number;
    longitude: number;
  };
  sourceType: string;
  publishedAt: Date;
  relevance?: 'supporting' | 'contradicting' | 'neutral';
}

export interface MapActor {
  id: string;
  name: string;
  type: string;
  location?: {
    latitude: number;
    longitude: number;
  };
}

export interface MapTension {
  id: string;
  description: string;
  intensity: string;
  location?: {
    latitude: number;
    longitude: number;
    radius: number;
  };
}

interface ValidityMapProps {
  events: MapEvent[];
  actors: MapActor[];
  tensions: MapTension[];
  onEventClick?: (event: MapEvent) => void;
  onActorClick?: (actor: MapActor) => void;
  center?: [number, number];
  zoom?: number;
}

// Layer visibility controls
function MapLayers({
  showEvents,
  showActors,
  showTensions,
  onToggle,
}: {
  showEvents: boolean;
  showActors: boolean;
  showTensions: boolean;
  onToggle: (layer: 'events' | 'actors' | 'tensions') => void;
}) {
  return (
    <div className="map-layers-control">
      <div className="layer-toggle">
        <input
          type="checkbox"
          id="layer-events"
          checked={showEvents}
          onChange={() => onToggle('events')}
        />
        <label htmlFor="layer-events">Events</label>
      </div>
      <div className="layer-toggle">
        <input
          type="checkbox"
          id="layer-actors"
          checked={showActors}
          onChange={() => onToggle('actors')}
        />
        <label htmlFor="layer-actors">Actors</label>
      </div>
      <div className="layer-toggle">
        <input
          type="checkbox"
          id="layer-tensions"
          checked={showTensions}
          onChange={() => onToggle('tensions')}
        />
        <label htmlFor="layer-tensions">Tensions</label>
      </div>
    </div>
  );
}

// Custom event marker icons
function createEventIcon(relevance?: string): L.DivIcon {
  const color = relevance === 'supporting' ? '#50c878' :
                relevance === 'contradicting' ? '#ff6b6b' : '#4a9eff';

  return L.divIcon({
    className: 'custom-event-marker',
    html: `<div class="event-marker" style="background: ${color}">
      <div class="event-pulse"></div>
    </div>`,
    iconSize: [24, 24],
    iconAnchor: [12, 12],
  });
}

// Custom actor marker icons
function createActorIcon(type: string): L.DivIcon {
  const colors: Record<string, string> = {
    nation: '#4a9eff',
    organization: '#50c878',
    individual: '#ffa500',
    non_state_actor: '#ff6b6b',
  };

  return L.divIcon({
    className: 'custom-actor-marker',
    html: `<div class="actor-marker" style="background: ${colors[type] || '#888'}">
      <span class="actor-icon">${type[0].toUpperCase()}</span>
    </div>`,
    iconSize: [32, 32],
    iconAnchor: [16, 16],
  });
}

export function ValidityMap({
  events,
  actors,
  tensions,
  onEventClick,
  onActorClick,
  center = [30, 0],
  zoom = 2,
}: ValidityMapProps) {
  const [showEvents, setShowEvents] = useState(true);
  const [showActors, setShowActors] = useState(true);
  const [showTensions, setShowTensions] = useState(true);

  const handleLayerToggle = (layer: 'events' | 'actors' | 'tensions') => {
    switch (layer) {
      case 'events': setShowEvents(!showEvents); break;
      case 'actors': setShowActors(!showActors); break;
      case 'tensions': setShowTensions(!showTensions); break;
    }
  };

  return (
    <div className="validity-map-container">
      <MapLayers
        showEvents={showEvents}
        showActors={showActors}
        showTensions={showTensions}
        onToggle={handleLayerToggle}
      />

      <MapContainer
        center={center}
        zoom={zoom}
        className="validity-map"
        scrollWheelZoom={true}
      >
        {/* Dark theme tile layer */}
        <TileLayer
          attribution='&copy; <a href="https://stadiamaps.com/">Stadia Maps</a>'
          url="https://tiles.stadiamaps.com/tiles/alidade_smooth_dark/{z}/{x}/{y}{r}.png"
        />

        {/* Tension circles */}
        {showTensions && tensions.filter(t => t.location).map(tension => (
          <Circle
            key={tension.id}
            center={[tension.location!.latitude, tension.location!.longitude]}
            radius={tension.location!.radius || 100000}
            pathOptions={{
              color: tension.intensity === 'critical' ? '#ff0000' :
                     tension.intensity === 'high' ? '#ff6b6b' :
                     tension.intensity === 'medium' ? '#ffa500' : '#ffcc00',
              fillOpacity: 0.2,
              weight: 2,
            }}
          >
            <Popup>
              <div className="map-popup">
                <span className={`intensity-badge ${tension.intensity}`}>
                  {tension.intensity}
                </span>
                <p>{tension.description}</p>
              </div>
            </Popup>
          </Circle>
        ))}

        {/* Event markers */}
        {showEvents && events.filter(e => e.location).map(event => (
          <Marker
            key={event.id}
            position={[event.location.latitude, event.location.longitude]}
            icon={createEventIcon(event.relevance)}
            eventHandlers={{ click: () => onEventClick?.(event) }}
          >
            <Popup>
              <div className="map-popup event-popup">
                <h4>{event.title}</h4>
                <p className="event-location">{event.location.name}</p>
                <span className="event-date">
                  {new Date(event.publishedAt).toLocaleDateString()}
                </span>
                {event.relevance && (
                  <span className={`relevance-badge ${event.relevance}`}>
                    {event.relevance}
                  </span>
                )}
              </div>
            </Popup>
          </Marker>
        ))}

        {/* Actor markers */}
        {showActors && actors.filter(a => a.location).map(actor => (
          <Marker
            key={actor.id}
            position={[actor.location!.latitude, actor.location!.longitude]}
            icon={createActorIcon(actor.type)}
            eventHandlers={{ click: () => onActorClick?.(actor) }}
          >
            <Popup>
              <div className="map-popup actor-popup">
                <span className={`actor-type-badge ${actor.type}`}>{actor.type}</span>
                <h4>{actor.name}</h4>
              </div>
            </Popup>
          </Marker>
        ))}
      </MapContainer>
    </div>
  );
}
```

```typescript
import { useCallback, useEffect, useState } from 'react';
import { ValidityMap, type MapEvent, type MapActor, type MapTension } from './ValidityMap';
import { GraphExplorer, type GraphData } from '../graph';
import './StrategicValidityDashboard.css';

interface Workspace {
  id: string;
  name: string;
  type: string;
  classification: string;
}

interface ValidityAlert {
  id: string;
  objectiveId: string;
  alertType: string;
  severity: string;
  title: string;
  details: string;
  createdAt: Date;
}

interface ObjectiveWithValidity {
  id: string;
  description: string;
  status: string;
  priority: string;
  validityScore?: number;
  validityTrend?: 'improving' | 'declining' | 'stable';
}

type ViewMode = 'map' | 'graph' | 'split';

export function StrategicValidityDashboard() {
  const [workspaces, setWorkspaces] = useState<Workspace[]>([]);
  const [selectedWorkspace, setSelectedWorkspace] = useState<string>('all');
  const [events, setEvents] = useState<MapEvent[]>([]);
  const [actors, setActors] = useState<MapActor[]>([]);
  const [tensions, setTensions] = useState<MapTension[]>([]);
  const [alerts, setAlerts] = useState<ValidityAlert[]>([]);
  const [objectives, setObjectives] = useState<ObjectiveWithValidity[]>([]);
  const [graphData, setGraphData] = useState<GraphData>({ nodes: [], edges: [] });
  const [viewMode, setViewMode] = useState<ViewMode>('map');
  const [selectedEvent, setSelectedEvent] = useState<MapEvent | null>(null);
  const [loading, setLoading] = useState(true);

  // Load workspaces
  useEffect(() => {
    fetch('/api/graph/workspaces')
      .then(res => res.json())
      .then(data => setWorkspaces(data.workspaces || []))
      .catch(console.error);
  }, []);

  // Load data for selected workspace
  useEffect(() => {
    setLoading(true);
    const wsParam = selectedWorkspace !== 'all' ? `?workspaceId=${selectedWorkspace}` : '';

    Promise.all([
      fetch(`/api/graph/osint/events${wsParam}`).then(r => r.json()),
      fetch(`/api/graph/actors${wsParam}`).then(r => r.json()),
      fetch(`/api/graph/tensions${wsParam}`).then(r => r.json()),
      fetch('/api/graph/validity/alerts').then(r => r.json()),
      fetch('/api/strategic/objectives').then(r => r.json()),
    ])
      .then(([eventsData, actorsData, tensionsData, alertsData, objectivesData]) => {
        setEvents(eventsData.events || []);
        setActors(actorsData.actors || []);
        setTensions(tensionsData.tensions || []);
        setAlerts(alertsData.alerts || []);
        setObjectives(objectivesData.objectives || []);

        // Build graph data
        const nodes = (actorsData.actors || []).map((a: MapActor) => ({
          id: a.id,
          label: a.name,
          type: a.type,
        }));
        // Note: edges would need to come from relationships endpoint
        setGraphData({ nodes, edges: [] });
      })
      .catch(console.error)
      .finally(() => setLoading(false));
  }, [selectedWorkspace]);

  const handleAcknowledgeAlert = useCallback(async (alertId: string) => {
    try {
      await fetch(`/api/graph/validity/alerts/${alertId}/acknowledge`, {
        method: 'POST',
      });
      setAlerts(prev => prev.filter(a => a.id !== alertId));
    } catch (error) {
      console.error('Failed to acknowledge alert:', error);
    }
  }, []);

  const criticalAlerts = alerts.filter(a => a.severity === 'critical' || a.severity === 'high');

  return (
    <div className="validity-dashboard">
      {/* Header */}
      <div className="dashboard-header">
        <div className="header-left">
          <h1>Strategic Intelligence Fusion</h1>
          <div className="workspace-selector">
            <select
              value={selectedWorkspace}
              onChange={e => setSelectedWorkspace(e.target.value)}
            >
              <option value="all">All Workspaces (Master View)</option>
              {workspaces.map(ws => (
                <option key={ws.id} value={ws.id}>
                  {ws.name} ({ws.type})
                </option>
              ))}
            </select>
          </div>
        </div>

        <div className="view-mode-toggle">
          <button
            className={viewMode === 'map' ? 'active' : ''}
            onClick={() => setViewMode('map')}
          >
            Map
          </button>
          <button
            className={viewMode === 'graph' ? 'active' : ''}
            onClick={() => setViewMode('graph')}
          >
            Graph
          </button>
          <button
            className={viewMode === 'split' ? 'active' : ''}
            onClick={() => setViewMode('split')}
          >
            Split
          </button>
        </div>
      </div>

      {/* Alert Banner */}
      {criticalAlerts.length > 0 && (
        <div className="alert-banner">
          <span className="alert-icon">⚠️</span>
          <span className="alert-count">{criticalAlerts.length} critical alert(s)</span>
          <div className="alert-preview">
            {criticalAlerts.slice(0, 2).map(alert => (
              <div key={alert.id} className="alert-item">
                <span className={`severity-badge ${alert.severity}`}>{alert.severity}</span>
                <span className="alert-title">{alert.title}</span>
                <button onClick={() => handleAcknowledgeAlert(alert.id)}>Dismiss</button>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Main Content */}
      <div className={`dashboard-content view-${viewMode}`}>
        {loading ? (
          <div className="loading-state">Loading strategic environment...</div>
        ) : (
          <>
            {(viewMode === 'map' || viewMode === 'split') && (
              <div className="map-section">
                <ValidityMap
                  events={events}
                  actors={actors}
                  tensions={tensions}
                  onEventClick={setSelectedEvent}
                />
              </div>
            )}

            {(viewMode === 'graph' || viewMode === 'split') && (
              <div className="graph-section">
                <GraphExplorer
                  data={graphData}
                  workspaceId={selectedWorkspace !== 'all' ? selectedWorkspace : undefined}
                  height={viewMode === 'split' ? 400 : 600}
                />
              </div>
            )}
          </>
        )}
      </div>

      {/* Sidebar */}
      <div className="dashboard-sidebar">
        {/* Event Timeline */}
        <div className="sidebar-section">
          <h3>Recent Events</h3>
          <div className="event-timeline">
            {events.slice(0, 10).map(event => (
              <div
                key={event.id}
                className={`timeline-event ${selectedEvent?.id === event.id ? 'selected' : ''}`}
                onClick={() => setSelectedEvent(event)}
              >
                <div className="event-time">
                  {new Date(event.publishedAt).toLocaleDateString()}
                </div>
                <div className="event-title">{event.title}</div>
                <div className="event-source">{event.sourceType}</div>
              </div>
            ))}
          </div>
        </div>

        {/* Objectives with Validity */}
        <div className="sidebar-section">
          <h3>Objective Validity</h3>
          <div className="objectives-list">
            {objectives.slice(0, 5).map(obj => (
              <div key={obj.id} className="objective-card">
                <div className="objective-header">
                  <span className={`priority-badge ${obj.priority.toLowerCase()}`}>
                    {obj.priority}
                  </span>
                  {obj.validityScore !== undefined && (
                    <span className={`validity-score ${
                      obj.validityScore >= 70 ? 'good' :
                      obj.validityScore >= 40 ? 'moderate' : 'low'
                    }`}>
                      {obj.validityScore}%
                    </span>
                  )}
                </div>
                <p className="objective-description">
                  {obj.description.slice(0, 100)}...
                </p>
                {obj.validityTrend && (
                  <span className={`trend-indicator ${obj.validityTrend}`}>
                    {obj.validityTrend === 'improving' ? '↑' :
                     obj.validityTrend === 'declining' ? '↓' : '→'}
                    {obj.validityTrend}
                  </span>
                )}
              </div>
            ))}
          </div>
        </div>

        {/* Alerts */}
        <div className="sidebar-section">
          <h3>Active Alerts ({alerts.length})</h3>
          <div className="alerts-list">
            {alerts.map(alert => (
              <div key={alert.id} className={`alert-card ${alert.severity}`}>
                <div className="alert-header">
                  <span className={`severity-badge ${alert.severity}`}>
                    {alert.severity}
                  </span>
                  <span className="alert-type">{alert.alertType.replace('_', ' ')}</span>
                </div>
                <h4>{alert.title}</h4>
                <p>{alert.details}</p>
                <button
                  className="acknowledge-btn"
                  onClick={() => handleAcknowledgeAlert(alert.id)}
                >
                  Acknowledge
                </button>
              </div>
            ))}
          </div>
        </div>
      </div>

      {/* Event Detail Modal */}
      {selectedEvent && (
        <div className="event-detail-modal">
          <div className="modal-content">
            <button className="close-btn" onClick={() => setSelectedEvent(null)}>
              &times;
            </button>
            <h2>{selectedEvent.title}</h2>
            <div className="event-meta">
              <span>{selectedEvent.location?.name}</span>
              <span>{new Date(selectedEvent.publishedAt).toLocaleString()}</span>
              <span className="source-type">{selectedEvent.sourceType}</span>
            </div>
            {selectedEvent.relevance && (
              <span className={`relevance-badge ${selectedEvent.relevance}`}>
                {selectedEvent.relevance} evidence
              </span>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
```
