import { z } from 'zod';
import type { MCPToolDefinition } from '../../agents/tool-registry.js';
import { actorStore } from '../raft/actor-store.js';
import { relationshipStore } from '../raft/relationship-store.js';
import { tensionStore } from '../raft/tension-store.js';
import { getNeo4jDriver } from '../neo4j-client.js';

export const createActorTool: MCPToolDefinition = {
  name: 'create_actor',
  description: 'Add an actor node to the RAFT graph',
  inputSchema: z.object({
    name: z.string().describe('Actor name'),
    type: z.enum(['nation', 'organization', 'individual', 'non_state_actor']),
    aliases: z.array(z.string()).optional(),
    attributes: z.record(z.unknown()).optional(),
    workspaceId: z.string().optional(),
    sourceDocumentIds: z.array(z.string()).default([]),
  }),
  outputSchema: z.object({ actorId: z.string() }),
  execute: async (input) => {
    const actor = await actorStore.createActor(input);
    return { actorId: actor.id };
  },
};

export const createRelationshipTool: MCPToolDefinition = {
  name: 'create_relationship',
  description: 'Add a relationship edge between two actors',
  inputSchema: z.object({

    targetActorId: z.string(),
    type: z.enum(['alliance', 'conflict', 'dependency', 'competition', 'cooperation']),
    strength: z.number().min(-1).max(1).default(0),
    description: z.string().optional(),
    evidence: z.array(z.string()).default([]),
    workspaceId: z.string().optional(),
    sourceDocumentIds: z.array(z.string()).default([]),
  }),
  outputSchema: z.object({ relationshipId: z.string() }),

export const queryGraphTool: MCPToolDefinition = {
  name: 'query_graph',
  description: 'Execute a Cypher query against the RAFT graph',
  inputSchema: z.object({
    cypher: z.string().describe('Cypher query to execute'),
    parameters: z.record(z.unknown()).optional(),
  }),
  outputSchema: z.object({
    records: z.array(z.any()),
    summary: z.any(),
  }),
  execute: async (input) => {
    const driver = getNeo4jDriver();
    const { records, summary } = await driver.executeQuery(
      input.cypher,
      input.parameters || {}
    );
    return {
      records: records.map(r => r.toObject()),
      summary: {
        counters: summary.counters.updates(),
        queryType: summary.queryType,
      },
    };
  },
};

export const runGraphAlgorithmTool: MCPToolDefinition = {
  name: 'run_graph_algorithm',
  description: 'Run a graph algorithm (PageRank, community detection, shortest path)',
  inputSchema: z.object({
    algorithm: z.enum(['pagerank', 'louvain', 'shortest_path', 'betweenness_centrality']),
    params: z.object({
      sourceActorId: z.string().optional().describe('For path algorithms'),
      targetActorId: z.string().optional().describe('For path algorithms'),
      workspaceId: z.string().optional(),
      iterations: z.number().optional().describe('For iterative algorithms'),
    }).optional(),
  }),
  outputSchema: z.object({
    results: z.array(z.any()),
  }),
  execute: async (input) => {
    const driver = getNeo4jDriver();

    let cypher: string;
    const params: Record<string, unknown> = {};

    switch (input.algorithm) {
      case 'pagerank':
        // Simple PageRank approximation using relationship count
        cypher = `
          MATCH (a:Actor)
          OPTIONAL MATCH (a)-[r]-()
          WITH a, count(r) as degree
          RETURN a.id as actorId, a.name as name, degree,
                 toFloat(degree) / (SELECT max(count) FROM (MATCH (x:Actor)-[r2]-() RETURN count(r2) as count)) as pagerank
          ORDER BY degree DESC
          LIMIT 20
        `;
        break;

      case 'louvain':
        // Community detection via connected components approximation
        cypher = `
          MATCH (a:Actor)-[:ALLIED_WITH|COOPERATES_WITH*1..3]-(b:Actor)
          WHERE a.id < b.id
          RETURN collect(distinct a.id) + collect(distinct b.id) as community, count(*) as size
          ORDER BY size DESC
          LIMIT 10
        `;
        break;

      case 'shortest_path':
        if (!input.params?.sourceActorId || !input.params?.targetActorId) {
          throw new Error('shortest_path requires sourceActorId and targetActorId');
        }
        cypher = `
          MATCH path = shortestPath((a:Actor {id: $sourceId})-[*..10]-(b:Actor {id: $targetId}))
          RETURN [n in nodes(path) | n.name] as path, length(path) as hops
        `;
        params.sourceId = input.params.sourceActorId;
        params.targetId = input.params.targetActorId;
        break;

      case 'betweenness_centrality':
        // Simplified betweenness based on path participation
        cypher = `
          MATCH (a:Actor)
          OPTIONAL MATCH path = (x:Actor)-[*2..4]-(y:Actor)
          WHERE a IN nodes(path) AND x <> a AND y <> a
          WITH a, count(path) as pathCount
          RETURN a.id as actorId, a.name as name, pathCount as betweenness
          ORDER BY pathCount DESC
          LIMIT 20
        `;
        break;

      default:
        throw new Error(`Unknown algorithm: ${input.algorithm}`);
    }

    const { records } = await driver.executeQuery(cypher, params);
    return { results: records.map(r => r.toObject()) };
  },
};

export const getActorProfileTool: MCPToolDefinition = {
  name: 'get_actor_profile',
  description: 'Get comprehensive profile of an actor including all relationships',
  inputSchema: z.object({
    actorId: z.string(),
  }),
  outputSchema: z.any(),
  execute: async (input) => {
    const actor = await actorStore.getActor(input.actorId);
    if (!actor) return { error: 'Actor not found' };

    const relationships = await relationshipStore.getActorRelationships(input.actorId, 'both');
    const tensions = await tensionStore.getTensionsForActor(input.actorId);

    // Get related actors
    const relatedActorIds = new Set<string>();
    for (const rel of relationships) {
      relatedActorIds.add(rel.sourceActorId === input.actorId ? rel.targetActorId : rel.sourceActorId);
    }
    for (const tension of tensions) {
      tension.actorIds.forEach(id => {
        if (id !== input.actorId) relatedActorIds.add(id);
      });
    }

    return {
      actor,
      relationships,
      tensions,
      relatedActorCount: relatedActorIds.size,
      stats: {
        allianceCount: relationships.filter(r => r.type === 'alliance').length,
        conflictCount: relationships.filter(r => r.type === 'conflict').length,
        tensionCount: tensions.length,
        highIntensityTensions: tensions.filter(t => t.intensity === 'high' || t.intensity === 'critical').length,
      },
    };
  },
};

export const exportGraphVisualizationTool: MCPToolDefinition = {
  name: 'export_graph_visualization',
  description: 'Export graph data for visualization',
  inputSchema: z.object({
    workspaceId: z.string().optional(),
    format: z.enum(['nodes_edges', 'd3', 'cytoscape']).default('nodes_edges'),
    includeRelationshipTypes: z.array(z.string()).optional(),
    maxNodes: z.number().default(100),
  }),
  outputSchema: z.any(),
  execute: async (input) => {
    const actors = await actorStore.listActors(input.workspaceId);
    const limitedActors = actors.slice(0, input.maxNodes);
    const actorIds = new Set(limitedActors.map(a => a.id));

    // Get relationships between these actors
    const allRelationships: Array<{ source: string; target: string; type: string; strength: number }> = [];
    for (const actor of limitedActors) {
      const rels = await relationshipStore.getActorRelationships(actor.id, 'out');
      for (const rel of rels) {
        if (actorIds.has(rel.targetActorId)) {
          if (!input.includeRelationshipTypes || input.includeRelationshipTypes.includes(rel.type)) {
            allRelationships.push({
              source: rel.sourceActorId,
              target: rel.targetActorId,
              type: rel.type,
              strength: rel.strength,
            });
          }
        }
      }
    }

    if (input.format === 'nodes_edges') {
      return {
        nodes: limitedActors.map(a => ({
          id: a.id,
          label: a.name,
          type: a.type,
        })),
        edges: allRelationships,
      };
    } else if (input.format === 'd3') {
      return {
        nodes: limitedActors.map(a => ({
          id: a.id,
          name: a.name,
          group: a.type,
        })),
        links: allRelationships.map(r => ({
          source: r.source,
          target: r.target,
          value: Math.abs(r.strength) * 10,
        })),
      };
    } else {
      // cytoscape format
      return {
        elements: {
          nodes: limitedActors.map(a => ({
            data: { id: a.id, label: a.name, type: a.type },
          })),
          edges: allRelationships.map((r, i) => ({
            data: { id: `e${i}`, source: r.source, target: r.target, type: r.type },
          })),
        },
      };
    }
  },
};
