import { z } from 'zod';
import type { MCPToolDefinition } from '../../agents/tool-registry.js';
import { objectiveStore } from '../../strategic/objectives/store.js';

export const queryObjectivesTool: MCPToolDefinition = {
  name: 'query_objectives',
  description: 'Query strategic objectives with filters and pagination',
  inputSchema: z.object({
    documentId: z.string().optional().describe('Filter by document ID'),
    status: z.enum(['DRAFT', 'PENDING_REVIEW', 'APPROVED', 'REJECTED', 'OPERATIONALIZED']).optional(),
    priority: z.enum(['LOW', 'MEDIUM', 'HIGH', 'CRITICAL']).optional(),
    instrument: z.enum(['DIPLOMATIC', 'INFORMATION', 'MILITARY', 'ECONOMIC']).optional(),
    limit: z.number().default(20),
    offset: z.number().default(0),
  }),
  outputSchema: z.object({
    objectives: z.array(z.any()),
    total: z.number(),
  }),
  execute: async (input) => {
    return await objectiveStore.listObjectives(input);
  },
};

export const saveFusedObjectiveTool: MCPToolDefinition = {
  name: 'save_fused_objective',
  description: 'Save a fused/consolidated objective to the database',
  inputSchema: z.object({
    documentId: z.string().describe('Source document ID'),
    sourceReference: z.string().describe('Reference to source location'),
    description: z.string().describe('Consolidated objective description'),
    endsWaysMeans: z.object({
      ends: z.string(),
      ways: z.string(),
      means: z.string(),
    }),
    primaryInstrument: z.enum(['DIPLOMATIC', 'INFORMATION', 'MILITARY', 'ECONOMIC']),
    supportingInstruments: z.array(z.enum(['DIPLOMATIC', 'INFORMATION', 'MILITARY', 'ECONOMIC'])).optional(),
    parentObjectiveId: z.string().optional().describe('Parent objective for hierarchy'),
    createdBy: z.string().describe('DID of the creator'),
  }),
  outputSchema: z.object({
    objectiveId: z.string(),
  }),
  execute: async (input) => {
    const id = await objectiveStore.saveObjective({
      ...input,
      extractedBy: 'AI_FUSION',
    });
    return { objectiveId: id };
  },
};

export const getDocumentMetadataTool: MCPToolDefinition = {
  name: 'get_document_metadata',
  description: 'Get metadata for a strategic document',
  inputSchema: z.object({
    documentId: z.string().describe('Document ID'),
  }),
  outputSchema: z.any(),
  execute: async (input) => {
    // Import dynamically to avoid circular dependency
    const { documentStore } = await import('../../strategic/ingestion/document-store.js');
    return await documentStore.getDocument(input.documentId);
  },
};

export const queryObjectivesByThemeTool: MCPToolDefinition = {
  name: 'query_objectives_by_theme',
  description: 'Query objectives by DIME or MIDLIFE theme',
  inputSchema: z.object({
    theme: z.enum(['DIPLOMATIC', 'INFORMATION', 'MILITARY', 'ECONOMIC', 'LEGAL', 'INTELLIGENCE', 'FINANCIAL']),
    classification: z.enum(['UNCLASSIFIED', 'SECRET', 'TOPSECRET']).optional(),
    workspaceId: z.string().optional(),
  }),
  outputSchema: z.object({
    objectives: z.array(z.any()),
  }),
  execute: async (input) => {
    const { objectives } = await objectiveStore.listObjectives({
      instrument: input.theme as any,
    });
    return { objectives };
  },
};

export const linkObjectivesTool: MCPToolDefinition = {
  name: 'link_objectives',
  description: 'Create a hierarchical link between objectives',
  inputSchema: z.object({
    parentId: z.string().describe('Parent objective ID'),
    childId: z.string().describe('Child objective ID'),
    relationshipType: z.enum(['supports', 'derives_from', 'conflicts_with', 'duplicates']),
  }),
  outputSchema: z.object({
    success: z.boolean(),
  }),
  execute: async (input) => {
    const updated = await objectiveStore.updateObjective(input.childId, {
      parentObjectiveId: input.parentId,
    });
    return { success: updated };
  },
};

Create backend/src/graph/tools/entity-tools.ts with entity resolution tools:

```typescript
import { z } from 'zod';
import type { MCPToolDefinition } from '../../agents/tool-registry.js';
import { actorStore } from '../raft/actor-store.js';
import { entityResolutionService } from '../resolution/resolution-service.js';

export const searchEntitiesTool: MCPToolDefinition = {
  name: 'search_entities',
  description: 'Search the canonical entity registry for actors',
  inputSchema: z.object({
    query: z.string().describe('Search query'),
    entityType: z.enum(['nation', 'organization', 'individual', 'non_state_actor']).optional(),
    workspaceId: z.string().optional(),
  }),
  outputSchema: z.object({
    entities: z.array(z.any()),
  }),
  execute: async (input) => {
    const entities = await actorStore.findActorsByName(input.query, true);
    const filtered = input.entityType
      ? entities.filter(e => e.type === input.entityType)
      : entities;
    return { entities: filtered };
  },
};

export const createEntityAliasTool: MCPToolDefinition = {
  name: 'create_entity_alias',
  description: 'Register an alias for an existing entity',
  inputSchema: z.object({
    canonicalId: z.string().describe('ID of the canonical entity'),
    alias: z.string().describe('New alias to add'),
    source: z.string().describe('Document or source where alias was found'),
  }),
  outputSchema: z.object({
    success: z.boolean(),
  }),
  execute: async (input) => {
    const success = await actorStore.addAlias(input.canonicalId, input.alias);
    return { success };
  },
};

export const mergeEntitiesTool: MCPToolDefinition = {
  name: 'merge_entities',
  description: 'Merge duplicate entities into a canonical entity',
  inputSchema: z.object({
    sourceIds: z.array(z.string()).describe('IDs of entities to merge'),
    targetId: z.string().describe('ID of the canonical entity to keep'),
  }),
  outputSchema: z.object({
    mergedCount: z.number(),
    canonicalId: z.string(),
  }),
  execute: async (input) => {
    let mergedCount = 0;
    for (const sourceId of input.sourceIds) {
      if (sourceId !== input.targetId) {
        await actorStore.mergeActors(sourceId, input.targetId);
        mergedCount++;
      }
    }
    return { mergedCount, canonicalId: input.targetId };
  },
};

export const getEntityReferencesTool: MCPToolDefinition = {
  name: 'get_entity_references',
  description: 'Find all document references to an entity',
  inputSchema: z.object({
    entityId: z.string().describe('Entity ID'),
  }),
  outputSchema: z.object({
    documentIds: z.array(z.string()),
    objectiveIds: z.array(z.string()),
  }),
  execute: async (input) => {
    const actor = await actorStore.getActor(input.entityId);
    if (!actor) {
      return { documentIds: [], objectiveIds: [] };
    }
    // Get relationships to find linked objectives
    const { relationshipStore } = await import('../raft/relationship-store.js');
    const relationships = await relationshipStore.getActorRelationships(input.entityId, 'both');
    const objectiveIds = relationships.flatMap(r => r.evidence);

    return {
      documentIds: actor.sourceDocumentIds,
      objectiveIds: [...new Set(objectiveIds)],
    };
  },
};

export const entityTools = [
  searchEntitiesTool,
  createEntityAliasTool,
  mergeEntitiesTool,
  getEntityReferencesTool,
];
```
