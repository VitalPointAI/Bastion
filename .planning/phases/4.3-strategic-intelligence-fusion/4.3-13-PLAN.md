   ```typescript
   private async executeToolHandler(
     tool: MCPTool,
     input: Record<string, unknown>,
     context: ToolExecutionContext
   ): Promise<string> {
     const handler = allToolHandlers[tool.toolId];

     if (!handler) {
       console.warn(`[AgentWrapper] No handler found for tool: ${tool.toolId} (${tool.name})`);
       return JSON.stringify({
         error: `Unknown tool: ${tool.toolId}`,
         availableTools: Object.keys(allToolHandlers),
       });
     }

     try {
       console.log(`[AgentWrapper] Executing tool ${tool.toolId} for agent ${context.agentId}`);
       const result = await handler(input);
       return JSON.stringify(result);
     } catch (error) {
       console.error(`[AgentWrapper] Tool ${tool.toolId} failed:`, error);
       return JSON.stringify({
         error: `Tool execution failed: ${error instanceof Error ? error.message : String(error)}`,
         tool: tool.toolId,
       });
     }
   }
   ```

Key points:
- The handler map is a flat merge of all 5 handler objects. Tool IDs are unique across all domains (e.g., `create_actor`, `query_objectives`, `search_entities`, `fetch_osint_feeds`, `update_validity_score`).
- Dispatch uses `tool.toolId` (snake_case like `create_actor`), NOT `tool.name` (human-readable like "Create Actor"). The handler map keys match toolId.
- Unknown tool IDs return an error JSON string with available tools for debugging.
- Errors during handler execution are caught and returned as JSON error strings (LangChain expects string returns from DynamicStructuredTool).
- The `context` parameter is available for future use (e.g., classification filtering at tool level) but not needed by current handlers.
