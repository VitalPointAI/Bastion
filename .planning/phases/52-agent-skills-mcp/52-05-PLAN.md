```typescript
export type TaskStatus =
  | 'created'
  | 'dispatched'
  | 'agent_working'
  | 'collecting_results'
  | 'presenting'
  | 'awaiting_approval'
  | 'applying'
  | 'completed'
  | 'refining'
  | 'rejected'
  | 'failed';

export interface IronclawTask {
  taskId: string;
  problemSetId: string;
  userDid: string;
  title: string;
  description: string | null;
  status: TaskStatus;
  assignedAgents: string[];
  assignedTeam: string | null;
  threadId: string | null;
  steps: StepInfo[];
  currentStep: number;
  results: TaskResult[];
  suggestions: TaskSuggestion[];
  targetFields: Record<string, string>;  // field path -> label
  userFeedback: FeedbackEntry[];
  createdAt: Date;
  updatedAt: Date;
  completedAt: Date | null;
}

export interface CreateTaskParams {
  problemSetId: string;
  userDid: string;
  title: string;
  description?: string;
  targetFields: Record<string, string>;  // field path -> label
  agentHints?: string[];  // suggested agent IDs
}

export interface StepInfo {
  index: number;
  label: string;
  status: 'pending' | 'running' | 'complete' | 'failed';
  agentId?: string;
  startedAt?: string;
  completedAt?: string;
  result?: unknown;
}

export interface TaskResult {
  agentId: string;
  agentName: string;
  output: string;
  fieldPath?: string;
  timestamp: string;
}

export interface TaskSuggestion {
  id: string;
  fieldPath: string;
  fieldLabel: string;
  content: string;
  agentId: string;
  status: 'pending' | 'approved' | 'dismissed' | 'revision_requested';
}

export interface FeedbackEntry {
  timestamp: string;
  feedback: string;
  suggestionId?: string;
}

// Valid state transitions
export const VALID_TRANSITIONS: Record<TaskStatus, TaskStatus[]> = {
  created: ['dispatched', 'failed'],
  dispatched: ['agent_working', 'failed'],
  agent_working: ['collecting_results', 'failed'],
  collecting_results: ['presenting', 'failed'],
  presenting: ['awaiting_approval'],
  awaiting_approval: ['applying', 'refining', 'rejected'],
  applying: ['completed', 'failed'],
  refining: ['dispatched'],
  completed: [],
  rejected: [],
  failed: [],
};

export const ROLE_FIELD_PERMISSIONS: Record<string, string[]> = {
  commander: ['*'],
  xo: ['*'],
  j2_intelligence: ['design.problemFraming', 'design.cogAnalysis.*', 'intelligence.*'],
  j3_operations: ['design.operationalApproach', 'design.linesOfEffort', 'plan.coaDetails.*'],
  j4_logistics: ['plan.sustainment.*', 'logistics.*'],
  j5_plans: ['design.*', 'plan.*', 'campaign.*'],
  j6_communications: ['plan.commandSignal.*', 'communications.*'],
};
```

**task-orchestrator.ts** — The core orchestration service:

```typescript
export class TaskOrchestrator {
  private taskStore: TaskStore;

  async createTask(params: CreateTaskParams): Promise<IronclawTask> {
    // 1. Generate taskId: `task_${Date.now()}_${random}`
    // 2. Build steps from targetFields: one step per field + "Synthesize Results" final step
    // 3. Select agents: use agentHints if provided, else query AgentRegistry for agents
    //    matching the task domain (use agent capabilities field)
    // 4. Persist task via taskStore.createTask()
    // 5. Return task
  }

  async dispatchTask(taskId: string): Promise<void> {
    // 1. Load task from store
    // 2. Transition to 'dispatched'
    // 3. Get or create BastionSupervisor for the problem set
    // 4. Build SupervisorInput with task description + target fields
    // 5. Execute supervisor.execute() with a unique threadId (store on task)
    //    CRITICAL: This is async/background. Use setImmediate or similar pattern.
    //    Do NOT block. The caller gets immediate return after dispatch.
    // 6. Transition to 'agent_working'
    // 7. As supervisor progresses, update steps via handleStepComplete()
    // 8. On completion, call collectResults()
    // On error: transition to 'failed', log error
  }

  async handleStepComplete(taskId: string, stepIdx: number, result: unknown): Promise<void> {
    // 1. Update step status to 'complete' with result
    // 2. Advance currentStep
    // 3. Publish StepProgressData via publishToChannel for WebSocket
    //    Emit type: 'ironclaw.step-progress' with the task's StepProgressData
    // 4. If all steps complete, call collectResults()
  }

  private async collectResults(taskId: string): Promise<void> {
    // 1. Transition to 'collecting_results'
    // 2. Read task results from supervisor execution trace
    // 3. For each targetField, extract the relevant agent output
    // 4. Generate TaskSuggestion per field:
    //    { id: `sug_...`, fieldPath, fieldLabel, content: agentOutput, agentId, status: 'pending' }
    // 5. Persist suggestions to task
    // 6. Call presentResults()
  }

  async presentResults(taskId: string): Promise<void> {
    // 1. Transition to 'presenting' then 'awaiting_approval'
    // 2. For each suggestion in task.suggestions:
    //    Emit an ironclaw.response message with SuggestionData via publishToChannel
    //    This makes each suggestion appear as an IronclawSuggestion card in the drawer
    // 3. Emit a summary message: "Task complete. {N} suggestions ready for review."
  }

  async handleApproval(taskId: string, suggestionId: string, decision: 'approved' | 'dismissed'): Promise<void> {
    // 1. Update suggestion status
    // 2. If approved, queue the field write (will be applied in applyApproved)
    // 3. Check if all suggestions are resolved (approved or dismissed)
    // 4. If all resolved, call applyApproved()
  }

  async handleRefinement(taskId: string, feedback: string, suggestionId?: string): Promise<void> {
    // 1. Record feedback via taskStore.addFeedback()
    // 2. Transition to 'refining'
    // 3. Re-dispatch with feedback as additional context
    //    Build new supervisor input including previous results + user feedback
    // 4. This starts the refinement loop (dispatched -> agent_working -> ... -> awaiting_approval)
  }

  async applyApproved(taskId: string): Promise<void> {
    // 1. Transition to 'applying'
    // 2. For each approved suggestion:
    //    Call the suggestion accept endpoint logic (same as Plan 04 accept handler)
    //    Use the field write dispatch map to route to correct API
    // 3. Transition to 'completed' with completedAt timestamp
    // 4. Emit completion message to drawer
  }

  async init(): Promise<void> {
    // On startup: query stale tasks (status IN dispatched, agent_working)
    // Mark as 'failed' with 'server restart' error
    // Per research pitfall 7 — prevents zombie tasks
  }
}

export function getTaskOrchestrator(): TaskOrchestrator { /* singleton */ }
```

```typescript
if (parsed?.task_request) {
  const tr = parsed.task_request as Record<string, unknown>;
  const orchestrator = getTaskOrchestrator();
  const task = await orchestrator.createTask({
    problemSetId,
    userDid: session.userDid,
    title: tr.title as string,
    description: (tr.description as string) ?? null,
    targetFields: (tr.target_fields as Record<string, string>) ?? {},
    agentHints: (tr.agent_hints as string[]) ?? [],
  });

  // Emit initial progress message to drawer
  const stepProgress: StepProgressData = {
    steps: task.steps.map(s => ({ label: s.label, status: s.status })),
    current_step: 0,
    started_at: new Date().toISOString(),
  };
  // Persist a chat message with step progress
  await ironclawStore.addMessage({
    problemSetId,
    role: 'assistant',
    content: `Task created: ${task.title}. Dispatching agents...`,
    stepProgress,
  });

  // Dispatch agents in background (non-blocking per research pitfall 6)
  orchestrator.dispatchTask(task.taskId).catch(err =>
    console.error(`[ironclaw] Task dispatch failed: ${task.taskId}`, err)
  );
}
```
