```typescript
async function expeditedAuthorize(
  gateId: string,
  decision: 'approve' | 'deny',
  missionContext: Record<string, unknown>,
): Promise<{ txHash: string; gateStatus: string }> {
  // 1. Record on NEAR blockchain using tx-signer
  // Use signAndSubmitFunctionCall with the DID contract
  // Method: 'record_authorization' (or fallback to 'store_did' if contract doesn't have it)
  // Args: { gate_id: gateId, decision, context: JSON.stringify(missionContext), timestamp: new Date().toISOString() }
  const result = await signAndSubmitFunctionCall(
    commanderKeyBytes,  // Get from gate context or env
    process.env.DID_CONTRACT_ID || 'did.bastion.testnet',
    'record_authorization',  // See open question — may need to use store_did as fallback
    { gate_id: gateId, decision, context: missionContext },
  );

  // 2. Update gate status in gate store
  // Approved → GateStatus.approved, Denied → GateStatus.rejected
  await updateGateStatus(gateId, {
    status: decision === 'approve' ? 'approved' : 'rejected',
    proposal_id: result.txHash || `local-${Date.now()}`,
    decided_at: new Date().toISOString(),
  });

  return { txHash: result.txHash || '', gateStatus: decision === 'approve' ? 'approved' : 'rejected' };
}
```

Important: If `signAndSubmitFunctionCall` fails (e.g., NEAR testnet down), still update the gate status locally and log the blockchain failure. The demo should not stall on NEAR issues — record the decision locally with a `blockchain_status: 'pending'` flag.
