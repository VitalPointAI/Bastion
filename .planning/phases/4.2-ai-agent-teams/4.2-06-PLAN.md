   ```typescript
   import { createSupervisor, createHandoffTool } from "@langchain/langgraph-supervisor";

   export class BastionSupervisor {
     private graph: CompiledStateGraph;

     constructor(
       private config: SupervisorConfig,
       private agents: LangGraphAgentWrapper[],
       private checkpointer: PostgresSaver
     ) {}

     async initialize(): Promise<void> {
       // Create handoff tools with classification checks
       const handoffTools = this.agents.map(agent =>
         createHandoffTool({
           agentName: agent.agentId,
           description: agent.getDescription(),
           // Add pre-handoff classification filter
           preHandoff: (state) => {
             const agentClearance = agent.getClearance();
             return createClassificationFilterNode(agentClearance)(state);
           }
         })
       );

       // Create supervisor with routing prompt
       this.graph = createSupervisor({
         llm: this.config.supervisorModel,
         agents: this.agents.map(a => a.createAgent()),
         tools: handoffTools,
         prompt: this.buildSupervisorPrompt(),
       }).compile({
         checkpointer: this.checkpointer
       });
     }

     async execute(input: SupervisorInput): Promise<SupervisorOutput> {
       const config = {
         configurable: {
           thread_id: input.threadId,
         }
       };

       const result = await this.graph.invoke(
         { messages: input.messages, classification: input.classification },
         config
       );

       return {
         output: result,
         trace: this.extractTrace(result),
       };
     }
   }
   ```

   ```typescript
   export enum ExecutionPattern {
     Sequential = "sequential",   // Pipeline: A → B → C
     Parallel = "parallel",       // Fan-out/fan-in: [A, B, C] → merge
     Hierarchical = "hierarchical", // Manager delegates to workers
     Consensus = "consensus",     // All agents must agree
   }

   export class TaskExecutor {
     /**
      * Sequential execution - tasks run in order
      */
     async executeSequential(tasks: Task[], agents: Agent[]): Promise<TaskResults> {
       const graph = new StateGraph(BastionStateAnnotation);

       // Chain agents in sequence
       for (let i = 0; i < tasks.length; i++) {
         const task = tasks[i];
         const agent = this.findAgentForTask(task, agents);
         graph.addNode(task.id, agent.createAgent());

         if (i > 0) {
           graph.addEdge(tasks[i-1].id, task.id);
         }
       }

       graph.setEntryPoint(tasks[0].id);
       graph.setFinishPoint(tasks[tasks.length-1].id);

       return graph.compile({ checkpointer }).invoke(initialState);
     }

     /**
      * Parallel execution - tasks run concurrently
      */
     async executeParallel(tasks: Task[], agents: Agent[]): Promise<TaskResults> {
       const graph = new StateGraph(BastionStateAnnotation);

       // Fan-out node
       graph.addNode("fanout", (state) => ({
         ...state,
         branches: tasks.map(t => t.id)
       }));

       // Add each task as parallel branch
       for (const task of tasks) {
         const agent = this.findAgentForTask(task, agents);
         graph.addNode(task.id, agent.createAgent());
         graph.addEdge("fanout", task.id);
       }

       // Fan-in node to merge results
       graph.addNode("fanin", this.mergeResults);
       for (const task of tasks) {
         graph.addEdge(task.id, "fanin");
       }

       return graph.compile({ checkpointer }).invoke(initialState);
     }

     /**
      * Hierarchical execution - manager delegates
      */
     async executeHierarchical(tasks: Task[], manager: Agent, workers: Agent[]): Promise<TaskResults> {
       // Use supervisor pattern
       const supervisor = new BastionSupervisor(
         { supervisorModel: manager.getModel() },
         workers.map(w => new LangGraphAgentWrapper(...)),
         checkpointer
       );

       await supervisor.initialize();
       return supervisor.execute({ tasks, classification: tasks[0].classification });
     }
   }
   ```

     - Pause execution and await human input

   ```typescript
   export interface CheckpointTrigger {
     type: "classification_escalation" | "high_risk_action" | "explicit_request" | "autonomy_exceeded";
     condition: (state: BastionState) => boolean;
   }

   export class HumanCheckpointManager {
     private pendingCheckpoints: Map<string, PendingCheckpoint> = new Map();

     /**
      * Create a checkpoint node for the graph
      */
     createCheckpointNode(trigger: CheckpointTrigger): GraphNode {
       return async (state: BastionState) => {
         if (!trigger.condition(state)) {
           return state; // Continue without checkpoint
         }

         // Pause execution
         const checkpointId = crypto.randomUUID();
         this.pendingCheckpoints.set(checkpointId, {
           checkpointId,
           threadId: state.threadId,
           state,
           trigger: trigger.type,
           createdAt: new Date().toISOString(),
           status: "pending",
         });

         // Notify (via message bus from 4.2-05)
         await messageBus.publish({
           destination: { type: "channel", target: "system.human-checkpoints" },
           messageType: "checkpoint.required",
           payload: { checkpointId, trigger: trigger.type, context: state },
           attributes: { classification: state.classification, ... }
         });

         // Return interrupt signal
         return {
           ...state,
           __interrupt__: {
             checkpointId,
             reason: trigger.type,
           }
         };
       };
     }

     /**
      * Resume execution after human approval
      */
     async resume(checkpointId: string, decision: HumanDecision): Promise<ResumeResult> {
       const checkpoint = this.pendingCheckpoints.get(checkpointId);
       if (!checkpoint) {
         throw new Error(`Checkpoint ${checkpointId} not found`);
       }

       if (decision.approved) {
         // Resume with potentially modified state
         const resumeState = decision.modifications
           ? { ...checkpoint.state, ...decision.modifications }
           : checkpoint.state;

         // Continue graph execution from checkpoint
         return await this.continueExecution(checkpoint.threadId, resumeState);
       } else {
         // Terminate execution
         return { terminated: true, reason: decision.rejectionReason };
       }
     }
   }
   ```
