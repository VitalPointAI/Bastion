/**
 * Agent DID Management
 *
 * Helpers for creating and resolving agent decentralized identifiers (DIDs).
 * Agent DIDs follow format: did:near:agent-{agentId}
 */

import { hkdf } from '@noble/hashes/hkdf';
import { sha256 } from '@noble/hashes/sha256';
import { bytesToHex, utf8ToBytes } from '@noble/hashes/utils';
import { getAgentRegistry } from './registry.js';
import type { AgentManifest } from './types.js';

// System secret for deriving agent keys (should match DID service pattern)
const AGENT_KEY_CONTEXT = 'bastion-agent-did-v1';

/**
 * Generate deterministic keys for an agent DID.
 * Uses HKDF with agent ID to derive consistent keys.
 */
function deriveAgentKeys(agentId: string): {
  blindedKey: string;
  publicKey: string;
} {
  const systemSecret = process.env.ENCRYPTION_KEY || 'dev-secret-key';
  const info = utf8ToBytes(`${AGENT_KEY_CONTEXT}:${agentId}`);

  // Derive 64 bytes: 32 for blinded key, 32 for public key
  const derived = hkdf(sha256, utf8ToBytes(systemSecret), undefined, info, 64);

  return {
    blindedKey: bytesToHex(derived.slice(0, 32)),
    publicKey: bytesToHex(derived.slice(32, 64)),
  };
}

/**
 * Create a DID for an agent.
 * Returns the DID string and associated keys.
 */
export async function createAgentDID(agentId: string): Promise<{
  did: string;
  blindedKey: string;
  publicKey: string;
}> {
  // Generate deterministic keys
  const keys = deriveAgentKeys(agentId);

  // DID format: did:near:agent-{agentId}
  const did = `did:near:agent-${agentId}`;

  return {
    did,
    blindedKey: keys.blindedKey,
    publicKey: keys.publicKey,
  };
}

/**
 * Resolve an agent DID to its manifest.
 * Returns null if agent not found.
 */
export async function resolveAgentDID(agentDID: string): Promise<AgentManifest | null> {
  // Parse agent ID from DID
  const match = agentDID.match(/^did:near:agent-(.+)$/);
  if (!match) {
    return null;
  }

  const agentId = match[1];
  const registry = getAgentRegistry();
  const agent = registry.getAgent(agentId);

  return agent || null;
}

/**
 * Verify an agent's DID matches expected keys.
 * Used for authentication/verification of agent identity.
 */
export function verifyAgentDID(agentId: string, providedPublicKey: string): boolean {
  const keys = deriveAgentKeys(agentId);
  return keys.publicKey === providedPublicKey;
}

     }).optional(),
     maxAutonomy: z.enum(['NotAutonomous', 'SemiAutonomous', 'Autonomous']).optional().default('NotAutonomous'),
     isEnabled: z.boolean().default(true),
   });

   // GET /api/admin/agents - List all agents with configs
   router.get('/agents', requireAdmin, async (req, res) => {
     try {
       const registry = getAgentRegistry();
       const agents = registry.listAgents();
       const configService = getConfigService();
       const modelConfigs = await configService.listAgentModelConfigs();

       // Merge model configs with agent data
       const agentsWithConfigs = agents.map(agent => ({
         ...agent,
         customModelConfig: modelConfigs.find(c => c.agentId === agent.agentId) || null,
       }));

       res.json({ agents: agentsWithConfigs });
     } catch (error) {
       res.status(500).json({ error: 'Failed to list agents' });
     }
   });

   // POST /api/admin/agents - Create new agent
   router.post('/agents', requireAdmin, async (req, res) => {
     try {
       const definition = AgentDefinitionSchema.parse(req.body);
       const agentId = definition.id || randomUUID();

       // Create DID for agent
       const didResult = await createAgentDID(agentId);

       // Build manifest
       const manifest = {
         agentId,
         name: definition.name,
         description: definition.description,
         phase: AgentPhase[definition.phase as keyof typeof AgentPhase] || AgentPhase.Support,
         capabilities: definition.capabilities.map(c => c as AgentCapability),
         maxAutonomy: AutonomyLevel[definition.maxAutonomy as keyof typeof AutonomyLevel] || AutonomyLevel.NotAutonomous,
         allowedProposalKinds: [ProposalKind.Custom],
         requiresHumanApproval: [ProposalKind.StrikeAuthorization],
         createdAt: new Date(),
         createdBy: req.userDID || 'admin',
         active: definition.isEnabled,
         agentDID: didResult.did,
         agentBlindedKey: didResult.blindedKey,
         agentPublicKey: didResult.publicKey,
         modelConfig: definition.modelConfig,
       };

       const registry = getAgentRegistry();
       await registry.registerAgent(manifest);

       res.status(201).json({
         agentId,
         agentDID: didResult.did,
         created: true,
       });
     } catch (error) {
       if (error instanceof z.ZodError) {
         res.status(400).json({ error: 'Invalid agent definition', details: error.errors });
       } else {
         res.status(500).json({ error: 'Failed to create agent' });
       }
     }
   });

   // GET /api/admin/agents/:agentId - Get single agent
   router.get('/agents/:agentId', requireAdmin, async (req, res) => {
     const { agentId } = req.params;
     const registry = getAgentRegistry();
     const agent = registry.getAgent(agentId as string);
     if (!agent) {
       return res.status(404).json({ error: 'Agent not found' });
     }
     res.json(agent);
   });

   // PUT /api/admin/agents/:agentId - Update agent
   router.put('/agents/:agentId', requireAdmin, async (req, res) => {
     const { agentId } = req.params;
     const registry = getAgentRegistry();
     const agent = registry.getAgent(agentId as string);
     if (!agent) {
       return res.status(404).json({ error: 'Agent not found' });
     }

     // Update allowed fields
     if (req.body.isEnabled !== undefined) {
       agent.active = req.body.isEnabled;
     }
     if (req.body.modelConfig) {
       agent.modelConfig = req.body.modelConfig;
     }

     res.json({ updated: true, agent });
   });

   // DELETE /api/admin/agents/:agentId - Deactivate agent
   router.delete('/agents/:agentId', requireAdmin, async (req, res) => {
     const { agentId } = req.params;
     try {
       const registry = getAgentRegistry();
       registry.deactivateAgent(agentId as string);
       res.json({ deactivated: true });
     } catch (error) {
       res.status(404).json({ error: 'Agent not found' });
     }
   });

   // GET /api/admin/agents/:agentId/did - Get agent DID info
   router.get('/agents/:agentId/did', requireAdmin, async (req, res) => {
     const { agentId } = req.params;
     const registry = getAgentRegistry();
     const agent = registry.getAgent(agentId as string);
     if (!agent) {
       return res.status(404).json({ error: 'Agent not found' });
     }
     res.json({
       agentId: agent.agentId,
       agentDID: agent.agentDID || null,
       agentPublicKey: agent.agentPublicKey || null,
       hasDID: !!agent.agentDID,
     });
   });
