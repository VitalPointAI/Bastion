1. Import AgentModelConfig and AgentModelConfigSchema at the top.

2. Add method to get per-agent model config:
   ```typescript
   async getAgentModelConfig(agentId: string): Promise<AgentModelConfig | null> {
     const key = `agents.${agentId}.model`;
     const result = await pool.query(
       'SELECT value FROM system_config WHERE key = $1',
       [key]
     );
     if (result.rows.length === 0) return null;
     try {
       return AgentModelConfigSchema.parse(JSON.parse(result.rows[0].value));
     } catch {
       return null;
     }
   }
   ```

3. Add method to set per-agent model config with audit trail:
   ```typescript
   async setAgentModelConfig(agentId: string, config: AgentModelConfig, reason: string): Promise<void> {
     const key = `agents.${agentId}.model`;
     const value = JSON.stringify(config);

     // Get previous value for audit
     const prev = await this.getAgentModelConfig(agentId);

     await pool.query(
       `INSERT INTO system_config (key, value, updated_at)
        VALUES ($1, $2, NOW())
        ON CONFLICT (key) DO UPDATE SET value = $2, updated_at = NOW()`,
       [key, value]
     );

     // Log to audit trail (using existing audit pattern if available, or insert to config_audit)
     await this.logConfigChange('agent_model_config', key, prev, config, reason);
   }
   ```

4. Add method to list all agent model configs:
   ```typescript
   async listAgentModelConfigs(): Promise<AgentModelConfig[]> {
     const result = await pool.query(
       "SELECT value FROM system_config WHERE key LIKE 'agents.%.model'"
     );
     return result.rows.map(row => {
       try {
         return AgentModelConfigSchema.parse(JSON.parse(row.value));
       } catch {
         return null;
       }
     }).filter((c): c is AgentModelConfig => c !== null);
   }
   ```

5. Add method to delete agent model config:
   ```typescript
   async deleteAgentModelConfig(agentId: string, reason: string): Promise<void> {
     const key = `agents.${agentId}.model`;
     const prev = await this.getAgentModelConfig(agentId);
     await pool.query('DELETE FROM system_config WHERE key = $1', [key]);
     if (prev) {
       await this.logConfigChange('agent_model_config', key, prev, null, reason);
     }
   }
   ```

Note: If logConfigChange doesn't exist, add a helper that inserts into config_audit table with category, key, previous_value, new_value, reason, timestamp columns. Check existing audit patterns in the service first.
