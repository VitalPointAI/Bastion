**Option B — No callback; intercept registration route:**

Or use a more robust pattern: wrap the package router with middleware that intercepts the `/register/finish` response:
```typescript
app.use('/api/auth', (req, res, next) => {
  if (req.method === 'POST' && req.path === '/register/finish') {
    const originalJson = res.json.bind(res);
    res.json = (body: any) => {
      // Trigger funding after successful registration
      if (res.statusCode < 400 && body?.accountId) {
        const fundingService = getFundingService();
        if (fundingService.isEnabled()) {
          fundingService.fundAccount(body.accountId).catch(err =>
            console.error('NEAR funding failed:', err)
          );
        }
      }
      return originalJson(body);
    };
  }
  next();
}, auth.router);
```
