  <name>Task 1: BrainController master component</name>
  <files>frontend/src/components/brain/BrainController.tsx</files>
  <action>
Create the master controller that orchestrates all brain sub-components and manages shared state.

Props:
- `problemSetId: string`

This component is the brain's "nervous system" — it manages state that multiple sub-components need and passes it down.

**State management:**

1. Initialize all hooks:
   - `const { mode } = useMode()` — import from `../../context/ModeContext.js`
   - `const { data, loading: dataLoading, refetch } = useBrainData(problemSetId)`
   - `const { events, activeProcesses, particlesRef, isConnected } = useBrainIngestion(problemSetId)`
   - `const { clusterMode, setClusterMode, clusterLabels } = useBrainClustering(fgRef, data?.nodes ?? [])`
   - `const { annotations, createAnnotation, updateAnnotation, deleteAnnotation, getNodeAnnotations } = useBrainAnnotations(problemSetId)`
   - `const { selectedTime, timeRange, futureTime, isLive, loading: timeLoading, setSelectedTime, goLive, effectiveData } = useBrainTimeline(problemSetId, data)`
   - `const { snapshots, saveSnapshot } = useBrainSnapshots(problemSetId)`
   - `const { gaps, gapCount, markGapNodes } = useBrainGaps(problemSetId)`
   - `const { alerts, unreadCount, markAsRead, markAllAsRead } = useBrainPatterns(problemSetId)`

2. **Node selection state:**
   - `selectedNodeId: string | null`
   - `selectedNodeIds: string[]` (for multi-select)
   - `rightPanelOpen: boolean` (derived: true if any selection exists)

3. **Search state:**
   - `searchMatchIds: string[] | null` (null = no search active, show all)

4. **Modal state:**
   - `snapshotModalOpen: boolean`

5. **Processed graph data:**
   - Take `effectiveData` from timeline hook
   - Apply gap marking: `markGapNodes(effectiveData.nodes)`
   - Apply search filtering: if `searchMatchIds` is set, dim non-matching nodes (set a `isSearchDimmed` flag)
   - This becomes the data passed to BrainVisualization

**Event handlers:**

- `handleNodeClick(node: BrainNode)`: If shift key held, add to selectedNodeIds (multi-select). Otherwise, set selectedNodeId = node.id, clear multi-select. Open right panel.
- `handleClosePanel()`: Clear selectedNodeId and selectedNodeIds. Close right panel.
- `handleSearchResults(matchingIds: string[])`: Set searchMatchIds. If empty array, clear search.
- `handleNodeFocus(nodeId: string)`: Call `fgRef.current?.centerAt(node.x, node.y, 500)` and `fgRef.current?.zoom(3, 500)` to zoom to the node.
- `handleAlertClick(alert: PatternAlert)`: Highlight alert's relatedNodeIds in the visualization (set them as searchMatchIds temporarily).
- `handleLassoSelect(nodeIds: string[])`: Set `selectedNodeIds = nodeIds` (multi-select). Open right panel with comparison view.

**Render:**

```tsx
<BrainLayout
  topBar={
    <BrainToolbar
      nodes={processedData.nodes}
      clusterMode={clusterMode}
      onClusterModeChange={setClusterMode}
      onSearchResults={handleSearchResults}
      onNodeFocus={handleNodeFocus}
      onSnapshotClick={() => setSnapshotModalOpen(true)}
      gapCount={gapCount}
    />
  }
  leftSidebar={
    <IngestionSidebar
      problemSetId={problemSetId}
      mode={mode}
    />
  }
  center={
    <>
      <BrainVisualization
        data={processedData}
        selectedNodeId={selectedNodeId}
        selectedNodeIds={selectedNodeIds}
        onNodeClick={handleNodeClick}
        onLassoSelect={handleLassoSelect}
        clusterMode={clusterMode}
      />
      <ParticleOverlay
        particlesRef={particlesRef}
        width={centerWidth}
        height={centerHeight}
        sidebarWidth={280}
      />
    </>
  }
  rightPanel={
    rightPanelOpen ? (
      <BrainDetailPanel
        selectedNode={selectedNode}
        selectedNodes={selectedNodes}
        graphData={processedData}
        annotations={selectedNode ? getNodeAnnotations(selectedNode.id) : []}
        onClose={handleClosePanel}
        onCreateAnnotation={createAnnotation}
        onUpdateAnnotation={updateAnnotation}
        onDeleteAnnotation={deleteAnnotation}
      />
    ) : undefined
  }
  rightPanelOpen={rightPanelOpen}
  timeline={
    <BrainTimeline
      timeRange={timeRange}
      futureTime={futureTime}
      selectedTime={selectedTime}
      isLive={isLive}
      loading={timeLoading}
      onTimeChange={setSelectedTime}
      onGoLive={goLive}
    />
  }
/>

{snapshotModalOpen && (
  <AIContextSnapshotModal
    isOpen={snapshotModalOpen}
    onClose={() => setSnapshotModalOpen(false)}
    problemSetId={problemSetId}
    nodeCount={processedData.nodes.length}
    edgeCount={processedData.edges.length}
    currentTimeScale={selectedTime ?? undefined}
  />
)}
```

Use `useRef` for fgRef (ForceGraphMethods) and pass it to both BrainVisualization and useBrainClustering. Use ResizeObserver on the center container to track centerWidth/centerHeight for ParticleOverlay.

import { useCallback } from 'react';
import { BrainController } from '../brain/BrainController.js';

import { useMode } from '../../context/ModeContext.js';
import { DecisionGateBanner, DecisionGateTimeline } from '../governance/index.js';
import type { DecisionGate } from '../../lib/gate-service';

interface UnderstandTabProps {
  problemSetId: string;
}

export function UnderstandTab({ problemSetId }: UnderstandTabProps) {
  const { mode } = useMode();
  const handleGateDetailClick = useCallback((gate: DecisionGate) => {
    console.log('[UnderstandTab] Gate detail:', gate.id, gate.gate_type, gate.status);
  }, []);

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%', overflow: 'hidden' }}>
      {/* Inherited context from parent problem sets */}
      {problemSetId && (
        <InheritedContextSection problemSetId={problemSetId} />
