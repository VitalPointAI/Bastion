  ```typescript
  // DesignStatus: per-section status tracking
  type SectionStatus = 'not-started' | 'in-progress' | 'complete';
  interface DesignStatus {
    problemFraming: SectionStatus;
    cogAnalysis: SectionStatus;
    linesOfEffort: SectionStatus;
    operationalApproach: SectionStatus;
  }

  // Problem Framing (JP 5-0 standard fields)
  interface ProblemFramingData {
    currentState: string;
    desiredEndState: string;
    problemStatement: string; // auto-generated from gap
    keyTensions: string[];
    obstacles: string[];
    opportunities: string[];
    assumptions: string[];
    constraints: string[];  // constraints/restraints
  }

  // CoG Analysis (Strange's CG-CC-CR-CV)
  interface CoGNode {
    id: string;
    type: 'cog' | 'critical-capability' | 'critical-requirement' | 'critical-vulnerability';
    label: string;
    description: string;
    children: CoGNode[];
  }
  interface CoGTree {
    root: CoGNode | null;
  }
  interface CoGAnalysis {
    friendly: CoGTree;
    adversary: CoGTree;
  }

  // Lines of Effort
  interface DecisivePoint {
    id: string;
    label: string;
    description: string;
    phase: string;      // which phase column it sits in
    position: number;   // x-position within phase
    cogLinks: LOECoGLink[];  // links to CoG vulnerability nodes
  }
  interface LOECoGLink {
    loeId: string;
    decisivePointId: string;
    cogNodeId: string;
    cogNodeType: CoGNode['type'];
  }
  interface LineOfEffort {
    id: string;
    name: string;
    description: string;
    objectiveId?: string;
    decisivePoints: DecisivePoint[];
    order: number;  // lane position
  }

  // Operational Approach
  interface OperationalApproach {
    phases: Array<{ id: string; name: string; description: string; order: number }>;
    transitions: Array<{ fromPhaseId: string; toPhaseId: string; conditions: string[] }>;
    decisionPoints: Array<{ id: string; label: string; phaseId: string; criteria: string[] }>;
    narrative: string;  // synthesized description
  }

  // Design-to-Plan handoff payload
  interface DesignHandoffPayload {
    problemStatement: string;
    cogAnalysis: CoGAnalysis;
    linesOfEffort: LineOfEffort[];
    phases: OperationalApproach['phases'];
    objectives: string[];
    assumptions: string[];
    constraints: string[];
  }

  // Top-level operational design record (one per problem set)
  interface OperationalDesign {
    id: string;
    problemSetId: string;
    problemFraming: ProblemFramingData;
    cogAnalysis: CoGAnalysis;
    linesOfEffort: LineOfEffort[];
    operationalApproach: OperationalApproach;
    status: DesignStatus;
    createdBy: string;
    createdAt: string;
    updatedAt: string;
  }
  ```

Add index on problem_set_id.
  - `getByProblemSetId(problemSetId)`: Get or auto-create design record for problem set. If none exists, create with defaults and return.

  - `getStatus(problemSetId)`: Return just the status JSONB.
  - `getHandoffPayload(problemSetId)`: Assemble and return DesignHandoffPayload from stored data.
