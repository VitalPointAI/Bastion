   import { SAFETY_MATRIX, ActivityCategory, AuthorityDesignation } from './types.js';

   export class SafetyMatrixEnforcer {
     /**
      * Validate that an authority designation is within the permitted range
      * for the given activity category.
      * INVARIANT 8: FullyDelegated scope restriction
      * INVARIANT 9: Safety matrix enforcement
      */
     validateAuthorityAssignment(
       category: ActivityCategory,
       requestedAuthority: AuthorityDesignation,
     ): { valid: boolean; reason?: string; matrixEntry?: SafetyMatrixEntry }

     /**
      * Check if an activity can be assigned to an agent at the given autonomy level.
      * Returns the safety matrix entry for the category.
      */
     getPermittedRange(
       category: ActivityCategory,
     ): { maxAuthority: AuthorityDesignation; minAuthority: AuthorityDesignation; fullyDelegatedPermitted: boolean }

     /**
      * Validate all activities in a phase against the safety matrix.
      * Returns list of violations.
      */
     validatePhaseActivities(
       phase: MDMPPhase,
       activityOverrides: Map<string, AuthorityDesignation>,
     ): Array<{ activityId: string; violation: string }>
   }

2. **Create `backend/src/mdmp/integration.ts`**:

   ```typescript
   export class MDMPIntegrationOrchestrator {
     constructor(
       private workflowService: MDMPWorkflowService,
       private safetyEnforcer: SafetyMatrixEnforcer,
       private activityRegistry: typeof import('./activity-registry.js'),
     )

     /**
      * Initialize a complete MDMP workflow for a mission.
      * Creates workflow, registers gates for Phase 1, validates initial state.
      */
     async initializeWorkflow(missionId: string, daoId: string, initiator: string): Promise<WorkflowInitResult>

     /**
      * Advance to next phase after validating all prerequisites:
      * - All gates satisfied (INVARIANT 2)
      * - All assumptions above threshold accepted (INVARIANT 3)
      * - All red team challenges addressed (INVARIANT 4)
      * - Safety matrix compliance for next phase activities (INVARIANT 9)
      */
     async advancePhase(missionId: string, authorizedBy: string, proposalId: number): Promise<PhaseAdvanceResult>

     /**
      * Handle assumption invalidation.
      * If Critical, creates mandatory replanning gate (INVARIANT 6).
      */
     async handleAssumptionInvalidation(
       missionId: string,
       assumptionId: string,
       evidence: string,
     ): Promise<InvalidationResult>

     /**
      * Validate that an agent output meets INVARIANT 5 (uncertainty transparency).
      * Wraps agent output with uncertainty metadata if missing.
      */
     validateAgentOutput(agentId: string, output: unknown): { valid: boolean; missingFields: string[] }

     /**
      * Record commander guidance (INVARIANT 7).
      * Creates CommanderGuidance proposal and tracks assumption modifications.
      */
     async recordCommanderGuidance(
       missionId: string,
       daoId: string,
       guidanceText: string,
       modifiesAssumptions: string[],
     ): Promise<GuidanceRecordResult>
   }
   ```

   Each method should:
   - Log structured audit events
   - Validate against safety matrix before proceeding
   - Return descriptive result objects with success/failure and reasoning
   - Handle edge cases (e.g., replanning from Phase 5 back to Phase 2)
