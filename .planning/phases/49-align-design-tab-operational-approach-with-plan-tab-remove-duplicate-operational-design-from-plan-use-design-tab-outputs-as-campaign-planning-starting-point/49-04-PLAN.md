   ```typescript
   export interface DesignRevision {
     id: string;
     problemSetId: string;
     artifactType: string;
     proposedBy: string;
     proposedAt: string;
     originalData: unknown;
     proposedData: unknown;
     rationale?: string;
     status: 'pending' | 'approved' | 'rejected' | 'merged';
     gateId?: string;
     reviewedBy?: string;
     reviewedAt?: string;
     mergedAt?: string;
   }

   export const designRevisionService = {
     async create(problemSetId: string, params: {
       artifactType: string;
       proposedData: unknown;
       originalData: unknown;
       rationale?: string;
     }): Promise<DesignRevision> {
       const res = await fetch(`/api/design/${problemSetId}/revisions`, {
         method: 'POST',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify(params),
       });
       const data = await res.json();
       return data.revision;
     },

     async list(problemSetId: string, status?: string): Promise<DesignRevision[]> { ... },
     async get(problemSetId: string, id: string): Promise<DesignRevision> { ... },
     async merge(problemSetId: string, id: string): Promise<DesignRevision> { ... },
   };
   ```
