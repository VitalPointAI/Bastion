   ```typescript
     const { artifact_type, proposed_data, original_data, rationale } = req.body;
     // Validate artifact_type is one of the 4 allowed types
     // Get accountId from req (check how other routes extract the authenticated user — likely req.headers or req.user)
     // Create revision via revisionStore.create()
     // Return { revision }
   });

   // GET /api/design/:problemSetId/revisions
   // Query: ?status=pending (optional filter)
   router.get('/', async (req, res) => {
     const { problemSetId } = req.params;
     const { status } = req.query;
     const revisions = await revisionStore.findByProblemSet(problemSetId, status as string);
     res.json({ revisions });
   });

   // GET /api/design/:problemSetId/revisions/:id
   router.get('/:id', async (req, res) => {
     const revision = await revisionStore.findById(req.params.id);
     if (!revision) return res.status(404).json({ error: 'Revision not found' });
     res.json({ revision });
   });

   // PATCH /api/design/:problemSetId/revisions/:id/merge
   // Called when DAO governance gate approves the revision
   router.patch('/:id/merge', async (req, res) => {
     const revision = await revisionStore.findById(req.params.id);
     if (!revision) return res.status(404).json({ error: 'Revision not found' });
     if (revision.status !== 'approved') return res.status(400).json({ error: 'Revision must be approved before merging' });
     const merged = await revisionStore.merge(req.params.id);
     res.json({ revision: merged });
   });

   // PATCH /api/design/:problemSetId/revisions/:id/status
   // Updates revision status (used by governance callback)
   router.patch('/:id/status', async (req, res) => {
     const { status, reviewed_by } = req.body;
     const updated = await revisionStore.updateStatus(req.params.id, status, reviewed_by);
     res.json({ revision: updated });
   });

   export { router };
   ```
