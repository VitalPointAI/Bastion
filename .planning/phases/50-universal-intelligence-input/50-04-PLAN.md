       ```typescript
       // Classification and routing events — forwarded to universal ingest hook
       const classifyEvents = ['classify:start', 'classify:result', 'route:selected', 'route:error', 'route:retry_success'];
       for (const eventName of classifyEvents) {
         es.addEventListener(eventName, (e: MessageEvent) => {
           const data = JSON.parse(e.data);
           // Forward to universal ingest handler if callback provided
           if (onUniversalIngestEvent) {
             onUniversalIngestEvent(eventName, data);
           }
         });
       }
       ```

    2. Add `onUniversalIngestEvent` as an optional parameter to `useBrainIngestion`:
