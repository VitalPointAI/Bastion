const problemSetSSEClients = new Map<string, Set<Response>>();

function createSSEProgressCallback(session: ProcessingSession): ProgressCallback {
  return (event: string, data: Record<string, unknown>) => {
    const enrichedData = { ...data, processId: session.processingId, documentId: session.documentId };
    const psClients = problemSetSSEClients.get(session.problemSetId);
    if (psClients) {
      for (const client of psClients) {
        client.write(`event: ${event}\n`);
        client.write(`data: ${JSON.stringify(enrichedData)}\n\n`);
      }
    }
  };
}

  <files>backend/src/ingest/url-unfurler.ts, backend/src/ingest/url-unfurler.test.ts</files>
  <behavior>
    - unfurlUrl for URL returning Content-Type 'application/rss+xml' returns type:'rss'
    - unfurlUrl for URL returning Content-Type 'application/pdf' returns type:'pdf_url'
    - unfurlUrl for HTML page with link rel=alternate rss+xml returns type:'rss' with discoveredFrom
    - unfurlUrl for HTML page without RSS returns type:'article' with title from og:title
    - unfurlUrl for URL that times out returns type:'unknown' (does not throw)
  </behavior>
  <action>
    1. Create `backend/src/ingest/url-unfurler.ts`:
       - Export `unfurlUrl(url: string): Promise<UnfurlResult>` (import UnfurlResult from types.ts)
       - Step 1: HEAD request with `AbortSignal.timeout(5000)`, `redirect: 'follow'`. Wrap in try/catch — on timeout/error, return `{ type: 'unknown', url }`.
       - Step 2: Check Content-Type header:
         - 'application/rss+xml' or 'application/atom+xml' -> return `{ type: 'rss', url: headRes.url }`
         - 'application/pdf' -> return `{ type: 'pdf_url', url: headRes.url }`
       - Step 3: If Content-Type includes 'text/html':
         a. GET the URL body with `AbortSignal.timeout(10000)`
         b. Parse with `new JSDOM(body)` (import from 'jsdom')
         c. Check for `<link type="application/rss+xml">` — if found, return `{ type: 'rss', url: feedUrl, discoveredFrom: url }`
         d. Try `rssParser.parseURL(url)` with short timeout — some feeds serve text/html
         e. Extract og:title from `meta[property="og:title"]` or `<title>`, og:description
         f. Return `{ type: 'article', url, title, description, htmlBody: body }`
       - Step 4: Fallback -> `{ type: 'unknown', url }`
       - Wrap entire function in try/catch — never throw, always return a result

       - Test: HTML article with OG tags -> article with title/description
       - Test: Timeout -> unknown (not an error)
       - Test: Network error -> unknown (not an error)
