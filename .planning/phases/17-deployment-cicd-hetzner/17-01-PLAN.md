**File 1: `nginx/nginx.prod.conf`**

Create the host-level nginx configuration for TLS termination. This is installed on the Hetzner server at `/etc/nginx/sites-available/bastion` (NOT inside a Docker container). The frontend container's `nginx.conf` handles SPA routing and /api/ proxy inside the Docker network; this host nginx handles internet-facing TLS.

```
# HTTP → HTTPS redirect
server {
    listen 80;
    server_name ${DOMAIN};

    # Let's Encrypt ACME challenge
    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }

    location / {
        return 301 https://$server_name$request_uri;
    }
}

# HTTPS server
server {
    listen 443 ssl http2;
    server_name ${DOMAIN};

    ssl_certificate /etc/letsencrypt/live/${DOMAIN}/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/${DOMAIN}/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # Proxy all traffic to frontend container (port 8080)
    # Frontend container's nginx handles SPA routing and /api/ proxy to backend
    location / {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_cache_bypass $http_upgrade;

        # SSE support (for channel feed events)
        proxy_buffering off;
        proxy_read_timeout 86400;
    }
}
```

Use `${DOMAIN}` as a placeholder. The server-setup.sh script or manual setup will replace it with the actual domain. Add a header comment: "# Host-level nginx for Bastion — TLS termination. **File 2: `scripts/server-setup.sh`**

Create the one-time server bootstrap script. This runs ONCE on a fresh Ubuntu 24.04 Hetzner server (CAX21 or CAX31).

Script structure:
```bash
#!/usr/bin/env bash
set -euo pipefail

# ============================================
# Bastion Server Setup - One-Time Bootstrap
# Run as root on a fresh Ubuntu 24.04 server
# Usage: bash server-setup.sh <domain> <ssh-public-key>
# ============================================

DOMAIN="${1:?Usage: server-setup.sh <domain> <ssh-public-key>}"
SSH_PUB_KEY="${2:?Usage: server-setup.sh <domain> <ssh-public-key>}"

echo "=== Setting up Bastion server for $DOMAIN ==="

# 1. Create deploy user with Docker group
useradd -m -s /bin/bash -G sudo deploy
mkdir -p /home/deploy/.ssh
echo "$SSH_PUB_KEY" > /home/deploy/.ssh/authorized_keys
chmod 700 /home/deploy/.ssh
chmod 600 /home/deploy/.ssh/authorized_keys
chown -R deploy:deploy /home/deploy/.ssh
echo "deploy ALL=(ALL) NOPASSWD:ALL" > /etc/sudoers.d/deploy

# 2. Install Docker
apt-get update
apt-get install -y apt-transport-https ca-certificates curl gnupg
install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | gpg --dearmor -o /etc/apt/keyrings/docker.gpg
chmod a+r /etc/apt/keyrings/docker.gpg
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | tee /etc/apt/sources.list.d/docker.list > /dev/null
apt-get update
apt-get install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin
usermod -aG docker deploy

# 3. Install nginx and certbot
apt-get install -y nginx certbot python3-certbot-nginx

# 4. Configure UFW firewall
ufw default deny incoming
ufw default allow outgoing
ufw allow 22/tcp
ufw allow 80/tcp
ufw allow 443/tcp
ufw --force enable

# 5. Create application directory
mkdir -p /home/deploy/bastion
chown deploy:deploy /home/deploy/bastion

# 6. Clone repository (for docker-compose.prod.yml and database init files)
su - deploy -c "git clone https://github.com/VitalPointAI/Bastion.git /home/deploy/bastion"

# 7. Install host nginx config
cp /home/deploy/bastion/nginx/nginx.prod.conf /etc/nginx/sites-available/bastion
sed -i "s/\${DOMAIN}/$DOMAIN/g" /etc/nginx/sites-available/bastion
ln -sf /etc/nginx/sites-available/bastion /etc/nginx/sites-enabled/bastion
rm -f /etc/nginx/sites-enabled/default

# 8. Obtain TLS certificate
certbot --nginx -d "$DOMAIN" --non-interactive --agree-tos --email admin@$DOMAIN --redirect

# 9. Disable root SSH login
sed -i 's/#\?PermitRootLogin.*/PermitRootLogin no/' /etc/ssh/sshd_config
systemctl reload sshd

# 10. Create .env.prod template
cat > /home/deploy/bastion/.env.prod << 'ENVEOF'
# Production environment - fill in values
POSTGRES_PASSWORD=CHANGE_ME
NEO4J_PASSWORD=CHANGE_ME
# Add other secrets from GitHub Secrets that backend needs at runtime
ENVEOF
chown deploy:deploy /home/deploy/bastion/.env.prod
chmod 600 /home/deploy/bastion/.env.prod

echo ""
echo "=== Setup complete ==="
echo "1. Edit /home/deploy/bastion/.env.prod with real passwords"
echo "2. Log in as deploy user: ssh deploy@$DOMAIN"
echo "3. GitHub Actions will handle deployment from here"
```
