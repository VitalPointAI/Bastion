**Structure:**

```yaml
name: CI

on:
  push:
    branches: [master]
  pull_request:
    branches: [master]

jobs:
  ci-backend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 10
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "pnpm"
          cache-dependency-path: backend/pnpm-lock.yaml
      - run: pnpm install --frozen-lockfile
        working-directory: backend
      - run: pnpm build
        working-directory: backend

  ci-frontend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 10
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "pnpm"
          cache-dependency-path: frontend/pnpm-lock.yaml
      - run: pnpm install --frozen-lockfile
        working-directory: frontend
      - run: pnpm build
        working-directory: frontend
```

**Notes:**
- Backend and frontend CI jobs run in parallel (both ubuntu-latest, no dependency between them).
- Use `pnpm/action-setup@v4` with version 10. Use `actions/setup-node@v4` with `cache: "pnpm"` and workspace-specific `cache-dependency-path`.
- Use `--frozen-lockfile` for deterministic installs.
- `pnpm build` for both — this compiles TypeScript (backend) and bundles with Vite (frontend), catching type errors and build failures.
- Do NOT add `pnpm test` or `pnpm lint` steps yet unless those scripts exist in package.json. Check `backend/package.json` and `frontend/package.json` for available scripts. If `test` script exists and is not a placeholder (`echo "no test"`), include it. If `lint` script exists, include it. Otherwise omit to avoid CI failures on missing scripts.

Create `.github/workflows/deploy.yml` — runs on push to master only. Two jobs: `build-and-push` (builds Docker images, pushes to GHCR) and `deploy` (SSHs to Hetzner, pulls images, restarts).

**Full workflow structure:**

```yaml
name: Build & Deploy

on:
  push:
    branches: [master]

env:
  REGISTRY: ghcr.io
  IMAGE_PREFIX: ghcr.io/vitalpointai/bastion

jobs:
  build-and-push:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write
    steps:
      - uses: actions/checkout@v4

      - name: Log in to GHCR
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      # Backend image
      - name: Backend metadata
        id: meta-backend
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.IMAGE_PREFIX }}/backend
          tags: |
            type=sha,prefix=
            type=raw,value=latest,enable={{is_default_branch}}

      - name: Build and push backend
        uses: docker/build-push-action@v6
        with:
          context: ./backend
          target: runtime
          push: true
          tags: ${{ steps.meta-backend.outputs.tags }}
          labels: ${{ steps.meta-backend.outputs.labels }}

      # Frontend image
      - name: Frontend metadata
        id: meta-frontend
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.IMAGE_PREFIX }}/frontend
          tags: |
            type=sha,prefix=
            type=raw,value=latest,enable={{is_default_branch}}

      - name: Build and push frontend
        uses: docker/build-push-action@v6
        with:
          context: ./frontend
          target: production
          push: true
          tags: ${{ steps.meta-frontend.outputs.tags }}
          labels: ${{ steps.meta-frontend.outputs.labels }}
          build-args: |
            VITE_PRIVY_APP_ID=${{ secrets.VITE_PRIVY_APP_ID }}
            VITE_NEAR_NETWORK=${{ vars.VITE_NEAR_NETWORK }}
            VITE_NEAR_RPC=${{ vars.VITE_NEAR_RPC }}
            VITE_PINATA_GATEWAY=${{ secrets.VITE_PINATA_GATEWAY }}
            VITE_BACKEND_URL=${{ vars.VITE_BACKEND_URL }}

  deploy:
    needs: build-and-push
    runs-on: ubuntu-latest
    steps:
      - name: Deploy to Hetzner
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.HETZNER_HOST }}
          username: ${{ secrets.HETZNER_USER }}
          key: ${{ secrets.HETZNER_SSH_KEY }}
          port: 22
          script: |
            set -e
            cd /home/deploy/bastion

            # Pull latest code (for docker-compose.prod.yml, database files)
            git pull origin master

            # Log in to GHCR (needed for private repo images)
            echo "${{ secrets.GHCR_TOKEN }}" | docker login ghcr.io -u ${{ secrets.GHCR_USER }} --password-stdin

            # Pull latest images
            docker compose -f docker-compose.prod.yml pull

            # Restart services with new images (never use down -v!)
            docker compose -f docker-compose.prod.yml up -d --remove-orphans

            # Clean up old images
            docker image prune -f

            # Verify backend health
            sleep 15
            curl -f http://localhost:3001/health || exit 1
            echo "Deployment successful"
```

**CRITICAL implementation notes:**

1. **GHCR image paths:** Use `ghcr.io/vitalpointai/bastion/backend` and `ghcr.io/vitalpointai/bastion/frontend`. The `IMAGE_PREFIX` env var should be `ghcr.io/vitalpointai/bastion` (lowercase — GHCR requires lowercase).

2. **Frontend build-args:** ALL `VITE_*` variables MUST be passed as `build-args` because Vite bakes them into the static bundle at build time. The frontend Dockerfile already has matching `ARG` declarations.

3. **Deploy job:** Uses `appleboy/ssh-action@v1`. The script:
   - Does `git pull origin master` to get latest docker-compose.prod.yml and SQL files
   - Logs in to GHCR (needed if repo is private — the `GHCR_TOKEN` and `GHCR_USER` secrets are optional; if repo is public, remove the docker login line)
   - Pulls and restarts. NEVER uses `down -v` (would destroy database volumes)
   - Waits 15 seconds then checks backend /health endpoint

4. **Secrets required (documented for Plan 03 checkpoint):**
   - `HETZNER_HOST` — server IP or domain
   - `HETZNER_USER` — `deploy` (from server-setup.sh)
   - `HETZNER_SSH_KEY` — private key (ed25519, no passphrase)
   - `VITE_PRIVY_APP_ID` — Privy dashboard
   - `VITE_PINATA_GATEWAY` — Pinata gateway URL
   - `GHCR_TOKEN` — PAT with `read:packages` scope (only if repo is private)
   - `GHCR_USER` — GitHub username (only if repo is private)

5. **Variables required (non-secret):**
   - `VITE_NEAR_NETWORK` — `testnet` or `mainnet`
   - `VITE_NEAR_RPC` — `https://rpc.testnet.near.org`
   - `VITE_BACKEND_URL` — `https://your-domain.com` (the production URL)

6. **Backend health check:** The backend Dockerfile and docker-compose.yml both define `/health` endpoint checks. The deploy script verifies this after restart. The curl check hits localhost:3001 which will work only if docker-compose.prod.yml exposes backend port 3001 to localhost. WAIT — the prod compose does NOT expose backend port 3001 to host. The health check should instead use `docker compose -f docker-compose.prod.yml exec backend node -e "fetch('http://localhost:3001/health').then(r => r.ok ? process.exit(0) : process.exit(1)).catch(() => process.exit(1))"` OR check via the frontend proxy: `curl -f https://localhost/api/health` (but TLS on localhost won't work). **Solution:** Use `docker compose -f docker-compose.prod.yml ps --format json` to verify containers are running, then `docker compose -f docker-compose.prod.yml exec -T backend node -e "fetch('http://localhost:3001/health').then(r=>r.ok?process.exit(0):process.exit(1)).catch(()=>process.exit(1))"` for the actual health check. Replace the `curl` line in the deploy script with this approach.
