```typescript
  /**
   * Get funding contract status
   */
  async getFundingStatus(): Promise<FundingStatus> {
    const response = await fetch(`${this.baseUrl}/admin/funding/status`, {
      headers: this.getHeaders(),
    });

    if (!response.ok) {
      throw new Error('Failed to get funding status');
    }

    return response.json();
  }

  /**
   * Get funding activity history
   */
  async getFundingHistory(fromIndex: number = 0, limit: number = 20): Promise<FundingHistory> {
    const response = await fetch(
      `${this.baseUrl}/admin/funding/history?from=${fromIndex}&limit=${limit}`,
      {
        headers: this.getHeaders(),
      }
    );

    if (!response.ok) {
      throw new Error('Failed to get funding history');
    }

    return response.json();
  }

  /**
   * Check if a specific account has been funded
   */
  async checkAccountFunding(accountId: string): Promise<{ enabled: boolean; funded: boolean }> {
    const response = await fetch(`${this.baseUrl}/admin/funding/check/${accountId}`, {
      headers: this.getHeaders(),
    });

    if (!response.ok) {
      throw new Error('Failed to check account funding');
    }

    return response.json();
  }
```

```tsx
/**
 * FundingPanel Component
 *
 * Admin panel for monitoring the NEAR funding contract.
 * Shows contract balance, funding statistics, and recent activity.
 */

import { useState, useEffect, useCallback } from 'react';
import { adminService, FundingStatus, FundingHistoryItem } from '../../lib/admin-service';
import './FundingPanel.css';

// Warning thresholds
const LOW_BALANCE_THRESHOLD = 10; // Warn when < 10 accounts can be funded
const CRITICAL_BALANCE_THRESHOLD = 3; // Critical when < 3 accounts

export function FundingPanel() {
  const [status, setStatus] = useState<FundingStatus | null>(null);
  const [history, setHistory] = useState<FundingHistoryItem[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [checkAccountId, setCheckAccountId] = useState('');
  const [checkResult, setCheckResult] = useState<{ funded: boolean } | null>(null);
  const [isChecking, setIsChecking] = useState(false);

  const loadData = useCallback(async () => {
    setIsLoading(true);
    setError(null);

    try {
      const [statusData, historyData] = await Promise.all([
        adminService.getFundingStatus(),
        adminService.getFundingHistory(0, 20)
      ]);

      setStatus(statusData);
      setHistory(historyData.history);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to load funding data');
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    loadData();
  }, [loadData]);

  const handleCheckAccount = async () => {
    if (!checkAccountId || checkAccountId.length !== 64) {
      setError('Account ID must be 64 characters (hex)');
      return;
    }

    setIsChecking(true);
    setCheckResult(null);

    try {
      const result = await adminService.checkAccountFunding(checkAccountId);
      setCheckResult(result);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to check account');
    } finally {
      setIsChecking(false);
    }
  };

  const getBalanceStatus = () => {
    if (!status || !status.accountsRemaining) return 'unknown';
    if (status.accountsRemaining < CRITICAL_BALANCE_THRESHOLD) return 'critical';
    if (status.accountsRemaining < LOW_BALANCE_THRESHOLD) return 'warning';
    return 'healthy';
  };

  const balanceStatus = getBalanceStatus();

  if (isLoading) {
    return (
      <div className="funding-panel loading">
        <p>Loading funding contract status...</p>
      </div>
    );
  }

  if (!status?.enabled) {
    return (
      <div className="funding-panel disabled">
        <div className="disabled-message">
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
            <circle cx="12" cy="12" r="10" />
            <line x1="12" y1="8" x2="12" y2="12" />
            <line x1="12" y1="16" x2="12.01" y2="16" />
          </svg>
          <h3>Funding Contract Not Configured</h3>
          <p>{status?.message || 'Configure NEAR_FUNDING_CONTRACT_ID in environment variables to enable account funding.'}</p>
        </div>
      </div>
    );
  }

  return (
    <div className="funding-panel">
      {error && (
        <div className="funding-error">
          <span>{error}</span>
          <button onClick={() => setError(null)}>Dismiss</button>
        </div>
      )}

      <div className="funding-header">
        <h2>Funding Contract</h2>
        <button className="refresh-button" onClick={loadData} disabled={isLoading}>
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
            <path d="M23 4v6h-6" />
            <path d="M1 20v-6h6" />
            <path d="M3.51 9a9 9 0 0 1 14.85-3.36L23 10M1 14l4.64 4.36A9 9 0 0 0 20.49 15" />
          </svg>
          Refresh
        </button>
      </div>

      <div className="contract-info">
        <span className="contract-label">Contract ID:</span>
        <code className="contract-id">{status.contractId}</code>
      </div>

      {/* Balance Warning Banner */}
      {balanceStatus !== 'healthy' && (
        <div className={`balance-warning ${balanceStatus}`}>
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
            <path d="M10.29 3.86L1.82 18a2 2 0 0 0 1.71 3h16.94a2 2 0 0 0 1.71-3L13.71 3.86a2 2 0 0 0-3.42 0z" />
            <line x1="12" y1="9" x2="12" y2="13" />
            <line x1="12" y1="17" x2="12.01" y2="17" />
          </svg>
          <span>
            {balanceStatus === 'critical'
              ? 'CRITICAL: Contract nearly empty! Top up immediately.'
              : `Low balance: Only ${status.accountsRemaining} accounts can be funded.`}
          </span>
        </div>
      )}

      {/* Stats Grid */}
      <div className="funding-stats">
        <div className={`stat-card balance ${balanceStatus}`}>
          <div className="stat-label">Available Balance</div>
          <div className="stat-value">{status.availableBalance} NEAR</div>
          <div className="stat-sublabel">({status.accountsRemaining} accounts)</div>
        </div>

        <div className="stat-card">
          <div className="stat-label">Total Balance</div>
          <div className="stat-value">{status.balance} NEAR</div>
        </div>

        <div className="stat-card">
          <div className="stat-label">Funding Amount</div>
          <div className="stat-value">{status.fundingAmountPerAccount} NEAR</div>
          <div className="stat-sublabel">per account</div>
        </div>

        <div className="stat-card">
          <div className="stat-label">Total Funded</div>
          <div className="stat-value">{status.totalAccountsFunded}</div>
          <div className="stat-sublabel">accounts activated</div>
        </div>
      </div>

      {/* Account Check Tool */}
      <div className="account-check-section">
        <h3>Check Account Funding</h3>
        <div className="account-check-form">
          <input
            type="text"
            placeholder="Enter 64-character account ID"
            value={checkAccountId}
            onChange={(e) => setCheckAccountId(e.target.value)}
            maxLength={64}
          />
          <button onClick={handleCheckAccount} disabled={isChecking}>
            {isChecking ? 'Checking...' : 'Check'}
          </button>
        </div>
        {checkResult && (
          <div className={`check-result ${checkResult.funded ? 'funded' : 'not-funded'}`}>
            {checkResult.funded
              ? 'Account has been funded'
              : 'Account has NOT been funded'}
          </div>
        )}
      </div>

      {/* Recent Activity */}
      <div className="activity-section">
        <h3>Recent Funding Activity</h3>
        {history.length === 0 ? (
          <p className="no-activity">No funding activity recorded yet.</p>
        ) : (
          <div className="activity-table-wrapper">
            <table className="activity-table">
              <thead>
                <tr>
                  <th>Account ID</th>
                  <th>Amount</th>
                  <th>Time</th>
                  <th>Block</th>
                </tr>
              </thead>
              <tbody>
                {history.map((item, index) => (
                  <tr key={index}>
                    <td className="account-cell" title={item.accountId}>
                      {item.accountId.substring(0, 8)}...{item.accountId.substring(56)}
                    </td>
                    <td>{item.amount}</td>
                    <td>{new Date(item.timestamp).toLocaleString()}</td>
                    <td>{item.blockHeight}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>

      {/* Top-up Instructions */}
      <div className="topup-section">
        <h3>Top Up Contract</h3>
        <p>To add NEAR to the funding contract, use the NEAR CLI:</p>
        <code className="topup-command">
          near send YOUR_ACCOUNT.testnet {status.contractId} AMOUNT
        </code>
        <p className="topup-note">
          Example: <code>near send admin.testnet {status.contractId} 10</code> to add 10 NEAR
        </p>
      </div>
    </div>
  );
}
```
